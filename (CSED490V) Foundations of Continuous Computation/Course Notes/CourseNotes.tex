\section{Computable Reals}

\begin{definition}[Borel Computability]
A real number $r$ is \emph{Borel-computable} if there exists a computable
sequence $(q_n)_{n\in\Nzero}\subseteq\Q$ such that
\[
  |r-q_n|<2^{-n}\qquad(n\in\Nzero).
\]
\end{definition}

\begin{definition}[Turing Computability]
A nonnegative real number $r$ is \emph{Turing-computable} if it has a binary
expansion
\[
 r=a+\sum_{i=0}^{\infty}b_i2^{-(i+1)},
 \qquad a\in\Nzero,\quad b_i\in\{0,1\},
\]
for which $i\mapsto b_i$ is computable. A negative real number is
Turing-computable when its absolute value is Turing-computable. For dyadic
rationals we may use the terminating binary expansion.
\end{definition}

\begin{theorem}[Equivalence of Computability Definitions]
For $r\in\R$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
 \item $r$ is Borel-computable.
 \item $r$ is Turing-computable.
\end{enumerate}
\end{theorem}
\begin{proof}
We may consider $r\ge0$: changing the sign of a computable rational
approximation is effective, and the negative case of binary computability is
defined using the absolute value.

Suppose first that $r$ is Turing-computable, with integer part $a$ and
computable bits $b_i$. Set
\[
 q_n=a+\sum_{i=0}^{n}b_i2^{-(i+1)}\in\Q.
\]
The sum is computable and
\[
 0\le r-q_n\le\sum_{i=n+1}^{\infty}2^{-(i+1)}
   =2^{-(n+1)}<2^{-n}.
\]
Thus $r$ is Borel-computable.

Conversely, suppose $(q_m)_{m\in\Nzero}$ computes $r$ with
$|q_m-r|<2^{-m}$. Fix the integer part $a=\lfloor r\rfloor$ as part of the
existential description of a binary expansion. We distinguish two cases.

If $r$ is dyadic, it admits an eventually zero binary expansion. Its finite
sequence of nonzero digits can be built into an algorithm, and hence $r$ is
Turing-computable.

Now assume $r$ is not dyadic. Fix $n\in\Nzero$. The non-dyadic number $r$
lies strictly between two consecutive grid points of mesh $2^{-(n+1)}$.
Search over $m\in\Nzero$ until the rational interval
\[
  I_m=(q_m-2^{-m},\,q_m+2^{-m})
\]
is contained strictly in a single grid cell
\[
 \left(\frac{k}{2^{n+1}},\frac{k+1}{2^{n+1}}\right),
 \qquad k\in\Nzero.
\]
Containment is decidable by rational arithmetic: for a given $m$ there is
at most one candidate $k$, which can be found by integer division.
The search terminates, since $r$ has positive distance from both endpoints
of its grid cell and $q_m\to r$ with an effective error bound.
Moreover $r\in I_m$, so the returned $k$ satisfies
$k=\lfloor2^{n+1}r\rfloor$. The $n$th binary digit of $r$ is therefore
\[
 b_n=k\bmod 2.
\]
This procedure computes each bit, completing the proof.
\end{proof}

\begin{remark}
The two definitions characterize the same set of real numbers, but their
representations do not admit symmetric \emph{uniform} conversions. Given a
computable binary expansion one can effectively obtain rational approximations.
In general there is no algorithm that takes an arbitrary rational-approximation
procedure and uniformly produces its binary digits: the latter operation must
resolve boundary cases at dyadic rationals. This does not contradict the
existence of a computable binary expansion for each Borel-computable real.
\end{remark}

\begin{definition}[Computable Real Numbers]
The set of \emph{computable real numbers} is
\[
  \Rc\coloneqq\{r\in\R\mid r\text{ is computable}\}.
\]
By the preceding theorem, either notion of computability can be used.
\end{definition}

\begin{proposition}[Closure under Arithmetic]
The set $\Rc$ is a subfield of $\R$. In particular,
\[
 x,y\in\Rc\implies x+y,\ x-y,\ xy\in\Rc,
 \qquad y\ne0\implies\frac{x}{y}\in\Rc.
\]
\end{proposition}

\begin{proposition}[Real Algebraic Closure]
If $p\in\Rc[X]$ is nonzero, then every real root of $p$ lies in $\Rc$.
In particular, $\Rc$ is a real-closed field.
\end{proposition}

\begin{proposition}[Computability of $\pi$ and $e$]
The real numbers $\pi$ and $e$ are computable.
\end{proposition}
\begin{proof}
The alternating arctangent series gives
\[
 \frac{\pi}{4}=\arctan 1=\sum_{k=0}^{\infty}\frac{(-1)^k}{2k+1}.
\]
For $m\in\Nzero$ put
\[
 p_m=4\sum_{k=0}^{m}\frac{(-1)^k}{2k+1}\in\Q.
\]
The alternating-series estimate yields
\[
 |\pi-p_m|\le\frac{4}{2m+3}.
\]
Choosing $m=2^{n+1}$ gives
$|\pi-p_m|<2^{-n}$ for every $n\in\Nzero$. The finite sum $p_m$ is
computable from $n$.

The computability of $e=\exp(1)$ follows from the computability of
$\exp\colon\R\to\R$ proved in Theorem~\ref{thm:exp}.
\end{proof}

\begin{proposition}[Countability of the Computable Reals]
The set $\Rc$ is countably infinite. Consequently, there exist
noncomputable real numbers.
\end{proposition}
\begin{proof}
There are countably many finite descriptions of algorithms, and every
computable real has at least one such description. Hence $\Rc$ is at most
countable. On the other hand, every rational number is computable via its
constant approximation sequence, so $\Q\subseteq\Rc$ and $\Rc$ is infinite.
Since $\R$ is uncountable, $\R\setminus\Rc\ne\varnothing$.
\end{proof}

\begin{fact}[Hilbert's Tenth Problem]
There is no algorithm that decides, for every multivariate polynomial with
integer coefficients, whether it has an integer zero. In particular, the set
of integer polynomials admitting an integer solution is not decidable.
\end{fact}

\begin{proposition}[A Noncomputable Real]
Let $P$ be the set of all integer-coefficient polynomials in finitely
many variables, and let
$\phi\colon P\to\N$ be a computable injection.
Write $P_{\mathrm{root}}$ for the polynomials having an integer zero.
Then
\[
 r\coloneqq\sum_{p\in P}
 2^{-\phi(p)}\,\indicator{P_{\mathrm{root}}}(p)
\]
is a noncomputable real in $[0,1]$.
\end{proposition}
\begin{proof}
Injectivity gives the binary coefficient at $2^{-\phi(p)}$:
\[
 b_{\phi(p)}=\indicator{P_{\mathrm{root}}}(p).
\]
Both $P_{\mathrm{root}}$ and $P\setminus P_{\mathrm{root}}$ are infinite,
so the expansion has infinitely many zeros and ones; in particular,
$r$ is not dyadic. If $r\in\Rc$, its unique binary expansion is
computable by the equivalence theorem. Hence
\[
 p\longmapsto b_{\phi(p)}=\indicator{P_{\mathrm{root}}}(p)
\]
would decide Hilbert's Tenth Problem, a contradiction.
\end{proof}

\section{Computable Functions}

\begin{definition}[Computable Function]
Let $D\subseteq\R$. A function $f\colon D\to\R$ is \emph{computable} if
there exists an oracle algorithm $\A$ with the following property. For every
$x\in D$, every function $\alpha\colon\Nzero\to\Q$ satisfying
\[
 |\alpha(i)-x|<2^{-i}\qquad(i\in\Nzero),
\]
and every precision $n\in\Nzero$, the algorithm $\A$, given access to
$\alpha$ and the input $n$, halts and returns a rational number
$\A(\alpha,n)$ such that
\[
 |\A(\alpha,n)-f(x)|<2^{-n}.
\]
The algorithm may query finitely many values $\alpha(i)$ during a computation.
The definition extends componentwise to functions between subsets of
$\R^k$ and $\R^m$.
\end{definition}

\begin{remark}
A name $\alpha$ need not be computable: $\A$ must work for
\emph{every} rational name, even one naming a noncomputable real.
Unlike a binary-digit oracle, rational approximation does not require
a uniform decision at dyadic boundaries (cf. Section~1).
\end{remark}

In particular, every computable $f\colon D\to\R$ satisfies
$f(D\cap\Rc)\subseteq\Rc$.

\begin{example}[Basic Computable Functions]
For $r\in\Rc$, both $x\mapsto x+r$ and $x\mapsto rx$ are computable.
Other examples are $|x|$, $e^x$, $\sin x$, $\sqrt{x}$ (for $x\ge0$),
and $\log x$ (for $x>0$).

In several variables, $+$, $-$, multiplication, $\min$, and $\max$
are computable; division is computable when the divisor is nonzero.
\end{example}

\begin{theorem}[Closure under Composition]
Let $f\colon D\to E$ and $g\colon E\to\R^m$ be computable, where
$D\subseteq\R^k$ and $E\subseteq\R^\ell$. Then $g\circ f$ is computable.
\end{theorem}

\begin{corollary}[Computable Function Operations]
If $f,g\colon D\to\R$ are computable, then so are
\[
 |f|,\quad f+g,\quad fg,\quad\min(f,g),\quad\max(f,g),
 \quad\exp\circ f.
\]
Moreover, $f/g$ is computable on $\{x\in D\mid g(x)\ne0\}$.
\end{corollary}

\begin{theorem}[Computability Implies Continuity]
Every computable function $f\colon D\to\R^m$, with
$D\subseteq\R^k$, is continuous on $D$ (with the relative topology).
\end{theorem}
\begin{proof}
Fix $\mathbf{x}\in D$ and $\varepsilon>0$; choose $n\in\Nzero$
with $2^{1-n}<\varepsilon$. Choose a rational name
$\alpha\colon\Nzero\to\Q^k$ of $\mathbf{x}$ with slack:
\[
\|\alpha(i)-\mathbf{x}\|_\infty<2^{-(i+2)}
\qquad(i\in\Nzero).
\]
At precision $n$, the algorithm queries a finite set of indices $I$.
For some $\delta>0$, every $\mathbf{y}\in D$ with
$\|\mathbf{x}-\mathbf{y}\|_\infty<\delta$ also admits a name agreeing
with $\alpha$ on $I$. The algorithm therefore returns the
same $\mathbf{q}\in\Q^m$ on both names, whence
\[
\|f(\mathbf{x})-f(\mathbf{y})\|_\infty
\le\|f(\mathbf{x})-\mathbf{q}\|_\infty
+\|\mathbf{q}-f(\mathbf{y})\|_\infty
<2^{1-n}<\varepsilon.
\]
Thus $f$ is continuous at $\mathbf{x}$.
\end{proof}

\begin{remark}
Continuity is necessary, not sufficient, for computability: if
$c\in\R\setminus\Rc$, the constant function $x\mapsto c$ is continuous
but not computable. In particular, a function with a discontinuity at a
point of its domain cannot be computable there.
\end{remark}

\begin{example}[Discontinuous Functions and Their Domains]
The step function $x\mapsto\indicator{\{x>0\}}$ is not computable on
$\R$ but is computable on $\R\setminus\{0\}$: refine an input
approximation until its sign is certified. The same principle applies
to strict comparison on $x\ne y$.

A set-valued extension, outside the single-valued definition above, can
instead allow both outputs at the discontinuity:
\[
\operatorname{sgn}'(x)=
\begin{cases}
\{1\},&x>0,\\
\{0\},&x<0,\\
\{0,1\},&x=0.
\end{cases}
\]
\end{example}

\begin{theorem}[Computability of the Exponential Function]\label{thm:exp}
The function $\exp\colon\R\to\R$ is computable.
\end{theorem}
\begin{proof}
First let $x\in[-\tfrac12,\tfrac12]$, let $\alpha$ be a rational name
of $x$, and fix $n\in\Nzero$. Set $m=d=n+3$ and
\[
 \widehat{x}=\alpha(m),\qquad
 \widehat{y}=P_d(\widehat{x}),\qquad
 P_d(t)=\sum_{j=0}^{d}\frac{t^j}{j!}.
\]
The rational $\widehat{y}$ is computed by finite arithmetic. Since
$|x-\widehat{x}|<2^{-m}$ and $|\widehat{x}|<1$, the mean value theorem
and $e<3$ give
\[
 |e^x-e^{\widehat{x}}|<3\cdot2^{-m}.
\]
For $d\ge3$, the Taylor remainder satisfies
\[
 |e^{\widehat{x}}-P_d(\widehat{x})|
 \le\sum_{j=d+1}^{\infty}\frac{1}{j!}
 \le\sum_{j=d+1}^{\infty}2^{-j}=2^{-d}.
\]
Consequently,
\[
 |e^x-\widehat{y}|<(3+1)2^{-(n+3)}<2^{-n}.
\]

For arbitrary $x\in\R$, one can effectively choose $k\in\Nzero$ with
$|x|/2^k\le\tfrac12$. Write $u=e^{x/2^k}$ and $N=2^k$.
The preceding algorithm computes $u\in(0,2)$ to arbitrary precision.
For any rational $v$ with $|u-v|<1$,
\[
 |u^N-v^N|\le N3^{N-1}|u-v|.
\]
Thus a sufficiently accurate rational approximation $v$ gives the
required approximation $v^N$ to $e^x=u^N$.
\end{proof}

\begin{theorem}[Noncomputability of Strict Comparison]
The strict-comparison map $\R^2\to\{0,1\}$ defined by
\[
 (x,y)\mapsto\begin{cases}1,&x<y,\\0,&x\ge y\end{cases}
\]
is not computable.
\end{theorem}
\begin{proof}
Suppose an algorithm computes strict comparison. Because its values lie in $\{0,1\}$,
requesting an approximation with error below $1/4$ and testing whether the
result exceeds $1/2$ yields an algorithm returning the exact bit.
Give this decision algorithm the two names
\[
 \alpha(i)=0,\qquad \beta(i)=2^{-(i+1)}\qquad(i\in\Nzero),
\]
both of which represent the real number $0$. Hence the algorithm must
halt and output $0$. During this run, only finitely many indices of
$\beta$ are queried; let $N\in\Nzero$ exceed or equal all of them.
Define another rational name by
\[
 \beta'(i)=\begin{cases}
    2^{-(i+1)},&i\le N,\\
    2^{-(N+1)},&i>N.
 \end{cases}
\]
For $i\le N$, its distance from $2^{-(N+1)}$ is strictly smaller than
$2^{-i}$, and for $i>N$ the distance is zero. Thus $\beta'$ is a valid
name of $y'=2^{-(N+1)}>0$.
The algorithm cannot distinguish $(\alpha,\beta')$ from
$(\alpha,\beta)$ during its finite computation; it still returns $0$.
But strict comparison at $(0,y')$ equals $1$, a contradiction.
\end{proof}

\begin{theorem}[Comparison of Unequal Real Numbers]
Strict comparison is computable on
\[
\{(x,y)\in\R^2\mid x\ne y\}.
\]
\end{theorem}

\begin{remark}
If $x\ne y$, sufficiently accurate rational approximations eventually
certify their order. Without this promise, neither equality nor strict
comparison is decidable on all of $\R^2$: their indicator maps are
discontinuous along $x=y$.
\end{remark}

\section{Floating-Point Arithmetic}

\begin{definition}[Floating-Point Representation]
Fix $E\ge2$ and $M\ge1$. A floating-point bit string consists of a sign
bit $s$, $E$ exponent bits, and $M$ fraction bits. Let
\[
 k=(e_1\cdots e_E)_2,\qquad
 t=(0.m_1\cdots m_M)_2=\sum_{j=1}^M m_j2^{-j},
 \qquad b_E=2^{E-1}-1.
\]
The interpretation map $\gamma_{E,M}\colon\{0,1\}^{1+E+M}\to
\R\cup\{-\infty,+\infty,\mathrm{NaN}\}$ is defined by
\[
\gamma_{E,M}(s,e_1,\ldots,e_E,m_1,\ldots,m_M)=
\begin{cases}
 (-1)^s2^{k-b_E}(1+t),&1\le k\le2^E-2,\\
 (-1)^s2^{1-b_E}t,&k=0,\\
 (-1)^s\infty,&k=2^E-1,\ t=0,\\
 \mathrm{NaN},&k=2^E-1,\ t\ne0.
\end{cases}
\]
The first case represents \emph{normal} values. The second represents
\emph{subnormal} values when $t\ne0$ and zero when $t=0$; the
remaining cases represent infinities and \emph{not a number} (NaN).
The normal exponent bounds are
\[
 e_{\min}=1-b_E=2-2^{E-1},\qquad
 e_{\max}=2^E-2-b_E=b_E.
\]
Write
\[
\overline{\mathbb F}_{E,M}=\operatorname{im}\gamma_{E,M},\qquad
\mathbb F_{E,M}=\overline{\mathbb F}_{E,M}
\setminus\{-\infty,+\infty,\mathrm{NaN}\}.
\]
Thus $\mathbb F_{E,M}\subseteq\R$ is the set of finite floating-point values.
\end{definition}

\begin{remark}
The two signed-zero bit strings both represent $0\in\mathbb F_{E,M}$,
while several distinct bit strings represent NaN. The map $\gamma_{E,M}$
identifies these representations as values. In the usual binary32 and
binary64 formats, $(E,M)=(8,23)$ and $(11,52)$, respectively.
\end{remark}

\begin{definition}[Rounding]
Let
\[
 F_{\max}=(2-2^{-M})2^{b_E},\qquad
 K_{E,M}=(2-2^{-M-1})2^{b_E}.
\]
Here $F_{\max}$ is the largest finite floating-point value and
$K_{E,M}$ is the positive overflow threshold. The
\emph{round-to-nearest, ties-to-even} map
$\rnd_{E,M}\colon\R\to\overline{\mathbb F}_{E,M}$ is defined by
\[
\rnd_{E,M}(x)=
\begin{cases}
 -\infty,&x\le-K_{E,M},\\
 \displaystyle\operatorname*{arg\,min}_{y\in\mathbb F_{E,M}}|x-y|,
 &|x|<K_{E,M},\\
 +\infty,&x\ge K_{E,M}.
\end{cases}
\]
If two finite values minimize the distance, choose the one whose least
significant stored significand bit is even. In particular, halfway cases
near $0$ round to $0$; the chosen value always exists and is unique.
\end{definition}

\begin{proposition}[Monotonicity of Rounding]
For $x,y\in\R$,
\[
 x\le y\quad\Longrightarrow\quad
 \rnd_{E,M}(x)\le\rnd_{E,M}(y),
\]
with the usual order on $\R\cup\{-\infty,+\infty\}$.
\end{proposition}

\begin{definition}[Floating-Point Operations]
For $a,b,c\in\mathbb F_{E,M}$, define rounded arithmetic by
\[
\begin{aligned}
 a\oplus b&=\rnd_{E,M}(a+b),&
 a\ominus b&=\rnd_{E,M}(a-b),\\
 a\otimes b&=\rnd_{E,M}(ab),&
 a\odiv b&=\rnd_{E,M}(a/b)\quad(b\ne0).
\end{aligned}
\]
The secondary notations are $\odot$ for multiplication and $\oslash$
for division. \emph{Fused multiply-add} performs only one rounding:
\[
 \fma(a,b,c)=\rnd_{E,M}(ab+c).
\]
These operations have values in $\overline{\mathbb F}_{E,M}$;
operations involving infinities, NaN, or signed-zero bit patterns are
outside the present finite-input definitions.
\end{definition}

\begin{remark}[Algebraic Properties]
Rounded addition and multiplication are commutative, with identities
$0$ and $1$, but generally neither associative nor distributive. For
$(E,M)=(3,2)$, ties-to-even gives
\[
\begin{aligned}
 (1\oplus\tfrac18)\oplus(-1)&=0
     \ne\tfrac18=1\oplus(\tfrac18\oplus(-1)),\\
 2\otimes(\tfrac34\otimes\tfrac18)&=\tfrac14
     \ne\tfrac3{16}=(2\otimes\tfrac34)\otimes\tfrac18,\\
 3\otimes(1\oplus\tfrac32)&=8
     \ne7=(3\otimes1)\oplus(3\otimes\tfrac32).
\end{aligned}
\]
Every $a\in\mathbb F_{E,M}$ has the unique additive inverse $-a$,
since $a\oplus b=0\iff b=-a$ (all finite values are integer
multiples of the smallest subnormal spacing). Multiplicative inverses
need not exist or be unique. In the same format,
\[
 \tfrac32\otimes\tfrac58=1=\tfrac32\otimes\tfrac34,
 \qquad \tfrac1{16}\otimes b\ne1\quad(b\in\mathbb F_{3,2}).
\]
Moreover, rounded operations may overflow to $\pm\infty$, outside
$\mathbb F_{E,M}$; hence they do not form a field.
\end{remark}

\section{Floating-Point Errors}

Fix the format $(E,M)$ of Section 3. In this section write
$\mathbb F=\mathbb F_{E,M}$ and $\rnd=\rnd_{E,M}$, and let
$F_{\max}$, $e_{\min}$, and $e_{\max}$ retain their earlier meanings.
For $x\in\R$, call $x$ in the \emph{normal range} when
$2^{e_{\min}}\le |x|\le F_{\max}$ and in the \emph{subnormal range}
when $|x|<2^{e_{\min}}$.

\begin{definition}[Absolute and Relative Errors]
For an exact value $x\in\R$ and an approximation $\widehat{x}\in\mathbb F$,
set
\[
\begin{aligned}
 \operatorname{err}_{\mathrm{abs}}(x,\widehat{x})
   &\coloneqq |x-\widehat{x}|,\\
 \operatorname{err}_{\mathrm{rel}}(x,\widehat{x})
   &\coloneqq \frac{|x-\widehat{x}|}{|x|}\qquad(x\ne0).
\end{aligned}
\]
\end{definition}

\begin{definition}[Unit in the Last Place]
The \emph{unit in the last place} is the function
$\ulp\colon\R\to\R_{>0}$ defined by
\[
\ulp(x)\coloneqq
\begin{cases}
 2^{e-M},&2^e\le|x|<2^{e+1}
       \text{ for some integer }e\ge e_{\min},\\
 2^{e_{\min}-M},&|x|<2^{e_{\min}}.
\end{cases}
\]
The \emph{ULP error} of $\widehat{x}\in\mathbb F$ is
\[
 \operatorname{err}_{\mathrm{ulp}}(x,\widehat{x})
   \coloneqq\frac{|x-\widehat{x}|}{\ulp(x)}.
\]
Thus ULP error measures the discrepancy in units of the local
floating-point spacing (within a common binade). The
\emph{machine epsilon} (unit roundoff) of the format is
\[
 \varepsilon\coloneqq\tfrac12\ulp(1)=2^{-M-1}.
\]
\end{definition}

\begin{proposition}[Relative and ULP Errors]
Let $x\in\R\setminus\{0\}$ and $\widehat{x}\in\mathbb F$.
Then
\[
 \operatorname{err}_{\mathrm{ulp}}(x,\widehat{x})
 \le\frac{\operatorname{err}_{\mathrm{rel}}(x,\widehat{x})}
 {\varepsilon}.
\]
If $x$ is in the normal range, then also
\[
 \operatorname{err}_{\mathrm{ulp}}(x,\widehat{x})
 \ge\frac{\operatorname{err}_{\mathrm{rel}}(x,\widehat{x})}
 {2\varepsilon}.
\]
\end{proposition}
\begin{proof}
The definitions give, without dividing by an error that might vanish,
\[
 \operatorname{err}_{\mathrm{ulp}}(x,\widehat{x})
 =\operatorname{err}_{\mathrm{rel}}(x,\widehat{x})
   \frac{|x|}{\ulp(x)}.
\]
For $2^e\le|x|<2^{e+1}$ with $e\ge e_{\min}$,
\[
 2^M\le\frac{|x|}{\ulp(x)}<2^{M+1}=\frac1\varepsilon.
\]
In the subnormal range, the same ratio is at most $2^M$;
this proves the first inequality. In the normal range its lower
bound is $2^M=1/(2\varepsilon)$, proving the second.

The normal-range hypothesis cannot be dropped. Indeed, for
$x=2^{e_{\min}-M}$ and $\widehat{x}=0$, both the relative and ULP
errors equal $1$, but $1/(2\varepsilon)=2^M>1$.
\end{proof}

\begin{proposition}[Rounding Error Bounds]
For round-to-nearest, ties-to-even, the following estimates hold:
\begin{enumerate}[label=(\roman*)]
 \item If $x$ is in the normal range, then
 \[
 \operatorname{err}_{\mathrm{rel}}(x,\rnd(x))\le\varepsilon.
 \]
 \item If $x$ is in the subnormal range, then
 \[
 \operatorname{err}_{\mathrm{abs}}(x,\rnd(x))
 \le 2^{e_{\min}}\varepsilon.
 \]
 \item If $|x|\le F_{\max}$, then
 \[
 \operatorname{err}_{\mathrm{ulp}}(x,\rnd(x))\le\tfrac12.
 \]
\end{enumerate}
\end{proposition}
\begin{proof}
For $2^e\le|x|<2^{e+1}$ in the normal range, consecutive
floating-point values are separated by at most $2^{e-M}$ locally.
Thus nearest rounding gives
\[
 \frac{|x-\rnd(x)|}{|x|}
 \le\frac{2^{e-M-1}}{2^e}=\varepsilon.
\]
In the subnormal range the spacing is $2^{e_{\min}-M}$, so
\[
 |x-\rnd(x)|\le\tfrac12 2^{e_{\min}-M}
 =2^{e_{\min}}\varepsilon.
\]
Finally, for $|x|\le F_{\max}$, nearest rounding is within half
the local spacing $\ulp(x)$, giving (iii).
\end{proof}

\begin{definition}[Underflow and Overflow]
For an exact nonzero arithmetic result $z$, the \emph{underflow range}
is $0<|z|<2^{e_{\min}}$. A result beyond the finite range satisfies
$|z|>F_{\max}$; under round-to-nearest, actual \emph{overflow to infinity}
occurs when $|z|\ge K_{E,M}$. These two upper thresholds need not
coincide.
\end{definition}

\begin{theorem}[Error of Floating-Point Operations]
Let $a,b\in\mathbb F$ and choose corresponding operations
\[
 \ast\in\{+,-,\times,\div\},\qquad
 \circledast\in\{\oplus,\ominus,\otimes,\odiv\},
\]
with $b\ne0$ for division. If $z=a\ast b$ satisfies
$|z|\le F_{\max}$, then
\[
 a\circledast b=z(1+\delta)+\delta',\qquad
 |\delta|\le\varepsilon,\quad
 |\delta'|\le2^{e_{\min}}\varepsilon,
\]
where one may choose $\delta'=0$ in the normal range and
$\delta=0$ in the subnormal range (including $z=0$).
For addition and subtraction, one may always choose $\delta'=0$.
\end{theorem}
\begin{proof}
By definition, $a\circledast b=\rnd(z)$. In the normal range,
the relative rounding bound yields
$\rnd(z)=z(1+\delta)$ with $|\delta|\le\varepsilon$.
In the subnormal range, the absolute bound yields
$\rnd(z)=z+\delta'$ with
$|\delta'|\le2^{e_{\min}}\varepsilon$.

For $\ast\in\{+,-\}$, every element of $\mathbb F$ is an integer
multiple of $2^{e_{\min}-M}$, so $a\pm b$ is also such a multiple.
If this exact sum or difference is in the subnormal range, it is
representable and rounding is exact. Thus $\delta'=0$ in both ranges.
\end{proof}

\begin{example}[Error in Repeated Multiplication]
Let $a,b,c\in\mathbb F\setminus\{0\}$, and suppose both intermediate
exact products are in the normal range and do not overflow.
By the preceding theorem, for some $|\delta_1|,|\delta_2|\le\varepsilon$,
\[
\begin{aligned}
 (a\otimes b)\otimes c
 &=\bigl(ab(1+\delta_1)c\bigr)(1+\delta_2)\\
 &=abc(1+\delta_1)(1+\delta_2).
\end{aligned}
\]
Consequently, with $x=abc$ and $\widehat{x}=(a\otimes b)\otimes c$,
\[
\begin{aligned}
 \operatorname{err}_{\mathrm{rel}}(x,\widehat{x})
 &\le |\delta_1|+|\delta_2|+|\delta_1\delta_2|
 \le 2\varepsilon+\varepsilon^2,\\
 \operatorname{err}_{\mathrm{ulp}}(x,\widehat{x})
 &\le 2+\varepsilon.
\end{aligned}
\]
To first order the relative error is at most $2\varepsilon$.
\end{example}

\begin{example}[Error in Repeated Addition]
Let $a,b,c\in\mathbb F$ and set $s=a+b$, $u=a+b+c\ne0$.
Assume $|s|, |(a\oplus b)+c|\le F_{\max}$. For rounded addition,
the theorem gives $\delta'=0$ even for subnormal results; hence
\[
\begin{aligned}
 (a\oplus b)\oplus c
 &=\bigl((a+b)(1+\delta_1)+c\bigr)(1+\delta_2)\\
 &=u(1+\delta_2)+s\delta_1(1+\delta_2),
 \qquad |\delta_1|,|\delta_2|\le\varepsilon.
\end{aligned}
\]
Writing $x=u$ and $\widehat{x}=(a\oplus b)\oplus c$, we obtain
\[
\begin{aligned}
 \operatorname{err}_{\mathrm{rel}}(x,\widehat{x})
 &\le\varepsilon\left(1+\frac{|a+b|}{|a+b+c|}\right)
   +\varepsilon^2\frac{|a+b|}{|a+b+c|},\\
 \operatorname{err}_{\mathrm{ulp}}(x,\widehat{x})
 &\le 1+\frac{|a+b|}{|a+b+c|}
       +\varepsilon\frac{|a+b|}{|a+b+c|}.
\end{aligned}
\]
The ratio $|a+b|/|a+b+c|$ can amplify the first rounding error.

For example, when $2^{M+1}\in\mathbb F$, take
$a=1$, $b=2^{M+1}$, and $c=-2^{M+1}$. Ties-to-even yields
\[
 (a\oplus b)\oplus c=0,
 \qquad a+b+c=1,
\]
with relative error $1$ and ULP error $2^M$.
\end{example}

\begin{remark}[Catastrophic Cancellation]
When large, nearly equal quantities cancel, an error previously
negligible relative to the intermediate values can dominate the
final result. The preceding example loses the small term $a$ in
the first rounded addition.

The same phenomenon can reveal \emph{pre-existing input errors}.
If $\widehat{x},\widehat{y}\in\mathbb F$ approximate $x,y\in\R$,
then even when their subtraction is exact,
\[
 |(\widehat{x}-\widehat{y})-(x-y)|
 \le |\widehat{x}-x|+|\widehat{y}-y|.
\]
For $x\ne y$, dividing by $|x-y|$ gives the relative error;
dividing by $\ulp(x-y)$ gives the ULP error. Both can be large
when $x\approx y$, even if the subtraction itself is exact.
\end{remark}


\begin{example}[Stable Evaluation of $a^2-b^2$]
For $a, b\in\mathbb F$, $a^2\ne b^2$, compare
\[
 u=(a\otimes a)\ominus(b\otimes b), \qquad
 v=(a\ominus b)\otimes(a\oplus b).
\]
Assume that the operations do not overflow or underflow and that the
usual relative-error model applies to all intermediate results.
Writing $|\delta_i|\le\varepsilon$ and retaining terms through first
order gives
\[
\begin{aligned}
 u&=(a^2(1+\delta_1)-b^2(1+\delta_2))(1+\delta_3), \\
 \frac{|u-(a^2-b^2)|}{|a^2-b^2|}
 &\le\left(1+\frac{a^2+b^2}{|a^2-b^2|}\right)\varepsilon
   +O(\varepsilon^2), \\
 v&=(a-b)(a+b)(1+\delta_1)(1+\delta_2)(1+\delta_3), \\
 \frac{|v-(a^2-b^2)|}{|a^2-b^2|}
 &\le3\varepsilon+O(\varepsilon^2).
\end{aligned}
\]
For a normal exact result, the corresponding ULP-error bounds are
$1+(a^2+b^2)/|a^2-b^2|+O(\varepsilon)$ and $3+O(\varepsilon)$,
respectively. The first can be arbitrarily large when $|a|\approx|b|$;
the factored expression avoids subtracting two rounded squares.
\end{example}

\begin{remark}
For $r, r'\in\R$ with
$\operatorname{err}_{\mathrm{ulp}}(r, r')\le1$ and finite $\rnd(r')$,
\[
 \operatorname{err}_{\mathrm{ulp}}(r, \rnd(r'))
 \le\operatorname{err}_{\mathrm{ulp}}(r, r')+\tfrac12.
\]
Indeed, if $\ulp(r')\le\ulp(r)$, the triangle inequality and nearest
rounding suffice. Otherwise $r'$ lies across a power-of-two boundary
from $r$. The one-ULP hypothesis forces $r'$ to round back to that
boundary, closer to $r$ than $r'$ was. The estimate thus holds in both
cases.
\end{remark}

\begin{example}[Stable Quadratic Roots]
For $a\ne0$ and $\Delta=b^2-4ac\ge0$, the usual quadratic formula is
\[
 x_{\pm}=\frac{-b\pm\sqrt\Delta}{2a}.
\]
If $b<0$ and $b^2\gg|4ac|$, evaluating
$(-b-\sqrt\Delta)/(2a)$ may suffer cancellation. Rationalization gives
\[
 \frac{-b-\sqrt\Delta}{2a}
   =\frac{2c}{-b+\sqrt\Delta},
\]
provided the denominator is nonzero. The analogous rationalization
applies to the other root when $b>0$. Moreover, forming $b^2$ can
overflow even if the roots are finite. For $b\ne0$, the identity
\[
 \sqrt{b^2-4ac}=|b|\sqrt{1-\frac{4ac}{b^2}}
\]
illustrates a rescaling that avoids directly squaring a large $b$;
its implementation must also guard the intermediate quotient.
When $|4ac/b^2|\ll1$, the Taylor approximation
$\sqrt{1-t}=1-\tfrac12t+O(t^2)$ explains the sensitivity of the
small root. In particular, for $b^2\gg|4ac|$,
\[
 x_-\approx\begin{cases}
 -b/a+c/b,&b>0, \\
 -c/b,&b<0.
 \end{cases}
\]
Higher-order terms follow from
$\sqrt{1-t}=1-\tfrac12t-\tfrac18t^2+O(t^3)$.
\end{example}

\begin{example}[Two Ways to Compute a Mean]
For $a, b\in\mathbb F$ compare
\[
 \widehat m=(a\oplus b)\odiv2,
 \qquad \widehat m'=(a\odiv2)\oplus(b\odiv2).
\]
If $|a+b|\le F_{\max}$, the first is correctly rounded:
\[
 \widehat m=\rnd\!\left(\frac{a+b}{2}\right),
 \qquad \min(a, b)\le\widehat m\le\max(a, b).
\]
It may nevertheless overflow when $|a+b|$ exceeds the finite range,
even though the mean is representable. The second avoids such
intermediate overflow; it is correctly rounded when the inputs are
sufficiently large, and otherwise has ULP error at most $1$.
\end{example}

\begin{theorem}[Correctly Rounded Mean]
Let $C\in[2^{e_{\min}+M+3},2^{e_{\max}}]\cap\mathbb F$. For
$a, b\in\mathbb F$, define
\[
 m(a, b)=\begin{cases}
 (a\oplus b)\odiv2,&|a|<C\ \text{and}\ |b|<C, \\
 (a\odiv2)\oplus(b\odiv2),&\text{otherwise}.
 \end{cases}
\]
Then, assuming the stated floating-point operations are defined,
\[
 m(a, b)=\rnd\!\left(\frac{a+b}{2}\right)
 \qquad(a, b\in\mathbb F).
\]
\end{theorem}

\section{Error-Free Computations}

Fix $\mathbb F=\mathbb F_{E, M}$ and $\rnd=\rnd_{E, M}$ as before.
For $a\in\mathbb F\setminus\{0\}$ let $e_a\in[e_{\min}, e_{\max}]\cap\mathbb Z$
denote its floating-point exponent, taking $e_a=e_{\min}$ for subnormals.
Set $e_0=e_{\min}$ when an exponent is needed for zero.
Write $\Delta_{\min}\coloneqq2^{e_{\min}-M}$ for the least positive
subnormal spacing. All exactness assertions below concern finite
results with the indicated range conditions.

\begin{lemma}[Integer Significands]
A real number $a$ is in $\mathbb F$ if and only if there exist
$m, e\in\mathbb Z$ with
\[
 a=m2^{e-M}, \qquad |m|<2^{M+1}, \qquad
 e_{\min}\le e\le e_{\max}.
\]
In particular, every $a\in\mathbb F$ is an integer multiple of
$\Delta_{\min}$.
\end{lemma}
\begin{proof}
For normal values take $e=e_a$ and let $m$ be the integer encoded
by the $(M+1)$ significand bits; for subnormals take $e=e_{\min}$
and $|m|<2^M$. Conversely, for any displayed pair $(m, e)$, a
significand with at most $M+1$ bits is either normal at a suitable
exponent or subnormal after shifting to $e_{\min}$. If a shift would
exceed $e_{\max}$, the bound $|m|<2^{M+1}$ prevents it.
\end{proof}

\begin{theorem}[Hauser: Exact Addition]
For $a, b\in\mathbb F$,
\[
 |a+b|\le2^{e_{\min}+1}\implies a\oplus b=a+b.
\]
\end{theorem}
\begin{proof}
Both $a$ and $b$ are integer multiples of $\Delta_{\min}$, hence
so is $a+b$. Every such multiple in
$[-2^{e_{\min}+1},2^{e_{\min}+1}]$ is representable. Therefore
$a+b\in\mathbb F$ and rounding does nothing.
\end{proof}

\begin{theorem}[Sterbenz: Exact Subtraction]
For $a, b\in\mathbb F$,
\[
 \frac a2\le b\le2a\implies a\ominus b=a-b.
\]
Equivalently, one may assume $b/2\le a\le2b$.
\end{theorem}
\begin{proof}
The hypothesis forces $a, b\ge0$. The zero case is immediate.
By exchanging $a, b$ and changing the sign if necessary, assume
$a\ge b$, so $b\le a\le2b$.
Write $a=m_a2^{e_a-M}$ and $b=m_b2^{e_b-M}$ by the lemma.
As $a\ge b$, we have $e_a\ge e_b$ (with the convention for
subnormals), and thus
\[
 a-b=(m_a2^{e_a-e_b}-m_b)2^{e_b-M}=m'2^{e_b-M},
 \qquad m'\in\mathbb Z.
\]
Since $0\le a-b\le b$, one has $0\le m'\le m_b<2^{M+1}$,
so $a-b\in\mathbb F$. Thus $a\ominus b=a-b$.
\end{proof}

\begin{proposition}[Exact Scaling and Multiplication]
Suppose $a, b\in\mathbb F\setminus\{0\}$ have
$|a|=(\alpha_0.\alpha_1\dots\alpha_M)_2\,2^{e_a}$ and
$|b|=2^{e_b}$. If $e_a+e_b\in[e_{\min}, e_{\max}]$, then
\[
 a\otimes b=ab.
\]
Similarly, if $e_a-e_b\in[e_{\min}, e_{\max}]$, then
$a\odiv b=a/b$. More generally, a product is exact whenever its
significand has at most $M+1$ significant binary digits and its
exponent lies in the representable range.
\end{proposition}

\begin{remark}
Sterbenz's theorem does not preclude catastrophic cancellation:
subtraction of nearby \emph{represented} floats may be exact, while
an error inherited from earlier rounded operations becomes large
relative to the small difference.
\end{remark}

\begin{theorem}[Dekker: Exact Addition Residual]
For $a, b\in\mathbb F$ with $|a+b|\le F_{\max}$, put
\[
 s=a\oplus b, \qquad r=(a+b)-s.
\]
Then $r\in\mathbb F$. If $e_a\ge e_b$, the algorithm
\[
 \begin{aligned}
 s&=a\oplus b, \\
 b'&=s\ominus a, \\
 z&=b\ominus b'
 \end{aligned}
 \qquad\text{(Fast2Sum)}
\]
returns $(s, z)=(a\oplus b, r)$; in particular, $a+b=s+z$
\emph{exactly}.
\end{theorem}
\begin{proof}[Sketch]
The integer-significand representation shows that the exact sum and
its rounded value have a common binary grid at the smaller operand's
exponent. Their difference has at most $M+1$ significant bits, hence
$r\in\mathbb F$. For Fast2Sum, it suffices to verify
\[
 s-a\in\mathbb F, \qquad b-(s\ominus a)\in\mathbb F.
\]
Under $e_a\ge e_b$ these exactness properties imply
$b'=s-a$ and $z=b-b'=r$. The two auxiliary verifications are the
steps isolated for Homework~2 in the lecture notes.
\end{proof}

\begin{theorem}[FMA Recovery of a Product Residual]
For $a, b\in\mathbb F$ with $|ab|\le F_{\max}$, let
$s=a\otimes b$ and $r=ab-s$. If
\[
 e_a+e_b\ge e_{\min}+M,
\]
then $r\in\mathbb F$ and
\[
 r=\fma(a, b,-s).
\]
\end{theorem}
\begin{remark}
The exponent condition controls whether the low-order bits of the
exact product fit in the floating-point format. The corresponding
failure at the weaker exponent bound $e_a+e_b=e_{\min}+M-1$ is
examined in Homework~2.
\end{remark}

\section{Automatic Error Analysis}

Fix $N\in\N$ input variables and consider expressions generated by
\[
 e::=c\mid x_i\mid -e\mid e+e\mid e\times e,
 \qquad c\in\mathbb F, \quad i\in\{1, \dots, N\}.
\]
Only expressions whose floating-point evaluation does not overflow
are considered. The expression $e$ is syntax; its \emph{semantics}
is a function of the input valuation
$\sigma=(\sigma_1, \dots, \sigma_N)\in\mathbb F^N$.

\begin{definition}[Floating-Point and Real-Valued Semantics]
The \emph{floating-point semantics} and \emph{real-valued semantics}
of an expression $e$ are functions
\[
 \llbracket e\rrbracket\colon\mathbb F^N\to\mathbb F,
 \qquad
 \llbracket e\rrbracket_{\mathrm r}\colon\mathbb F^N\to\mathbb R,
\]
recursively defined by
\[
\begin{aligned}
 \llbracket c\rrbracket(\sigma)&=c,&
 \llbracket x_i\rrbracket(\sigma)&=\sigma_i, \\
 \llbracket-e\rrbracket(\sigma)&=-\llbracket e\rrbracket(\sigma),&
 \llbracket e_1+e_2\rrbracket(\sigma)
   &=\llbracket e_1\rrbracket(\sigma)\oplus\llbracket e_2\rrbracket(\sigma), \\
 \llbracket e_1\times e_2\rrbracket(\sigma)
   &=\llbracket e_1\rrbracket(\sigma)\otimes\llbracket e_2\rrbracket(\sigma).
\end{aligned}
\]
The real-valued semantics uses the same rules with $+$ and $\times$
in place of $\oplus$ and $\otimes$, respectively.
\end{definition}

\begin{example}[Different Semantics]
For $e=(1+x_1)+(-x_1)$, if $2^{M+1}\in\mathbb F$ and
$\sigma_1=2^{M+1}$, then ties-to-even gives
\[
 \llbracket e\rrbracket(\sigma)=0,
 \qquad \llbracket e\rrbracket_{\mathrm r}(\sigma)=1.
\]
Thus the same expression has different floating-point and real-valued
evaluations.
\end{example}

\begin{definition}[Automatic Range and Error Analysis]
Let $l_i, r_i\in\mathbb F$ with $l_i\le r_i$, and fix the set of
admissible input valuations
\[
 \Sigma\coloneqq\prod_{i=1}^N([l_i, r_i]\cap\mathbb F)
 \subseteq\mathbb F^N.
\]
A \emph{sound range analysis} produces intervals
$\operatorname{range}(e)\subseteq\mathbb R$ such that
\[
 \llbracket e\rrbracket(\sigma)\in\operatorname{range}(e)
 \qquad(\forall\sigma\in\Sigma).
\]
A \emph{sound absolute-error analysis} produces
$\operatorname{err}(e)\in\mathbb R_{\ge0}$ such that
\[
 |\llbracket e\rrbracket(\sigma)
 -\llbracket e\rrbracket_{\mathrm r}(\sigma)|
 \le\operatorname{err}(e)
 \qquad(\forall\sigma\in\Sigma).
\]
The dependence on the fixed $\Sigma$ is suppressed in this notation.
\end{definition}

\Needspace{15\baselineskip}
\begin{proposition}[Recursive Range Analysis]
For each subexpression $e$, set
\[
\begin{aligned}
 \operatorname{range}(c)&=[c, c],&
 \operatorname{range}(x_i)&=[l_i, r_i], \\
 \operatorname{range}(-e)&=[-b,-a]
   &&\text{if }\operatorname{range}(e)=[a, b].
\end{aligned}
\]
If $\operatorname{range}(e_i)=[a_i, b_i]$ for $i=1,2$, define
\[
 \operatorname{range}(e_1+e_2)
 \coloneqq[\rnd(a_1+a_2), \rnd(b_1+b_2)].
\]
These rules are sound, assuming no overflow in the analysis itself.
\end{proposition}
\begin{proof}
The constant, variable and negation cases follow directly from the
semantics. By induction,
$\llbracket e_i\rrbracket(\sigma)\in[a_i, b_i]$, so
\[
 a_1+a_2\le\llbracket e_1\rrbracket(\sigma)
  +\llbracket e_2\rrbracket(\sigma)\le b_1+b_2.
\]
Apply monotonicity of $\rnd$ to obtain the asserted enclosure for
$\llbracket e_1+e_2\rrbracket(\sigma)$.
\end{proof}

\begin{proposition}[Recursive Absolute-Error Analysis]
Set
\[
 \operatorname{err}(c)=\operatorname{err}(x_i)=0,
 \qquad \operatorname{err}(-e)=\operatorname{err}(e).
\]
If $\operatorname{range}(e_i)=[a_i, b_i]$ for $i=1,2$, define
\[
\begin{aligned}
 \operatorname{err}(e_1+e_2)\coloneqq{}&
 \operatorname{err}(e_1)+\operatorname{err}(e_2)\\
 &+\frac12\ulp\!\left(\max\{|a_1+a_2|,|b_1+b_2|\}\right).
\end{aligned}
\]
Under the same no-overflow assumptions, these rules are sound.
\end{proposition}
\begin{proof}
Put $u_i=\llbracket e_i\rrbracket(\sigma)$ and
$v_i=\llbracket e_i\rrbracket_{\mathrm r}(\sigma)$. Then
\[
\begin{aligned}
 |\rnd(u_1+u_2)-(v_1+v_2)|
 &\le|\rnd(u_1+u_2)-(u_1+u_2)|
   +|u_1-v_1|+|u_2-v_2|\\
 &\le\tfrac12\ulp(u_1+u_2)
    +\operatorname{err}(e_1)+\operatorname{err}(e_2).
\end{aligned}
\]
Since $u_i\in[a_i, b_i]$,
$|u_1+u_2|\le\max\{|a_1+a_2|,|b_1+b_2|\}$;
$\ulp$ is nondecreasing in absolute value, proving the claim.
The remaining cases are immediate.
\end{proof}

\begin{remark}[Limitations of Interval Analysis]
The rules bound subexpressions independently and need not preserve
dependencies between occurrences of the same variable. For
$e=x_1-x_1$ (shorthand for $x_1+(-x_1)$) and $x_1\in[-1,1]$,
ordinary interval arithmetic yields $[-2,2]$, although both
semantics evaluate to $0$. Likewise, a positive error bound can be
returned even when the actual rounding error vanishes. Refining the
analysis to account for dependencies is a separate problem.
The multiplication and square-root analysis rules discussed as
exercises in the lecture are addressed in Homework~2, not assumed
as proved results here.
\end{remark}
