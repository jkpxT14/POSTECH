\begin{definition}[Floating-Point Expression Syntax]
Fix $N\in\N$ and $\mathbb F=\mathbb F_{E, M}$.
Expressions are generated by the grammar
\[
 e::=c\mid x_i\mid-e\mid e+e\mid e\times e,
 \qquad c\in\mathbb F, \quad 1\le i\le N.
\]
An expression is a syntactic object, not yet a function.
\end{definition}

\begin{definition}[Floating-Point Semantics]
Let $D_e\subseteq\mathbb F^N$ be the set of input valuations
for which every rounded operation in $e$ has a finite result.
For $\sigma=(\sigma_1, \dots, \sigma_N)\in D_e$, the floating-point
semantics $\llbracket e\rrbracket\colon D_e\to\mathbb F$
is given recursively by
\[
\begin{aligned}
 \llbracket c\rrbracket(\sigma)&=c, &
 \llbracket x_i\rrbracket(\sigma)&=\sigma_i, \\
 \llbracket-e\rrbracket(\sigma)&=-\llbracket e\rrbracket(\sigma), &
 \llbracket e_1+e_2\rrbracket(\sigma)
 &=\llbracket e_1\rrbracket(\sigma)\oplus\llbracket e_2\rrbracket(\sigma), \\
 \llbracket e_1\times e_2\rrbracket(\sigma)
 &=\llbracket e_1\rrbracket(\sigma)\otimes\llbracket e_2\rrbracket(\sigma).
\end{aligned}
\]
For constants and variables, $D_e=\mathbb F^N$; the other
constructors restrict the domain to valuations for which their
subexpressions and the indicated rounded operation remain finite.
\end{definition}

\begin{definition}[Real-Valued Semantics]
The function
$\llbracket e\rrbracket_{\mathrm r}\colon\mathbb F^N\to\R$
is defined by the same recursion, replacing $\oplus$ and
$\otimes$ by exact $+$ and $\times$. Both interpretations
use the same input valuation $\sigma$.
\end{definition}

\begin{example}[Different Semantics]
For $e=(1+x_1)+(-x_1)$, suppose $2^{M+1}\in\mathbb F$
and take $\sigma_1=2^{M+1}$. Then
\[
 \llbracket e\rrbracket(\sigma)=0, \qquad
 \llbracket e\rrbracket_{\mathrm r}(\sigma)=1.
\]
\end{example}

\begin{definition}[Sound Range and Error Analysis]
Fix the admissible input set
\[
 \Sigma\coloneqq\prod_{i=1}^N([l_i, r_i]\cap\mathbb F)
 \subseteq D_e,
 \qquad l_i, r_i\in\mathbb F, \quad l_i\le r_i.
\]
The inclusion excludes intermediate overflow for inputs in
$\Sigma$. Assume also that rounding of endpoints used by
the analysis remains finite. A sound \emph{range analysis}
and \emph{absolute-error analysis} assign respectively an
interval $\operatorname{range}(e)$ and a bound
$\operatorname{err}(e)\ge0$ such that
\[
 \begin{aligned}
 \llbracket e\rrbracket(\sigma)&\in\operatorname{range}(e), \\
 |\llbracket e\rrbracket(\sigma)-\llbracket e\rrbracket_{\mathrm r}(\sigma)|
 &\le\operatorname{err}(e)
 \end{aligned}
 \qquad(\forall\sigma\in\Sigma).
\]
The fixed $\Sigma$ is suppressed in the notation.
We compute both bounds recursively from expression syntax.
\end{definition}

\begin{definition}[Recursive Range Bounds]
For constants and variables, set
\[
 \operatorname{range}(c)=[c, c], \qquad
 \operatorname{range}(x_i)=[l_i, r_i].
\]
If $\operatorname{range}(e)=[a, b]$, put
$\operatorname{range}(-e)=[-b, -a]$.
For $\operatorname{range}(e_i)=[a_i, b_i]$, $i=1, 2$, set
\[
 \operatorname{range}(e_1+e_2)
 =[\rnd(a_1+a_2), \rnd(b_1+b_2)].
\]
\end{definition}
Indeed, monotonicity of exact addition and of rounding gives
\[
 a_1+a_2\le
 \llbracket e_1\rrbracket(\sigma)+\llbracket e_2\rrbracket(\sigma)
 \le b_1+b_2,
\]
which proves the recursive enclosure.

\begin{definition}[Recursive Absolute-Error Bounds]
Set
\[
 \operatorname{err}(c)=\operatorname{err}(x_i)=0,
 \qquad\operatorname{err}(-e)=\operatorname{err}(e).
\]
For $\operatorname{range}(e_i)=[a_i, b_i]$, $i=1, 2$, define
\[
 \begin{aligned}
 \operatorname{err}(e_1+e_2)
 ={}&\operatorname{err}(e_1)+\operatorname{err}(e_2)\\
 &+\tfrac12\ulp(\max\{|a_1+a_2|, |b_1+b_2|\}).
 \end{aligned}
\]
\end{definition}
To verify this rule, write
$u_i=\llbracket e_i\rrbracket(\sigma)$ and
$v_i=\llbracket e_i\rrbracket_{\mathrm r}(\sigma)$.
The triangle inequality and half-ULP bound give
\[
 \begin{aligned}
 |\rnd(u_1+u_2)-(v_1+v_2)|
 &\le|\rnd(u_1+u_2)-(u_1+u_2)|\\
 &\quad+|u_1-v_1|+|u_2-v_2|\\
 &\le\tfrac12\ulp(u_1+u_2)
 +\operatorname{err}(e_1)+\operatorname{err}(e_2).
 \end{aligned}
\]
The monotonicity of $\ulp$ in $|x|$ and the range bounds
justify the displayed recursion.

\begin{definition}[Multiplication Bounds]
For $\operatorname{range}(e_i)=[a_i, b_i]$, $i=1, 2$, put
\[
 \begin{aligned}
 p_-&\coloneqq\min\{a_1a_2, a_1b_2, b_1a_2, b_1b_2\}, \\
 p_+&\coloneqq\max\{a_1a_2, a_1b_2, b_1a_2, b_1b_2\}.
 \end{aligned}
\]
Then
\[
 \operatorname{range}(e_1\times e_2)
 =[\rnd(p_-), \rnd(p_+)].
\]
Writing $U_i=\max\{|a_i|, |b_i|\}$ and
$\eta_i=\operatorname{err}(e_i)$, define
\[
 \begin{aligned}
 \operatorname{err}(e_1\times e_2)
 ={}&U_1\eta_2+U_2\eta_1+\eta_1\eta_2\\
 &+\tfrac12\ulp(\max\{|p_-|, |p_+|\}).
 \end{aligned}
\]
\end{definition}
The four endpoint products bound all exact products from
the two intervals; rounding preserves that enclosure.
For $u_i=\llbracket e_i\rrbracket(\sigma)$ and
$v_i=\llbracket e_i\rrbracket_{\mathrm r}(\sigma)$,
\[
 \begin{aligned}
 |\rnd(u_1u_2)-v_1v_2|
 &\le|\rnd(u_1u_2)-u_1u_2|+|u_1u_2-v_1v_2|\\
 &\le\tfrac12\ulp(\max\{|p_-|, |p_+|\})\\
 &\quad+U_1\eta_2+U_2\eta_1+\eta_1\eta_2,
 \end{aligned}
\]
proving soundness of the error rule.

\begin{definition}[Square-Root Extension]
The grammar may be extended by $\sqrt e$ on valuations where
both semantics of $e$ are nonnegative. Define
\[
 \begin{aligned}
 \llbracket\sqrt e\rrbracket(\sigma)
 &=\rnd\!\left(\sqrt{\llbracket e\rrbracket(\sigma)}\right), \\
 \llbracket\sqrt e\rrbracket_{\mathrm r}(\sigma)
 &=\sqrt{\llbracket e\rrbracket_{\mathrm r}(\sigma)}.
 \end{aligned}
\]
If $\operatorname{range}(e)=[a, b]\subseteq[0, \max\mathbb F]$,
set
\[
 \begin{aligned}
 \operatorname{range}(\sqrt e)
 &=[\rnd(\sqrt a), \rnd(\sqrt b)], \\
 \operatorname{err}(\sqrt e)
 &=\sqrt{\operatorname{err}(e)}+\tfrac12\ulp(\sqrt b).
 \end{aligned}
\]
\end{definition}
Range soundness follows from the monotonicity of square root and
rounding. For $u, v\ge0$,
\[
 |\sqrt u-\sqrt v|\le\sqrt{|u-v|}.
\]
Together with the triangle inequality and half-ULP rounding,
this proves the error rule.

\begin{remark}[Limitations of Interval Analysis]
Independent interval bounds lose correlations between repeated
occurrences of a variable. For $e=x_1-x_1$, understood as
$x_1+(-x_1)$, and $x_1\in[-1, 1]$, interval propagation gives
$[-2, 2]$, although both semantics are identically zero.
Soundness does not imply sharpness: an error upper bound
may be positive when the actual rounding error vanishes.
\end{remark}
