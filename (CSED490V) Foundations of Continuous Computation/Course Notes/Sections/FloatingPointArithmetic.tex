\section{Floating-Point Arithmetic}

\begin{definition}[Floating-Point Representation]
Fix $E\ge2$ and $M\ge1$. A floating-point bit string has a sign
bit $s$, $E$ exponent bits, and $M$ fraction bits. Write
\[
 k=(e_1\cdots e_E)_2, \qquad
 t=(0.m_1\cdots m_M)_2=\sum_{j=1}^Mm_j2^{-j},
 \qquad b_E=2^{E-1}-1.
\]
The representation map
$\gamma_{E, M}\colon\{0, 1\}^{1+E+M}\to
\R\cup\{-\infty, +\infty, \mathrm{NaN}\}$ is defined by
\[
 \gamma_{E, M}(s, e_1, \ldots, e_E, m_1, \ldots, m_M)=
 \begin{cases}
 (-1)^s2^{k-b_E}(1+t), &1\le k\le2^E-2, \\
 (-1)^s2^{1-b_E}t, &k=0, \\
 (-1)^s\infty, &k=2^E-1, \ t=0, \\
 \mathrm{NaN}, &k=2^E-1, \ t\ne0.
 \end{cases}
\]
The first case gives \emph{normal} values; the second gives
\emph{subnormal} values when $t\ne0$, and zero when $t=0$.
The remaining cases represent infinities and NaN. The normal
exponent bounds are
\[
 e_{\min}=1-b_E=2-2^{E-1}, \qquad
 e_{\max}=2^E-2-b_E=b_E.
\]
Define
\[
 \overline{\mathbb F}_{E, M}=\operatorname{im}\gamma_{E, M},
 \qquad
 \mathbb F_{E, M}=\overline{\mathbb F}_{E, M}
 \setminus\{-\infty, +\infty, \mathrm{NaN}\}.
\]
Thus $\mathbb F_{E, M}\subseteq\R$ consists of finite values.
\end{definition}

\begin{remark}
The bit strings for $+0$ and $-0$ both map to the real value $0$,
whereas several distinct bit strings represent NaN. Thus
$\gamma_{E, M}$ identifies bit strings that have the same
value in this abstract model. IEEE binary32 and binary64 have
$(E, M)=(8, 23)$ and $(11, 52)$, respectively; signed-zero behavior
in actual IEEE arithmetic is not modeled by this value map.
\end{remark}

\begin{definition}[Rounding]
The largest finite floating-point value and the positive overflow
threshold are, respectively,
\[
 \max\mathbb F_{E, M}=(2-2^{-M})2^{e_{\max}}, \qquad
 K_{E, M}=(2-2^{-M-1})2^{e_{\max}}.
\]
We also write $F_{\max}=\max\mathbb F_{E, M}$ when useful.
Fix \emph{round to nearest, ties to even} throughout the notes.
The rounding map
$\rnd_{E, M}\colon\R\to\overline{\mathbb F}_{E, M}$ is
\[
 \rnd_{E, M}(x)=
 \begin{cases}
 -\infty, &x\le-K_{E, M}, \\
 \displaystyle\operatorname*{arg\, min}_{y\in\mathbb F_{E, M}}
 |x-y|, &|x|<K_{E, M}, \\
 +\infty, &x\ge K_{E, M}.
 \end{cases}
\]
At equal distances choose the value with an even least significant
stored significand bit. In particular, a tie near zero is rounded
to zero. The selected result is unique.
\end{definition}

\begin{proposition}[Monotonicity of Rounding]
For $x, y\in\R$,
\[
 x\le y\implies\rnd_{E, M}(x)\le\rnd_{E, M}(y)
\]
with the usual order on $\R\cup\{-\infty, +\infty\}$.
\end{proposition}
Strict monotonicity fails: distinct real values can round to the
same floating-point value.

\begin{definition}[Floating-Point Operations]
For $a, b, c\in\mathbb F_{E, M}$, define
\[
 \begin{aligned}
 a\oplus b&=\rnd_{E, M}(a+b), &
 a\ominus b&=\rnd_{E, M}(a-b), \\
 a\otimes b&=\rnd_{E, M}(ab), &
 a\odiv b&=\rnd_{E, M}(a/b)\quad(b\ne0).
 \end{aligned}
\]
Secondary notations are $\odot$ for multiplication and
$\oslash$ for division. Fused multiply-add rounds only once:
\[
 \fma(a, b, c)=\rnd_{E, M}(ab+c).
\]
These operations take values in $\overline{\mathbb F}_{E, M}$;
operations on infinite values, NaN, or signed-zero bit patterns
are outside the present finite-input model.
\end{definition}

\begin{remark}[Algebraic Properties]
On finite inputs, rounded addition and multiplication are
commutative and have respective identities $0$ and $1$ whenever
the operations remain finite. Neither operation is generally
associative, and multiplication need not distribute over addition.
For $(E, M)=(3, 2)$, ties-to-even gives
\[
 \begin{aligned}
 (1\oplus\tfrac18)\oplus(-1)&=0
   \ne\tfrac18=1\oplus(\tfrac18\oplus(-1)), \\
 2\otimes(\tfrac34\otimes\tfrac18)&=\tfrac14
   \ne\tfrac3{16}=(2\otimes\tfrac34)\otimes\tfrac18, \\
 3\otimes(1\oplus\tfrac32)&=8
   \ne7=(3\otimes1)\oplus(3\otimes\tfrac32).
 \end{aligned}
\]
Every $a\in\mathbb F_{E, M}$ has the unique additive inverse
$-a$: since all floats lie on the lattice of spacing
$2^{e_{\min}-M}$, $a\oplus b=0$ holds exactly when $b=-a$.
Multiplicative inverses need neither exist nor be unique;
for the same $(3, 2)$ format,
\[
 \tfrac32\otimes\tfrac58=1=\tfrac32\otimes\tfrac34,
 \qquad
 \tfrac1{16}\otimes b\ne1\quad(b\in\mathbb F_{3, 2}).
\]
Finally, rounded operations may overflow to $\pm\infty$,
which are outside $\mathbb F_{E, M}$. Hence this is not a field.
\end{remark}
