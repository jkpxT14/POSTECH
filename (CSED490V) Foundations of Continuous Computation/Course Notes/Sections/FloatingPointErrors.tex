\section{Floating-Point Errors}

Fix a format $(E, M)$ as in Section~3. Write
$\mathbb F=\mathbb F_{E, M}$ and $\rnd=\rnd_{E, M}$,
and retain $e_{\min}$, $e_{\max}$, $K_{E, M}$, and
$\max\mathbb F$ from that section. The rounding mode remains
round to nearest, ties to even.

\begin{definition}[Normal and Subnormal Ranges]
A real number $x$ is in the \emph{normal range} if
\[
 2^{e_{\min}}\le|x|\le\max\mathbb F,
\]
and in the \emph{subnormal range} if
\[
 |x|<2^{e_{\min}}.
\]
The subnormal range includes $0$; a \emph{subnormal number}
is a nonzero representable value within that range.
\end{definition}

\begin{definition}[Absolute and Relative Errors]
For $x\in\R$ and an approximation $\widehat{x}\in\mathbb F$,
\[
 \begin{aligned}
 \operatorname{err}_{\mathrm{abs}}(x, \widehat{x})
  &\coloneqq|x-\widehat{x}|, \\
 \operatorname{err}_{\mathrm{rel}}(x, \widehat{x})
  &\coloneqq\frac{|x-\widehat{x}|}{|x|}\qquad(x\ne0).
 \end{aligned}
\]
\end{definition}

\begin{definition}[Unit in the Last Place]
The \emph{unit in the last place} is
$\ulp\colon\R\to\R_{>0}$, where
\[
 \ulp(x)\coloneqq
 \begin{cases}
 2^{e-M}, &2^e\le|x|<2^{e+1}, \quad e\in\mathbb Z, \ e\ge e_{\min}, \\
 2^{e_{\min}-M}, &|x|<2^{e_{\min}}.
 \end{cases}
\]
For an approximation $\widehat{x}\in\mathbb F$, set
\[
 \operatorname{err}_{\mathrm{ulp}}(x, \widehat{x})
 \coloneqq\frac{|x-\widehat{x}|}{\ulp(x)}.
\]
It measures error in units of the local floating-point spacing.
The \emph{unit roundoff} (denoted here by machine epsilon) is
\[
 \varepsilon\coloneqq\tfrac12\ulp(1)=2^{-M-1}.
\]
\end{definition}

\begin{proposition}[Relative and ULP Errors]
For $x\ne0$ and $\widehat{x}\in\mathbb F$,
\[
 \operatorname{err}_{\mathrm{ulp}}(x, \widehat{x})
 \le\frac{\operatorname{err}_{\mathrm{rel}}(x, \widehat{x})}
 {\varepsilon}.
\]
If $x$ is in the normal range, also
\[
 \operatorname{err}_{\mathrm{ulp}}(x, \widehat{x})
 \ge\frac{\operatorname{err}_{\mathrm{rel}}(x, \widehat{x})}
 {2\varepsilon}.
\]
\end{proposition}
\begin{proof}
By definition,
\[
 \operatorname{err}_{\mathrm{ulp}}(x, \widehat{x})
 =\operatorname{err}_{\mathrm{rel}}(x, \widehat{x})
 \frac{|x|}{\ulp(x)}.
\]
For $2^e\le|x|<2^{e+1}$, $e\ge e_{\min}$,
\[
 2^M\le\frac{|x|}{\ulp(x)}<2^{M+1}=\frac1\varepsilon.
\]
In the subnormal range, the ratio is less than $2^M$.
These bounds prove both assertions. The normal-range assumption
is essential: for $x=2^{e_{\min}-M}$ and $\widehat{x}=0$,
both errors equal $1$, but $1/(2\varepsilon)=2^M>1$.
\end{proof}

\begin{proposition}[Rounding Error Bounds]
For $x\in\R$ with $|x|\le\max\mathbb F$:
\begin{enumerate}[label=(\roman*)]
 \item If $x$ is in the normal range and $x\ne0$, then
 \[
 \operatorname{err}_{\mathrm{rel}}(x, \rnd(x))\le\varepsilon.
 \]
 \item If $x$ is in the subnormal range, then
 \[
 \operatorname{err}_{\mathrm{abs}}(x, \rnd(x))
 \le2^{e_{\min}}\varepsilon.
 \]
 \item In either range,
 \[
 \operatorname{err}_{\mathrm{ulp}}(x, \rnd(x))\le\tfrac12.
 \]
\end{enumerate}
\end{proposition}
\begin{proof}
For $2^e\le|x|<2^{e+1}$ in the normal range,
the local spacing is $2^{e-M}$. Nearest rounding gives
\[
 \frac{|x-\rnd(x)|}{|x|}
 \le\frac{2^{e-M-1}}{2^e}=\varepsilon.
\]
In the subnormal range the spacing is $2^{e_{\min}-M}$,
so $|x-\rnd(x)|\le2^{e_{\min}-M-1}$.
Both bounds come from rounding by at most half of $\ulp(x)$.
\end{proof}

\begin{theorem}[Error of Floating-Point Operations]
Let $a, b\in\mathbb F$, with $b\ne0$ for division, and suppose
the exact result $a*b$ satisfies $|a*b|\le\max\mathbb F$.
Pair $*\in\{+, -, \times, \div\}$ with its rounded counterpart
$\circledast\in\{\oplus, \ominus, \otimes, \odiv\}$.
Then the following cases hold:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Normal range} ($*\in\{+, -, \times, \div\}$):
 \[
 a\circledast b=(a*b)(1+\delta), \qquad|\delta|\le\varepsilon.
 \]
 \item \emph{Subnormal range}, $*\in\{+, -\}$:
 \[
 a\circledast b=a*b.
 \]
 \item \emph{Subnormal range}, $*\in\{\times, \div\}$:
 \[
 a\circledast b=a*b+\delta',
 \qquad|\delta'|\le2^{e_{\min}}\varepsilon.
 \]
\end{enumerate}
In particular, all cases admit
$a\circledast b=(a*b)(1+\delta)+\delta'$ with the displayed bounds,
where $\delta'=0$ in (i), $\delta=\delta'=0$ in (ii), and
$\delta=0$ in (iii).
\end{theorem}
\begin{proof}
By definition, $a\circledast b=\rnd(a*b)$; the normal
and subnormal rounding bounds establish (i) and (iii).
For (ii), every element of $\mathbb F$ is an integer
multiple of $2^{e_{\min}-M}$. Thus $a\pm b$ also lies on
this lattice, and a sum or difference in the subnormal
range is exactly representable.
\end{proof}

\begin{definition}[Underflow and Overflow]
For a nonzero exact arithmetic result $a*b$, distinguish:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Underflow range}: $0<|a*b|<2^{e_{\min}}$.
 \item \emph{Beyond the finite range}:
 $|a*b|>\max\mathbb F$.
 \item \emph{Overflow to infinity}:
 $|a*b|\ge K_{E, M}$, under the fixed rounding mode.
\end{enumerate}
These are not interchangeable: an underflow-range result
need not round to zero, and a value strictly between
$\max\mathbb F$ and $K_{E, M}$ may still round to
$\max\mathbb F$ (with the appropriate sign).
\end{definition}

\begin{example}[Error in Repeated Multiplication]
Let $a, b, c\in\mathbb F\setminus\{0\}$, and suppose that
$ab$, $(a\otimes b)c$, and $abc$ are in the normal range,
with no intermediate overflow. Then
\[
 (a\otimes b)\otimes c
 =abc(1+\delta_1)(1+\delta_2),
 \qquad|\delta_1|, |\delta_2|\le\varepsilon.
\]
Consequently,
\[
 \begin{aligned}
 \operatorname{err}_{\mathrm{rel}}
  (abc, (a\otimes b)\otimes c)
 &\le2\varepsilon+\varepsilon^2, \\
 \operatorname{err}_{\mathrm{ulp}}
  (abc, (a\otimes b)\otimes c)
 &\le2+\varepsilon.
 \end{aligned}
\]
The leading-order relative error is at most $2\varepsilon$.
\end{example}

\begin{example}[Error in Repeated Addition]
Let $a, b, c\in\mathbb F$ with $a+b+c\ne0$ and suppose
$|a+b|, |(a\oplus b)+c|\le\max\mathbb F$.
Rounded addition has no additive subnormal error, so
\[
 \begin{aligned}
 (a\oplus b)\oplus c
 &=\bigl((a+b)(1+\delta_1)+c\bigr)(1+\delta_2)\\
 &=(a+b+c)(1+\delta_2)
  +(a+b)\delta_1(1+\delta_2),
 \end{aligned}
\]
where $|\delta_1|, |\delta_2|\le\varepsilon$. Therefore
\[
 \begin{aligned}
 \operatorname{err}_{\mathrm{rel}}
 &(a+b+c, (a\oplus b)\oplus c)\\
 &\le\varepsilon\left(1+\frac{|a+b|}{|a+b+c|}\right)
 +\varepsilon^2\frac{|a+b|}{|a+b+c|}, \\
 \operatorname{err}_{\mathrm{ulp}}
 &(a+b+c, (a\oplus b)\oplus c)\\
 &\le1+\frac{|a+b|}{|a+b+c|}
 +\varepsilon\frac{|a+b|}{|a+b+c|}.
 \end{aligned}
\]
The ratio $|a+b|/|a+b+c|$ can be large. For instance, when
$2^{M+1}\in\mathbb F$, put $a=1$, $b=2^{M+1}$, and
$c=-2^{M+1}$. Ties-to-even gives
\[
 (a\oplus b)\oplus c=0,
 \qquad a+b+c=1,
\]
so the relative error is $1$ and the ULP error is $2^M$.
\end{example}

\begin{remark}[Catastrophic Cancellation]
When large, nearly equal quantities cancel, small errors inherited
from intermediate calculations may dominate their difference.
The preceding example loses the small term $a$ in its first addition.
The effect can also expose pre-existing input errors: if
$\widehat{x}, \widehat{y}\in\mathbb F$ approximate $x, y\in\R$,
then even when subtracting the represented operands is exact,
\[
 |(\widehat{x}-\widehat{y})-(x-y)|
 \le|\widehat{x}-x|+|\widehat{y}-y|.
\]
Dividing by $|x-y|$ or $\ulp(x-y)$ illustrates how relative or
ULP error can be large when $x\approx y$, independently of
whether the final subtraction incurs rounding.
\end{remark}

\begin{example}[Stable Evaluation of $a^2-b^2$]
For $a, b\in\mathbb F$ with $a^2\ne b^2$, compare
\[
 (a\otimes a)\ominus(b\otimes b),
 \qquad (a\ominus b)\otimes(a\oplus b).
\]
Assume all exact intermediate results are normal and within the
finite range. For some $|\delta_i|\le\varepsilon$, the direct
calculation gives
\[
 (a\otimes a)\ominus(b\otimes b)
 =(a^2(1+\delta_1)-b^2(1+\delta_2))(1+\delta_3).
\]
Its relative error is therefore bounded by
\[
 \left(1+\frac{a^2+b^2}{|a^2-b^2|}\right)\varepsilon
 +O(\varepsilon^2).
\]
By contrast, the factored calculation satisfies
\[
 (a\ominus b)\otimes(a\oplus b)
 =(a-b)(a+b)(1+\delta_1)(1+\delta_2)(1+\delta_3),
\]
with relative error at most $3\varepsilon+O(\varepsilon^2)$.
For a normal exact result, the corresponding ULP-error bounds
are respectively $1+(a^2+b^2)/|a^2-b^2|+O(\varepsilon)$
and $3+O(\varepsilon)$. The former can be large when
$|a|\approx|b|$; factoring avoids subtracting two rounded squares.
\end{example}

\begin{remark}[An ULP Bound]
Suppose $\operatorname{err}_{\mathrm{ulp}}(r, r')\le1$
and $\rnd(r')$ is finite. Then
\[
 \operatorname{err}_{\mathrm{ulp}}(r, \rnd(r'))
 \le\operatorname{err}_{\mathrm{ulp}}(r, r')+\tfrac12.
\]
If $\ulp(r')\le\ulp(r)$, apply the triangle inequality and the
half-ULP rounding bound. Otherwise $r'$ lies just across a
power-of-two boundary from $r$. The one-ULP assumption keeps
$r'$ in the first half of the new binade, so it rounds to that
boundary, no farther from $r$ than $r'$.
\end{remark}

\begin{example}[Stable Quadratic Roots]
Let $a\ne0$ and $\Delta=b^2-4ac\ge0$. The quadratic roots are
\[
 x_{\pm}=\frac{-b\pm\sqrt\Delta}{2a}.
\]
When $b^2\gg|4ac|$, the numerator $-b-\sqrt\Delta$
for $b<0$ (or $-b+\sqrt\Delta$ for $b>0$) subtracts
nearly equal quantities and may suffer cancellation.
Rationalization gives the exact alternatives
\[
 \begin{aligned}
 \frac{-b-\sqrt\Delta}{2a}&=\frac{2c}{-b+\sqrt\Delta}
   &&(b<0), \\
 \frac{-b+\sqrt\Delta}{2a}&=\frac{2c}{-b-\sqrt\Delta}
   &&(b>0),
 \end{aligned}
\]
provided the displayed denominators are nonzero.

An independent problem is that forming $b^2$ may overflow even
when both roots are finite. For $b\ne0$ the exact identity
\[
 \sqrt{b^2-4ac}=|b|\sqrt{1-\frac{4ac}{b^2}}
\]
suggests rescaling, although a real implementation must still
protect the intermediate quotient. When $|4ac/b^2|\ll1$,
\[
 \sqrt{1-t}=1-\tfrac12t-\tfrac18t^2+O(t^3)
\]
explains the cancellation and yields, for example,
\[
 x_-\approx
 \begin{cases}
 -b/a+c/b, &b>0, \\
 -c/b, &b<0.
 \end{cases}
\]
These are asymptotic approximations, not exact identities.
\end{example}

\begin{example}[Two Ways to Compute a Mean]
For $a, b\in\mathbb F$, compare
\[
 (a\oplus b)\odiv2,
 \qquad(a\odiv2)\oplus(b\odiv2).
\]
If $|a+b|\le\max\mathbb F$, the first is correctly rounded:
\[
 (a\oplus b)\odiv2
 =\rnd\!\left(\frac{a+b}{2}\right), \qquad
 \min(a, b)\le(a\oplus b)\odiv2\le\max(a, b).
\]
But computing $a\oplus b$ may overflow even when the exact
mean is representable. The second avoids this intermediate
overflow. In particular, if $a/2, b/2\in\mathbb F$, both halves
are exact and the second expression is correctly rounded.
When halving incurs underflow rounding, its ULP error is at
most $1$, rather than necessarily zero.
\end{example}

\begin{theorem}[Correctly Rounded Mean]
Let
\[
 C\in[2^{e_{\min}+M+3}, 2^{e_{\max}}]\cap\mathbb F.
\]
For $a, b\in\mathbb F$, define
\[
 m(a, b)=
 \begin{cases}
 (a\oplus b)\odiv2, &|a|<C\text{ and }|b|<C, \\
 (a\odiv2)\oplus(b\odiv2), &\text{otherwise}.
 \end{cases}
\]
Then, whenever the displayed floating-point operations are defined,
\[
 m(a, b)=\rnd\!\left(\frac{a+b}{2}\right)
 \qquad(a, b\in\mathbb F).
\]
\end{theorem}
