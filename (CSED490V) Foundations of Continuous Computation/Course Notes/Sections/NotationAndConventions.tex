\thispagestyle{coursenotationpage}
\markboth{Notation and Conventions}{}
\addcontentsline{toc}{section}{Notation and Conventions}
General notation and writing follow \textit{(MATH000) Mathematical Language}.
\begin{coursenotation}
$\Rc$ & Computable real numbers\\
$P_{\mathrm{root}}$ & Integer polynomials admitting an integer root\\
$\A$ & Oracle algorithm\\
$\gamma_{E,M}$ & Floating-point representation map\\
$\mathbb{F}_{E,M}$ & Finite floating-point values\\
$\overline{\mathbb{F}}_{E,M}$ & Extended floating-point values\\
$e_{\min}$ & Minimum normal exponent\\
$e_{\max}$ & Maximum normal exponent\\
$F_{\max}$ & Largest finite floating-point value\\
$K_{E,M}$ & Overflow threshold\\
$\rnd_{E,M}$ & Round to nearest, ties to even\\
$\oplus$ & Rounded addition\\
$\ominus$ & Rounded subtraction\\
$\otimes$, $\odot$ & Rounded multiplication\\
$\odiv$, $\oslash$ & Rounded division\\
$\fma$ & Fused multiply-add\\
$\operatorname{err}_{\mathrm{abs}}$ & Absolute error\\
$\operatorname{err}_{\mathrm{rel}}$ & Relative error\\
$\ulp$ & Unit in the last place\\
$\operatorname{err}_{\mathrm{ulp}}$ & ULP error\\
$e_a$ & Exponent of a nonzero floating-point value $a$\\
$\fasttwosum$ & Error-free summation algorithm\\
$\sigma$ & Input valuation\\
$\Sigma$ & Set of admissible input valuations\\
$\llbracket e\rrbracket$ & Floating-point semantics of $e$\\
$\llbracket e\rrbracket_{\mathrm r}$ & Real-valued semantics of $e$\\
$\operatorname{range}(e)$ & Range enclosure over $\Sigma$\\
$\operatorname{err}(e)$ & Absolute-error bound over $\Sigma$\\
$\varepsilon$ & Machine epsilon (unit roundoff)
\end{coursenotation}
