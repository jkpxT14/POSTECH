\thispagestyle{coursenotationpage}
\markboth{Notation and Conventions}{}
\addcontentsline{toc}{section}{Notation and Conventions}
General notation and writing follow \textit{(MATH000) Mathematical Language}.
\begin{coursenotation}
$\Rc$ & Computable real numbers\\
$P_{\mathrm{root}}$ & Integer polynomials admitting an integer zero\\
$\A$ & Oracle algorithm\\
$\gamma_{E, M}$ & Floating-point representation map\\
$\mathbb{F}_{E, M}$ & Finite floating-point values\\
$\overline{\mathbb{F}}_{E, M}$ & Extended floating-point values\\
$e_{\min}, \ e_{\max}$ & Minimum and maximum normal exponents\\
$\max\mathbb{F}_{E, M}, \ F_{\max}$ & Largest finite floating-point value\\
$K_{E, M}$ & Overflow threshold (rounding to infinity)\\
$\rnd_{E, M}$ & Round to nearest, ties to even\\
$\oplus, \ \ominus$ & Rounded addition and subtraction\\
$\otimes, \ \odot$ & Rounded multiplication\\
$\odiv, \ \oslash$ & Rounded division\\
$\fma$ & Fused multiply-add (one rounding)\\
$\operatorname{err}_{\mathrm{abs}}, \ \operatorname{err}_{\mathrm{rel}}$ & Absolute and relative errors\\
$\ulp, \ \operatorname{err}_{\mathrm{ulp}}$ & Unit in the last place and ULP error\\
$\varepsilon$ & Unit roundoff, $2^{-M-1}$\\
$e_a$ & Exponent assigned to $a\in\mathbb{F}\setminus\{0\}$\\
$\fasttwosum$ & Error-free summation algorithm\\
$\sigma, \ \Sigma$ & Input valuation and admissible input set\\
$\llbracket e\rrbracket, \ \llbracket e\rrbracket_{\mathrm r}$ & Floating-point and real-valued semantics\\
$\operatorname{range}(e), \ \operatorname{err}(e)$ & Range enclosure and absolute-error bound over $\Sigma$
\end{coursenotation}
