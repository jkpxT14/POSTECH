\section{Combinatorics and Discrete Mathematics}

\begin{notationtable}
$[n]=\{1,2,\dots,n\}$ & Standard $n$-element set; $[0]=\varnothing$ when needed \\
$n!$ & Factorial \\
$n!!$ & Double factorial: product of positive integers of the same parity not exceeding $n$, with $0!!=(-1)!!=1$ \\
$n^{\underline{k}}$ & Falling factorial, $n(n-1)\cdots(n-k+1)$ \\*
$n^{\overline{k}}$ & Rising factorial, $n(n+1)\cdots(n+k-1)$ \\
$(a)_k$ & Rising Pochhammer symbol $a(a+1)\cdots(a+k-1)$ in special-functions contexts \\
${}_nP_r$ & Number of ordered selections of $r$ distinct elements from an $n$-element set, $\displaystyle {}_nP_r=\frac{n!}{(n-r)!}$ for $0\le r\le n$ \\
$\displaystyle\binom{n}{r},\ {}_nC_r$ & Number of $r$-element subsets of an $n$-element set \\*
$\displaystyle\left(\!\binom{n}{r}\!\right),\ {}_nH_r$ & $r$-combinations with repetition from $n$ types; equals $\displaystyle\binom{n+r-1}{r}$ \\
$\displaystyle\binom{n}{n_1,\dots,n_k}$ & Multinomial coefficient, for nonnegative integers $n_1,\dots,n_k$ with $n_1+\cdots+n_k=n$ \\
$\displaystyle\genfrac{[}{]}{0pt}{}{n}{k},\ c(n,k)$ & Unsigned Stirling number of the first kind; permutations of $n$ elements with $k$ cycles \\
$s(n,k)$ & Signed Stirling number of the first kind; $s(n,k)=(-1)^{n-k}c(n,k)$ \\[0.45em]
$\displaystyle\genfrac{\{}{\}}{0pt}{}{n}{k},\ S(n,k)$ & Stirling number of the second kind; partitions of an $n$-element set into $k$ nonempty blocks \\
$D_n,\ !n$ & Number of derangements of $n$ elements; permutations with no fixed points \\
$C_n$ & $n$th Catalan number, $\displaystyle C_n=\frac{1}{n+1}\binom{2n}{n}$ for $n\ge0$ \\
$B_n$ & $n$th Bell number; number of partitions of an $n$-element set \\
$p(n)$ & Number of integer partitions of $n$, with $p(0)=1$ \\
$\displaystyle H_n=\sum_{k=1}^n\frac1k$ & $n$th harmonic number for $n\ge1$, with $H_0=0$ \\
$(X,\preceq)$ & Partially ordered set (poset) \\
$\prec,\ \succeq,\ \succ$ & Strict and reverse order relations associated with $\preceq$ \\
$\le_{\mathrm{lex}}$ & Lexicographic order \\
$x\lessdot y$ & Cover relation: $y$ covers $x$ \\
$B_n$ & Boolean lattice (Boolean poset) on an $n$-element ground set \\
$G=(V,E)$ & Graph with vertex set $V$ and edge set $E$ \\
$V(G)$ & Vertex set of $G$ \\
$E(G)$ & Edge set of $G$ \\
$N(v),\ \Gamma(v)$ & Neighborhood of vertex $v$ \\
$N[v]=N(v)\cup\{v\}$ & Closed neighborhood of the vertex $v$ \\
$\deg_G(v),\ \deg(v)$ & Degree of vertex $v$ \\
$\deg^+(v),\ \deg^-(v)$ & Out-degree and in-degree of $v$ in a directed graph \\
$\delta(G),\ \Delta(G)$ & Minimum and maximum vertex degrees, respectively, of a nonempty finite graph $G$ \\
$d_G(u,v),\ d(u,v)$ & Graph distance between vertices $u$ and $v$ \\
$G[S]$ & Subgraph of $G$ induced by the vertex set $S\subseteq V(G)$ \\
$G-v,\ G-S$ & Graph obtained by deleting the vertex $v$, or all vertices in $S$, together with their incident edges \\
$G-e$ & Graph obtained by deleting the edge $e$ \\
$G/e$ & Graph obtained by contracting the edge $e$ \\
$\overline G$ & Complement of the graph $G$ \\
$K_n$ & Complete graph on $n$ vertices \\
$K_{m,n}$ & Complete bipartite graph with parts of sizes $m$ and $n$ \\
$P_n$ & Path graph on $n$ vertices \\
$C_n$ & Cycle graph on $n$ vertices \\
$\mathbf A(G),\ \mathbf A_G$ & Adjacency matrix of $G$ \\
$\mathbf D(G),\ \mathbf D_G$ & Degree matrix of $G$, the diagonal matrix of vertex degrees in the chosen vertex order \\
$\mathbf L(G),\ \mathbf L_G$ & Graph Laplacian, usually $\mathbf D(G)-\mathbf A(G)$ \\
$\chi(G)$ & Chromatic number of $G$ \\
$\alpha(G)$ & Independence number of $G$: maximum size of an independent vertex set \\
$\omega(G)$ & Clique number of $G$: maximum size of a complete vertex set \\
$\Aut(G)$ & Automorphism group of the graph $G$ \\
$\tau(G),\ T(G)$ & Number of spanning trees of $G$ \\
$\begin{aligned} &R(k,\ell),\ r(k,\ell),\\[-0.15em] &R(k),\ r(k) \end{aligned}$ & Ramsey numbers; $R(k)=R(k,k)$ in the diagonal case \\
$\displaystyle A(x)=\sum_{n=0}^{\infty}a_nx^n$ & Ordinary generating function of the sequence $(a_n)_{n\ge0}$ \\
$\displaystyle\sum_{n=0}^{\infty}a_n\frac{x^n}{n!}$ & Exponential generating function of the sequence $(a_n)_{n\ge0}$ \\
$[x^n]F(x)$ & Coefficient of $x^n$ in the formal power series or generating function $F(x)$ \\
\end{notationtable}
