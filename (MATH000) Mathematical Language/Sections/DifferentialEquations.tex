\section{Differential Equations}

\begin{notationtable}
$y'$ & First derivative \\
$y''$ & Second derivative \\
$\dot{x}$ & First derivative of $x$ with respect to time \\
$\ddot{x}$ & Second derivative of $x$ with respect to time \\
$F(t, y, y', \dots, y^{(n)})=0$ & General $n$th-order ordinary differential equation \\
$y^{(n)}=f(t, y, \dots, y^{(n-1)})$ & $n$th-order ordinary differential equation in normal form \\
$y(t_0)=y_0$ & Initial condition for a scalar differential equation \\
$y'+p(t)y=g(t)$ & Standard form of a first-order linear differential equation \\
$\mu(t)$ & Integrating factor for a first-order linear differential equation \\
$M(x, y)\,\dd x+N(x, y)\,\dd y=0$ & Differential form of a first-order differential equation \\
$M_x, M_y, N_x, N_y$ & Subscript notation for the corresponding partial derivatives \\
$L[y]$ & Linear differential operator applied to $y$; in order $n$, $\displaystyle L[y]=\sum_{k=0}^{n}a_k(t)y^{(k)}$ with $a_n\not\equiv0$ \\
$L[y]=0$ & Homogeneous linear differential equation \\
$L[y]=g$ & Nonhomogeneous linear differential equation for nonzero $g$ \\
$y_{\mathrm{h}}, \ y_{\mathrm{c}}$ & Homogeneous or complementary part of a solution \\
$y_{\mathrm{p}}$ & Particular solution \\
$c_1, c_2, \dots$ & Arbitrary constants in a general solution \\
$\displaystyle\sum_{k=0}^{n}a_k r^k=0$ & Characteristic equation for a constant-coefficient homogeneous linear differential equation \\
$r_1, \dots, r_n$ & Characteristic roots \\
$W(y_1, \dots, y_n)(t)$ & Wronskian of $y_1, \dots, y_n$ \\
$\displaystyle y=\sum_{n=0}^{\infty}a_n(x-x_0)^n$ & Power-series form about the ordinary point $x_0$ \\
$\displaystyle y=\sum_{n=0}^{\infty}a_n(x-x_0)^{n+r}$ & Frobenius-series form about a regular singular point $x_0$; $r$ denotes an indicial exponent \\
$\mathbf{x}'=\mathbf{f}(t, \mathbf{x})$ & General first-order system \\
$\mathbf{x}(t_0)=\mathbf{x}_0$ & Initial condition for a first-order system \\
$\begin{aligned} &\mathbf{x}'=\mathbf{A}\mathbf{x}+\mathbf{g}(t), \\ &\dot{\mathbf{x}}=\mathbf{A}\mathbf{x}+\mathbf{g}(t)\end{aligned}$ & Linear first-order system in state-vector form \\
$\mathbf{x}'=\mathbf{f}(\mathbf{x})$ & Autonomous first-order system \\
$\mathbf{x}_{\mathrm{e}}$ & Equilibrium point of an autonomous system \\
$\boldsymbol{\Phi}(t)$ & Fundamental matrix of a linear system \\
$\mathcal{L}\{f(t)\}(s), \ \mathcal{L}[f](s)$ & One-sided Laplace transform of $f$ \\
$\mathcal{L}^{-1}\{F(s)\}(t), \ \mathcal{L}^{-1}[F](t)$ & Inverse Laplace transform of $F$ \\
$(f\ast g)(t)$ & One-sided convolution of $f$ and $g$ \\
$H(t-a), \ u(t-a)$ & Heaviside or unit-step function shifted to $t=a$; its value at the jump is specified when relevant \\
$\delta(t), \ \delta(t-a)$ & Dirac delta distribution, or unit impulse, centered at $0$ or shifted to $t=a$ \\
$u_x, u_t, u_{xx}, u_{xy}, u_{tt}, \dots$ & Subscript notation for partial derivatives of a scalar function $u$ \\
$\partial_{\mathbf{n}}u$ & Normal derivative of $u$ along the unit normal $\mathbf{n}$; outward on $\partial\Omega$ for boundary conditions \\
$u|_{\partial\Omega}=g$ & Dirichlet boundary condition on $\partial\Omega$ \\*
$\partial_{\mathbf{n}}u|_{\partial\Omega}=g$ & Neumann boundary condition on $\partial\Omega$ \\*
$\alpha u+\beta\,\partial_{\mathbf{n}}u=g$ & Robin boundary condition on $\partial\Omega$ \\
$a_n$ & Cosine coefficient of a real $2L$-periodic function: $\displaystyle a_n=\frac{1}{L}\int_{-L}^{L}f(x)\cos\frac{n\pi x}{L}\,\dd x$ for $n\geq0$ \\*
$b_n$ & Sine coefficient of a real $2L$-periodic function: $\displaystyle b_n=\frac{1}{L}\int_{-L}^{L}f(x)\sin\frac{n\pi x}{L}\,\dd x$ for $n\geq1$ \\*
$\begin{aligned} &f(x)\sim\frac{a_0}{2}+\sum_{n=1}^\infty\bigg(\\[-0.15em]&\quad a_n\cos\frac{n\pi x}{L}+b_n\sin\frac{n\pi x}{L}\bigg)\end{aligned}$ & Real Fourier series of a $2L$-periodic function \\
$\widehat f(\boldsymbol{\xi}), \ (\mathcal{F}f)(\boldsymbol{\xi})$ & Fourier transform: $\displaystyle\widehat f(\boldsymbol{\xi})=\int_{\R^n}f(\mathbf{x})e^{-2\pi i\mathbf{x}\cdot\boldsymbol{\xi}}\,\dd\mathbf{x}$ \\*
$\mathcal{F}^{-1}, \ (\mathcal{F}^{-1}\widehat f)(\mathbf{x})$ & Inverse Fourier transform under the stated normalization: $\displaystyle(\mathcal{F}^{-1}\widehat f)(\mathbf{x})=\int_{\R^n}\widehat f(\boldsymbol{\xi})e^{2\pi i\mathbf{x}\cdot\boldsymbol{\xi}}\,\dd\boldsymbol{\xi}$ \\
$-(py')'+qy=\lambda wy$ & Sturm--Liouville eigenvalue equation, where $p$, $q$, and $w$ are functions of $x$ \\
$\lambda_n$ & Sturm--Liouville eigenvalue \\
$\phi_n$ & Sturm--Liouville eigenfunction corresponding to $\lambda_n$ \\
$\langle f, g\rangle_w$ & Weighted inner product, $\displaystyle\langle f, g\rangle_w\coloneqq\int_a^b\overline{f(x)}g(x)w(x)\,\dd x$ \\
$\langle\phi_m, \phi_n\rangle_w=\delta_{mn}$ & Weighted orthonormality of eigenfunctions \\
$G(x, \xi), \ G(\mathbf{x}, \boldsymbol{\xi})$ & Green's function \\
$\operatorname{Ai}(x)$ & Airy function of the first kind \\
$\operatorname{Bi}(x)$ & Airy function of the second kind \\
$J_\nu(x)$ & Bessel function of the first kind of order $\nu$ \\
$Y_\nu(x), \ N_\nu(x)$ & Bessel function of the second kind of order $\nu$ \\
$I_\nu(x)$ & Modified Bessel function of the first kind of order $\nu$ \\
$K_\nu(x)$ & Modified Bessel function of the second kind of order $\nu$ \\
$H_\nu^{(1)}(x)$ & Hankel function of the first kind of order $\nu$ \\
$H_\nu^{(2)}(x)$ & Hankel function of the second kind of order $\nu$ \\
$P_n(x)$ & Legendre polynomial of degree $n$ \\
$P_\nu(x)$ & Legendre function of the first kind of degree $\nu$ \\
$Q_\nu(x)$ & Legendre function of the second kind of degree $\nu$ \\
$P_\ell^m(x)$ & Associated Legendre function for integer degree $\ell$ and order $m$ \\
$H_n(x)$ & Physicists' Hermite polynomial \\
$\mathrm{He}_n(x)$ & Probabilists' Hermite polynomial \\
$L_n(x)$ & Ordinary Laguerre polynomial $L_n^{(0)}(x)$ \\*
$L_n^{(\alpha)}(x)$ & Generalized Laguerre polynomial \\
$T_n(x)$ & Chebyshev polynomial of the first kind \\
$U_n(x)$ & Chebyshev polynomial of the second kind \\
$P_n^{(\alpha, \beta)}(x)$ & Jacobi polynomial \\
$C_n^{(\lambda)}(x)$ & Gegenbauer (ultraspherical) polynomial \\
$\operatorname{erf}(x)$ & Error function \\
$\operatorname{erfc}(x)$ & Complementary error function \\
${}_2F_1(a, b; c; z)$ & Gauss hypergeometric function \\
\end{notationtable}
