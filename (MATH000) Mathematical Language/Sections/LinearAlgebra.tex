\section{Linear Algebra}

\begin{notationtable}
$M_{m\times n}(F), \ F^{m\times n}$ & Space of $m\times n$ matrices over the field $F$ \\
$M_n(F)$ & Space (and matrix algebra) of $n\times n$ matrices over $F$ \\
$\mathbf{A}=(a_{ij}), \ \mathbf{A}=[a_{ij}]$ & Matrix with $(i, j)$-entry $a_{ij}$ \\
$\mathbf{A}^{\mathrm{T}}$ & Transpose of $\mathbf{A}$ \\
$\mathbf{A}^{\mathrm{H}}$ & Conjugate transpose of $\mathbf{A}$ \\
$\mathbf{A}^{-1}$ & Inverse of $\mathbf{A}$ \\
$\mathbf{I}_n, \ \mathbf{I}$ & Identity matrix \\
$\mathbf{0}$ & Zero vector or zero matrix, with dimension understood \\
$\mathbf{e}_i$ & $i$th standard basis vector \\
$\mathbf{i}, \ \mathbf{j}, \ \mathbf{k}$ & Standard basis vectors of $\R^3$, along the $x$-, $y$-, and $z$-axes, respectively \\
$\ones$ & All-ones vector \\
$\mathbf{D}$ & Diagonal matrix \\
$\diag(d_1, \dots, d_n)$ & Diagonal matrix with entries $d_1, \dots, d_n$ \\
$\mathbf{P}$ & Projection matrix \\
$\det(\mathbf{A}), \ |\mathbf{A}|$ & Determinant \\
$\rk(\mathbf{A}), \ \operatorname{rank}(\mathbf{A})$ & Rank \\
$\tr(\mathbf{A}), \ \trace(\mathbf{A})$ & Trace \\
$\adj(\mathbf{A})$ & Adjugate matrix \\
$M_{ij}$ & Minor of the $(i, j)$-entry \\
$C_{ij}=(-1)^{i+j}M_{ij}$ & Cofactor of the $(i, j)$-entry \\
$\mathbf{A}\otimes\mathbf{B}$ & Kronecker product of matrices $\mathbf{A}$ and $\mathbf{B}$ \\
$\Col(\mathbf{A})$ & Column space \\
$\Row(\mathbf{A})$ & Row space \\
$\ker\mathbf{A}, \ \Null(\mathbf{A})$ & Null space of $\mathbf{A}$ \\
$V, W$ & Vector spaces, usually over an understood field $F$ \\
$\Hom_F(V, W)$ & Vector space of $F$-linear maps from $V$ to $W$ \\
$\End_F(V)$ & $F$-algebra of $F$-linear endomorphisms of $V$ \\
$\mathcal{L}(X, Y)$ & Bounded linear maps between normed spaces $X$ and $Y$ \\
$\dim V$ & Dimension of $V$ \\
$\Span(\mathbf{v}_1, \dots, \mathbf{v}_k)$ & Span of the listed vectors \\
$U\oplus V$ & Direct sum \\
$V/U$ & Quotient vector space of $V$ by its vector subspace $U\subseteq V$; elements are cosets $\mathbf{v}+U$ \\
$\ker T$ & Kernel of a linear map $T$ \\
$\nullity(T)=\dim\ker T$ & Nullity of a linear map $T$ \\
$\im T$ & Image of a linear map $T$ \\
$\mathcal{B}=(\mathbf{b}_1, \dots, \mathbf{b}_n)$ & Ordered basis \\
$[\mathbf{v}]_{\mathcal{B}}$ & Coordinate vector of $\mathbf{v}$ with respect to the ordered basis $\mathcal{B}$ \\
$[T]_{\mathcal{C}\leftarrow\mathcal{B}}$ & Matrix of $T\colon V\to W$ relative to domain basis $\mathcal{B}$ and codomain basis $\mathcal{C}$ \\
$[T]_{\mathcal{B}}$ & Matrix of an endomorphism $T\colon V\to V$ when the same basis $\mathcal{B}$ is used in the domain and codomain \\
$\mathbf{P}_{\mathcal{C}\leftarrow\mathcal{B}}$ & Change-of-coordinate matrix from basis $\mathcal{B}$ to basis $\mathcal{C}$ \\
$V^*$ & Algebraic dual of a vector space $V$; for a normed space $X$, $X^*$ usually denotes the continuous dual \\
$\langle\mathbf{u}, \mathbf{v}\rangle, \ (\mathbf{u}, \mathbf{v})$ & Inner product. Over complex spaces, the inner product is conjugate-linear in the first argument and linear in the second \\
$\mathbf{u}\cdot\mathbf{v}, \ \mathbf{u}^{\mathrm{T}}\mathbf{v}$ & Standard Euclidean inner product on $\R^n$ \\
$\mathbf{u}^{\mathrm{H}}\mathbf{v}$ & Standard Hermitian inner product on $\C^n$ \\
$\|\mathbf{v}\|$ & Norm of $\mathbf{v}$ \\
$\|\mathbf{x}\|_p$ & Vector $p$-norm, for $1\le p<\infty$ \\
$\|\mathbf{x}\|_\infty$ & Vector infinity norm \\
$\|T\|_{\mathrm{op}}$ & Operator norm $\displaystyle\sup_{\|x\|\le1}\|Tx\|$ of a bounded linear map $T\colon X\to Y$ \\
$\|\mathbf{A}\|_{\mathrm{F}}$ & Frobenius norm of $\mathbf{A}$ \\
$\mathbf{u}\perp\mathbf{v}$ & Orthogonality \\
$U^\perp$ & Orthogonal complement of the subspace $U$ \\
$\proj_U\mathbf{x}$ & Orthogonal projection of $\mathbf{x}$ onto the subspace $U$ \\
$\proj_{\mathbf{v}}\mathbf{x}$ & Orthogonal projection of $\mathbf{x}$ onto $\Span(\mathbf{v})$ \\
$\mathbf{u}\times\mathbf{v}$ & Cross product in $\R^3$ \\
$\mathbf{A}^{\mathrm{T}}=\mathbf{A}$ & Real symmetric matrix \\
$\mathbf{A}^{\mathrm{T}}=-\mathbf{A}$ & Real skew-symmetric matrix \\
$\mathbf{A}^{\mathrm{H}}=\mathbf{A}$ & Hermitian matrix \\
$\mathbf{A}^{\mathrm{H}}=-\mathbf{A}$ & Skew-Hermitian matrix \\
$\mathbf{Q}^{\mathrm{T}}\mathbf{Q}=\mathbf{I}$ & Real orthogonal matrix $\mathbf{Q}$ \\
$\mathbf{U}^{\mathrm{H}}\mathbf{U}=\mathbf{I}$ & Unitary matrix $\mathbf{U}$ \\
$\mathbf{A}\mathbf{A}^{\mathrm{H}}=\mathbf{A}^{\mathrm{H}}\mathbf{A}$ & Normal matrix \\
$\mathbf{P}^2=\mathbf{P}$ & Projection matrix \\
$\mathbf{P}^2=\mathbf{P}=\mathbf{P}^{\mathrm{T}}$ & Orthogonal projection matrix over $\R$ \\
$\mathbf{P}^2=\mathbf{P}=\mathbf{P}^{\mathrm{H}}$ & Orthogonal projection matrix over $\C$ \\
$\chi_{\mathbf{A}}(t), \ p_{\mathbf{A}}(t)$ & Characteristic polynomial of $\mathbf{A}$, with $\chi_{\mathbf{A}}(t)=\det(t\mathbf{I}-\mathbf{A})$ \\*
$m_{\mathbf{A}}(t), \ \mu_{\mathbf{A}}(t)$ & Minimal polynomial of $\mathbf{A}$ \\
$\lambda, \ \lambda_i$ & Eigenvalue of $\mathbf{A}$ \\
$\mathbf{v}, \ \mathbf{v}_i$ & Eigenvector of $\mathbf{A}$ \\
$E_\lambda(\mathbf{A}), \ V_\lambda(\mathbf{A})$ & Eigenspace of $\mathbf{A}$ associated with $\lambda$ \\
$\mathbf{J}_k(\lambda)$ & Jordan block of size $k$ with eigenvalue $\lambda$ \\
$\sigma(\mathbf{A}), \ \spec(\mathbf{A})$ & Spectrum of $\mathbf{A}$ \\
$\rho(\mathbf{A})$ & Spectral radius \\
$\mathbf{A}\sim\mathbf{B}$ & Similar square matrices \\
$\mathbf{P}_\lambda$ & Spectral projection associated with the eigenvalue $\lambda$ \\
$\mathbf{A}=\mathbf{P}\mathbf{D}\mathbf{P}^{-1}$ & Diagonalization of a diagonalizable matrix \\
$\mathbf{A}=\mathbf{Q}\mathbf{D}\mathbf{Q}^{\mathrm{T}}$ & Orthogonal diagonalization of a real symmetric matrix \\
$\mathbf{A}=\mathbf{U}\mathbf{D}\mathbf{U}^{\mathrm{H}}$ & Unitary diagonalization of a complex normal matrix \\
$\displaystyle\mathbf{A}=\sum_{\lambda\in\sigma(\mathbf{A})} \lambda\mathbf{P}_\lambda$ & Spectral decomposition of $\mathbf{A}$ \\
$e^{\mathbf{A}}, \ \exp(\mathbf{A})$ & Matrix exponential \\
$\mathbf{A}=\mathbf{L}\mathbf{U}$ & LU factorization, with $\mathbf{L}$ lower triangular and $\mathbf{U}$ upper triangular \\
$\mathbf{A}=\mathbf{Q}\mathbf{R}$ & QR factorization \\
$\mathbf{A}=\mathbf{U}\boldsymbol{\Sigma}\mathbf{V}^{\mathrm{H}}$ & Singular value decomposition (SVD) of $\mathbf{A}$ \\
$\sigma_i(\mathbf{A})$ & $i$th singular value of $\mathbf{A}$ \\
$\mathbf{A}^\dagger$ & Moore--Penrose pseudoinverse of $\mathbf{A}$ \\
$\mathbf{A}\succ0$ & Positive definite Hermitian matrix (real symmetric in the real case) \\
$\mathbf{A}\succeq0$ & Positive semidefinite Hermitian matrix \\
$\mathbf{A}\prec0$ & Negative definite Hermitian matrix \\
$\mathbf{A}\preceq0$ & Negative semidefinite Hermitian matrix \\
$\mathbf{A}\succeq\mathbf{B}$ & L\"owner order: $\mathbf{A}-\mathbf{B}\succeq0$, for Hermitian matrices of the same size \\
$\mathbf{A}=\mathbf{L}\mathbf{L}^{\mathrm{H}}$ & Cholesky factorization of a positive definite Hermitian matrix \\
\end{notationtable}
