\section{Number Theory}

\begin{notationtable}
$a\mid b$ & $a$ divides $b$. \\
$a\nmid b$ & $a$ does not divide $b$. \\
$\gcd(a,b)$ & Greatest common divisor, taken nonnegative. \\
$\lcmop(a,b)$ & Least common multiple. \\
$a\equiv b\pmod n$ & $a$ is congruent to $b$ modulo $n$. \\
$[a]_n$ & Residue class of $a$ modulo $n$. \\
$\Z/n\Z$ & Ring of residue classes modulo $n$. \\
$(\Z/n\Z)^\times$ & Multiplicative group of units modulo $n$. \\
$a^{-1}\pmod n$ & Multiplicative inverse of $a$ modulo $n$. \\
$\displaystyle\left(\frac ap\right)$ & Legendre symbol. \\
$\varphi(n)$ & Euler totient function. \\
$\ord_n(a)$ & Multiplicative order of $a$ modulo $n$. \\
$v_p(n),\ \nu_p(n)$ & $p$-adic valuation of $n$. \\
$\tau(n)$ & Number of positive divisors of $n$. \\
$\sigma(n)$ & Sum of positive divisors of $n$. \\
$\mu(n)$ & M\"obius function. \\
$\rad(n)$ & Product of the distinct prime divisors of $n$. \\
$\pi(x)$ & Number of primes not exceeding $x$. \\
$\F_q,\ \operatorname{GF}(q)$ & Finite field with $q$ elements. \\
\end{notationtable}
