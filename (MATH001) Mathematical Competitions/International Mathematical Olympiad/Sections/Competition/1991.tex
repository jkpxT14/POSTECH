\competitionyear{1991}

\begin{problem}{1991}{1}
Let $I$ be the incentre of triangle $ABC$. The internal angle bisectors from $A$, $B$, and $C$ meet the opposite sides at $A'$, $B'$, and $C'$, respectively. Prove that
\[
    \frac14<\frac{AI\cdot BI\cdot CI}{AA'\cdot BB'\cdot CC'}\leq\frac8{27}.
\]
\end{problem}

\begin{problem}{1991}{2}
For an integer $n>6$, let $a_1, \dots, a_k$ be all positive integers less than $n$ and coprime to $n$. Suppose that
\[
    a_2-a_1=a_3-a_2=\cdots=a_k-a_{k-1}>0.
\]
Prove that $n$ is prime or a power of $2$.
\end{problem}

\begin{problem}{1991}{3}
Let $S=\{1, 2, \dots, 280\}$. Determine the least integer $n$ for which every $n$-element subset of $S$ contains five pairwise coprime integers.
\end{problem}

\begin{problem}{1991}{4}
Let $G$ be a connected simple undirected graph with $k$ edges. Show that its edges can be labelled bijectively by $1, \dots, k$ so that, at every vertex of degree at least $2$, the labels of the incident edges have greatest common divisor $1$.
\end{problem}

\begin{problem}{1991}{5}
Let $P$ be inside triangle $ABC$. Prove that
\[
    \min\{m\angle PAB, m\angle PBC, m\angle PCA\}\leq30^\circ.
\]
\end{problem}

\begin{problem}{1991}{6}
Given a real number $a>1$, construct a bounded sequence $(x_i)_{i\geq0}$ of real numbers satisfying
\[
    |x_i-x_j||i-j|^a\geq1
\]
for all distinct $i, j\in\Nzero$.
\end{problem}
