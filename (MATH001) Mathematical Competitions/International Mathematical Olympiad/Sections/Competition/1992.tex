\competitionyear{1992}

\begin{problem}{1992}{1}
Determine all integers $1<a<b<c$ such that
\[
    (a-1)(b-1)(c-1)\mid abc-1.
\]
\end{problem}

\begin{problem}{1992}{2}
Determine all functions $f\colon\R\to\R$ satisfying
\[
    f(x^2+f(y))=y+f(x)^2
\]
for all $x, y\in\R$.
\end{problem}

\begin{problem}{1992}{3}
Take nine points in space, no four coplanar. Join every pair by an edge, and colour each edge red or blue, or leave it uncoloured. Find the least $n$ for which every assignment with exactly $n$ coloured edges contains a triangle whose three edges have the same colour.
\end{problem}

\begin{problem}{1992}{4}
Let $C$ be a circle, $\ell$ a tangent to $C$, and $M\in\ell$. Determine the locus of points $P$ for which there exist $Q, R\in\ell$ with midpoint $M$ and with $C$ as the incircle of triangle $PQR$.
\end{problem}

\begin{problem}{1992}{5}
For a finite set $S\subseteq\R^3$, let $S_x$, $S_y$, and $S_z$ be its orthogonal projections onto the $yz$-, $zx$-, and $xy$-planes, respectively. Prove that
\[
    |S|^2\leq|S_x||S_y||S_z|.
\]
\end{problem}

\begin{problem}{1992}{6}
For $n\in\N$, define $S(n)$ to be the largest integer such that $n^2$ is expressible as a sum of $k$ positive squares for every positive integer $k\leq S(n)$.
\begin{problemsubparts}
    \item Show that $S(n)\leq n^2-14$ whenever $n\geq4$.
    \item Find an integer $n$ with $S(n)=n^2-14$.
    \item Show that infinitely many integers $n$ satisfy $S(n)=n^2-14$.
\end{problemsubparts}
\end{problem}
