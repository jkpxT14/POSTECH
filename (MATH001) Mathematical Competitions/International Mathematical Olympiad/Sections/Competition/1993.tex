\competitionyear{1993}

\begin{problem}{1993}{1}
For an integer $n>1$, prove that the polynomial $f(x)=x^n+5x^{n-1}+3$ is not a product of two nonconstant polynomials in $\Z[x]$.
\end{problem}

\begin{problem}{1993}{2}
Let $D$ lie inside acute triangle $ABC$ and satisfy
\[
    m\angle ADB=m\angle ACB+\frac{\pi}{2},\qquad AC\cdot BD=AD\cdot BC.
\]
\begin{problemsubparts}
    \item Determine $\displaystyle\frac{AB\cdot CD}{AC\cdot BD}$.
    \item Show that the tangents at $C$ to the circumcircles of triangles $ACD$ and $BCD$ are perpendicular.
\end{problemsubparts}
\end{problem}

\begin{problem}{1993}{3}
An infinite chessboard initially has one piece in each cell of an $n\times n$ block. A move jumps a piece horizontally or vertically over a neighbouring occupied cell into the next cell, which must be empty, and removes the jumped piece. Determine the values of $n$ for which a single piece can remain.
\end{problem}

\begin{problem}{1993}{4}
For points $P, Q, R$ in the plane, let $m(PQR)$ denote the shortest altitude of triangle $PQR$, taking $m(PQR)=0$ when the points are collinear. Show that for any points $A, B, C, X$,
\[
    m(ABC)\leq m(ABX)+m(AXC)+m(XBC).
\]
\end{problem}

\begin{problem}{1993}{5}
Does there exist a strictly increasing function $f\colon\N\to\N$ such that $f(1)=2$ and
\[
    f(f(n))=f(n)+n
\]
for every $n\in\N$?
\end{problem}

\begin{problem}{1993}{6}
Arrange $n>1$ lamps $L_0, \dots, L_{n-1}$ in a circle, with subscripts interpreted modulo $n$. All are initially lit. At step $s_i$ ($i=0, 1, \dots$), toggle $L_i$ if $L_{i-1}$ is lit; otherwise leave it unchanged. Prove that:
\begin{problemsubparts}
    \item For some positive integer $M(n)$, all lamps are lit after $M(n)$ steps.
    \item For $n=2^k$, one may take $M(n)=n^2-1$.
    \item For $n=2^k+1$, one may take $M(n)=n^2-n+1$.
\end{problemsubparts}
\end{problem}
