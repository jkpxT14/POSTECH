\competitionyear{1994}

\begin{problem}{1994}{1}
Let $m, n\in\N$ and let $a_1, \dots, a_m$ be distinct elements of $\{1, \dots, n\}$. Assume that for $1\leq i\leq j\leq m$, whenever $a_i+a_j\leq n$, there is a $k$ with $1\leq k\leq m$ and $a_k=a_i+a_j$. Prove that
\[
    \frac{a_1+\cdots+a_m}{m}\geq\frac{n+1}{2}.
\]
\end{problem}

\begin{problem}{1994}{2}
Let $ABC$ be an isosceles triangle with $AB=AC$. Suppose that:
\begin{problemclauses}
    \item $M$ is the midpoint of $BC$, and $O$ lies on line $AM$ with $\overleftrightarrow{OB}\perp\overleftrightarrow{AB}$;
    \item $Q$ lies strictly between $B$ and $C$;
    \item $E$ and $F$ lie on lines $AB$ and $AC$, respectively, and $E, Q, F$ are distinct and collinear.
\end{problemclauses}
Prove that $\overleftrightarrow{OQ}\perp\overleftrightarrow{EF}$ if and only if $QE=QF$.
\end{problem}

\begin{problem}{1994}{3}
For $k\in\N$, let $f(k)$ count the integers in $\{k+1, \dots, 2k\}$ whose binary expansions contain exactly three digits equal to $1$.
\begin{problemsubparts}
    \item Show that every $m\in\N$ is attained by $f$.
    \item Determine all $m\in\N$ attained for exactly one $k\in\N$.
\end{problemsubparts}
\end{problem}

\begin{problem}{1994}{4}
Determine all ordered pairs $(m, n)\in\N^2$ for which
\[
    \frac{n^3+1}{mn-1}\in\Z.
\]
\end{problem}

\begin{problem}{1994}{5}
Let $S=(-1, \infty)$. Determine all functions $f\colon S\to S$ satisfying:
\begin{problemclauses}
    \item $f(x+f(y)+xf(y))=y+f(x)+yf(x)$ for all $x, y\in S$;
    \item $f(x)/x$ is strictly increasing on each of $(-1, 0)$ and $(0, \infty)$.
\end{problemclauses}
\end{problem}

\begin{problem}{1994}{6}
Show that there is a set $A\subseteq\N$ with the following property: for every infinite set $S$ of primes, some $m\in A$ and $n\in\N\setminus A$ are each products of $k$ distinct elements of $S$, for the same integer $k\geq2$.
\end{problem}
