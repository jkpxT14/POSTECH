\competitionyear{1995}

\begin{problem}{1995}{1}
Four distinct points $A, B, C, D$ occur in that order on a line. The circles with diameters $\overline{AC}$ and $\overline{BD}$ meet at $X, Y$. Line $XY$ meets line $BC$ at $Z$. Take $P\in\overleftrightarrow{XY}$ with $P\ne Z$. Line $CP$ meets the first circle again at $M$, and line $BP$ meets the second again at $N$. Prove that lines $AM$, $DN$, and $XY$ are concurrent.
\end{problem}

\begin{problem}{1995}{2}
For positive real numbers $a, b, c$ with $abc=1$, prove that
\[
    \frac{1}{a^3(b+c)}+\frac{1}{b^3(c+a)}+\frac{1}{c^3(a+b)}\geq\frac32.
\]
\end{problem}

\begin{problem}{1995}{3}
Determine all integers $n>3$ for which there exist planar points $A_1, \dots, A_n$, no three collinear, and real numbers $r_1, \dots, r_n$ with
\[
    [A_iA_jA_k]=r_i+r_j+r_k\qquad(1\leq i<j<k\leq n).
\]
\end{problem}

\begin{problem}{1995}{4}
Find the maximum $x_0$ among sequences $(x_i)_{i=0}^{1995}$ of positive real numbers satisfying $x_0=x_{1995}$ and
\[
    x_{i-1}+\frac{2}{x_{i-1}}=2x_i+\frac1{x_i}\qquad(1\leq i\leq1995).
\]
\end{problem}

\begin{problem}{1995}{5}
Let $ABCDEF$ be a convex hexagon with $AB=BC=CD$, $DE=EF=FA$, and $m\angle BCD=m\angle EFA=\pi/3$. Interior points $G, H$ satisfy $m\angle AGB=m\angle DHE=2\pi/3$. Prove that
\[
    AG+GB+GH+DH+HE\geq CF.
\]
\end{problem}

\begin{problem}{1995}{6}
For an odd prime $p$, count the $p$-element subsets of $\{1, 2, \dots, 2p\}$ whose element sum is divisible by $p$.
\end{problem}
