\competitionyear{1996}

\begin{problem}{1996}{1}
Let $r\in\N$. A rectangular board $ABCD$ with $AB=20$ and $BC=12$ is partitioned into unit squares. Moves between squares are allowed only when their centres are at distance $\sqrt r$. The goal is to reach the square at corner $B$ from the square at corner $A$.
\begin{problemsubparts}
    \item Prove this is impossible if $2\mid r$ or $3\mid r$.
    \item Prove this is possible for $r=73$.
    \item Is it possible for $r=97$?
\end{problemsubparts}
\end{problem}

\begin{problem}{1996}{2}
Let $P$ lie inside triangle $ABC$ with
\[
    m\angle APB-m\angle ACB=m\angle APC-m\angle ABC.
\]
Let $D, E$ be the incentres of triangles $APB$ and $APC$, respectively. Prove that lines $AP$, $BD$, and $CE$ are concurrent.
\end{problem}

\begin{problem}{1996}{3}
Determine all functions $f\colon\Nzero\to\Nzero$ satisfying
\[
    f(m+f(n))=f(f(m))+f(n)
\]
for every $m, n\in\Nzero$.
\end{problem}

\begin{problem}{1996}{4}
Let $a, b$ be positive integers for which $15a+16b$ and $16a-15b$ are positive perfect squares. Find the minimum possible value of the smaller square.
\end{problem}

\begin{problem}{1996}{5}
A convex hexagon $ABCDEF$ has its opposite sides parallel. Let $R_A$, $R_C$, and $R_E$ be the circumradii of triangles $FAB$, $BCD$, and $DEF$, respectively, and let $P$ be the perimeter of the hexagon. Prove that
\[
    R_A+R_C+R_E\geq\frac P2.
\]
\end{problem}

\begin{problem}{1996}{6}
Let $p, q, n\in\N$ with $p+q<n$. Suppose that an integer tuple $(x_0, x_1, \dots, x_n)$ satisfies:
\begin{problemclauses}
    \item $x_0=x_n=0$;
    \item $x_i-x_{i-1}\in\{p, -q\}$ for $1\leq i\leq n$.
\end{problemclauses}
Prove that $x_i=x_j$ for some indices $0\leq i<j\leq n$ with $(i, j)\ne(0, n)$.
\end{problem}
