\competitionyear{1997}

\begin{problem}{1997}{1}
Colour the unit squares of the integer lattice alternately black and white as on a chessboard. For positive integers $m, n$, consider a right triangle with lattice vertices and legs of lengths $m, n$ along grid lines. Let $S_1, S_2$ be its black and white areas and define $f(m, n)\coloneqq|S_1-S_2|$.
\begin{problemsubparts}
    \item Determine $f(m, n)$ when $m, n$ have the same parity.
    \item Prove that $\displaystyle f(m, n)\leq\frac12\max\{m, n\}$ for all $m, n\in\N$.
    \item Show that $f$ is unbounded.
\end{problemsubparts}
\end{problem}

\begin{problem}{1997}{2}
Let the angle at $A$ be the smallest angle of triangle $ABC$. Take $U$ in the interior of the arc of its circumcircle from $B$ to $C$ not containing $A$. The perpendicular bisectors of $\overline{AB}$ and $\overline{AC}$ intersect line $AU$ at $V$ and $W$, respectively. Let lines $BV$ and $CW$ meet at $T$. Prove that $AU=TB+TC$.
\end{problem}

\begin{problem}{1997}{3}
Suppose real numbers $x_1, \dots, x_n$ satisfy
\[
    |x_1+\cdots+x_n|=1,\qquad |x_i|\leq\frac{n+1}{2}\quad(1\leq i\leq n).
\]
Show that some permutation $(y_1, \dots, y_n)$ of these numbers satisfies
\[
    |y_1+2y_2+\cdots+ny_n|\leq\frac{n+1}{2}.
\]
\end{problem}

\begin{problem}{1997}{4}
An $n\times n$ matrix with entries in $S=\{1, 2, \dots, 2n-1\}$ is called \emph{silver} if, for every $i=1, \dots, n$, its $i$-th row and $i$-th column jointly contain all of $S$.
\begin{problemsubparts}
    \item Show that no silver matrix exists for $n=1997$.
    \item Show that silver matrices exist for infinitely many $n$.
\end{problemsubparts}
\end{problem}

\begin{problem}{1997}{5}
Determine all pairs $(a, b)\in\N^2$ satisfying
\[
    a^{b^2}=b^a.
\]
\end{problem}

\begin{problem}{1997}{6}
For $n\in\N$, let $f(n)$ count the representations of $n$ as a sum of powers of $2$ with nonnegative integer exponents, disregarding the order of the summands. For example, $f(4)=4$, corresponding to
\[
    4,\quad 2+2,\quad 2+1+1,\quad 1+1+1+1.
\]
Prove that for every integer $n\geq3$,
\[
    2^{n^2/4}<f(2^n)<2^{n^2/2}.
\]
\end{problem}
