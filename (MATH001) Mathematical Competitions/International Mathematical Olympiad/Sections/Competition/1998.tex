\competitionyear{1998}

\begin{problem}{1998}{1}
Let $ABCD$ be a convex quadrilateral with perpendicular diagonals and nonparallel opposite sides $AB$ and $DC$. Suppose the perpendicular bisectors of $\overline{AB}$ and $\overline{DC}$ meet at an interior point $P$. Prove that $ABCD$ is cyclic if and only if $[ABP]=[CDP]$.
\end{problem}

\begin{problem}{1998}{2}
A contest has $a$ contestants and an odd number $b\geq3$ of judges. Every judge assigns every contestant either a pass or a fail. If the judgments of any two judges agree for at most $k$ contestants, prove that
\[
    \frac ka\geq\frac{b-1}{2b}.
\]
\end{problem}

\begin{problem}{1998}{3}
Let $\tau(n)$ be the number of positive divisors of $n\in\N$. Determine all positive integers $k$ for which
\[
    \frac{\tau(n^2)}{\tau(n)}=k
\]
for some $n\in\N$.
\end{problem}

\begin{problem}{1998}{4}
Determine all pairs $(a, b)\in\N^2$ satisfying
\[
    ab^2+b+7\mid a^2b+a+b.
\]
\end{problem}

\begin{problem}{1998}{5}
Let $I$ be the incentre of triangle $ABC$. Its incircle touches sides $BC$, $CA$, and $AB$ at $K$, $L$, and $M$, respectively. The line through $B$ parallel to $MK$ intersects lines $LM$ and $LK$ at $R$ and $S$, respectively. Prove that $\angle RIS$ is acute.
\end{problem}

\begin{problem}{1998}{6}
Consider functions $f\colon\N\to\N$ with
\[
    f(t^2f(s))=sf(t)^2
\]
for every $s, t\in\N$. Determine the minimum possible $f(1998)$.
\end{problem}
