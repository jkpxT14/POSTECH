\competitionyear{1999}

\begin{problem}{1999}{1}
Determine all finite planar sets $S$ with $|S|\geq3$ such that, for any distinct $A, B\in S$, reflection in the perpendicular bisector of $\overline{AB}$ leaves $S$ invariant.
\end{problem}

\begin{problem}{1999}{2}
Fix an integer $n\geq2$.
\begin{problemsubparts}
    \item Find the least constant $C$ for which
    \[
        \sum_{1\leq i<j\leq n}x_ix_j(x_i^2+x_j^2)
        \leq C\left(\sum_{i=1}^{n}x_i\right)^4
    \]
    for all nonnegative real numbers $x_1, \dots, x_n$.
    \item For that $C$, determine all equality cases.
\end{problemsubparts}
\end{problem}

\begin{problem}{1999}{3}
Let $n$ be a fixed positive even integer. Mark $N$ cells of an $n\times n$ board so that every cell, marked or not, shares an edge with a marked cell distinct from itself. Find the minimum possible $N$.
\end{problem}

\begin{problem}{1999}{4}
Determine all pairs $(n, p)\in\N^2$ satisfying:
\begin{problemclauses}
    \item $p$ is prime;
    \item $n\leq2p$;
    \item $n^{p-1}\mid (p-1)^n+1$.
\end{problemclauses}
\end{problem}

\begin{problem}{1999}{5}
Circles $G_1, G_2$ lie inside a circle $G$ and touch it internally at distinct points $M, N$, respectively. The centre of $G_2$ lies on $G_1$. The common chord line of $G_1$ and $G_2$ meets $G$ at $A, B$. Lines $MA$ and $MB$ meet $G_1$ again at $C$ and $D$, respectively. Prove that line $CD$ is tangent to $G_2$.
\end{problem}

\begin{problem}{1999}{6}
Determine all functions $f\colon\R\to\R$ satisfying
\[
    f(x-f(y))=f(f(y))+xf(y)+f(x)-1
\]
for every $x, y\in\R$.
\end{problem}
