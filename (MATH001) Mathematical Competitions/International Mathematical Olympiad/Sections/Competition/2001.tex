\competitionyear{2001}

\begin{problem}{2001}{1}
Let $ABC$ be an acute triangle with circumcentre $O$, and let $P$ be the foot of the altitude from $A$ to the side $BC$. Suppose that $m\angle BCA\geq m\angle ABC+30^\circ$. Prove that $m\angle CAB+m\angle COP<90^\circ$.
\end{problem}

\begin{problem}{2001}{2}
For positive real numbers $a, b, c$, prove the inequality
\[
    \frac{a}{\sqrt{a^2+8bc}}+\frac{b}{\sqrt{b^2+8ca}}+\frac{c}{\sqrt{c^2+8ab}}\geq1.
\]
\end{problem}

\begin{problem}{2001}{3}
A mathematics competition has $21$ girls and $21$ boys as contestants. No contestant solves more than six problems, and every pair consisting of a girl and a boy solves at least one common problem. Prove that some problem is solved by at least three girls and at least three boys.
\end{problem}

\begin{problem}{2001}{4}
Let $n>1$ be an odd integer, and fix integers $k_1, \dots, k_n$. For every permutation $a=(a_1, \dots, a_n)$ of $1, \dots, n$, define
\[
    S(a)\coloneqq\sum_{i=1}^n k_i a_i.
\]
Prove that two distinct permutations $b, c$ satisfy $n!\mid S(b)-S(c)$.
\end{problem}

\begin{problem}{2001}{5}
In triangle $ABC$, the internal bisectors of $\angle BAC$ and $\angle ABC$ meet the sides $BC$ and $CA$ at $P$ and $Q$, respectively. Assume that $m\angle BAC=60^\circ$ and $AB+BP=AQ+QB$. Determine the possible angle measures of triangle $ABC$.
\end{problem}

\begin{problem}{2001}{6}
Suppose integers $a>b>c>d>0$ satisfy
\[
    ac+bd=(b+d+a-c)(b+d-a+c).
\]
Prove that $ab+cd$ is not prime.
\end{problem}
