\competitionyear{2002}

\begin{problem}{2002}{1}
Let $S=\{(h, k)\in\Nzero^2\mid h+k<n\}$. Colour every element of $S$ red or blue, with the rule that whenever $(h, k)$ is red, every $(h', k')\in S$ satisfying $h'\leq h$ and $k'\leq k$ is red as well. A subset of $S$ is of type 1 if it consists of $n$ blue elements with pairwise distinct first coordinates; it is of type 2 if it consists of $n$ blue elements with pairwise distinct second coordinates. Prove that the two types occur equally often.
\end{problem}

\begin{problem}{2002}{2}
Let $BC$ be a diameter of a circle centred at $O$. Choose a point $A$ on the circle with $m\angle AOC>60^\circ$. Let $EF$ be the chord on the perpendicular bisector of $\overline{AO}$, and let $D$ be the midpoint of $\arc{AB}$. The line through $O$ parallel to the line $AD$ meets the line $AC$ at $J$. Prove that $J$ is the incentre of triangle $CEF$.
\end{problem}

\begin{problem}{2002}{3}
Determine all integer pairs $(m, n)$ with $m, n>2$ for which
\[
    k^n+k^2-1\mid k^m+k-1
\]
holds for infinitely many positive integers $k$.
\end{problem}

\begin{problem}{2002}{4}
For an integer $n>1$, list its positive divisors as $1=d_1<d_2<\cdots<d_k=n$. Define
\[
    d\coloneqq d_1d_2+d_2d_3+\cdots+d_{k-1}d_k.
\]
Prove that $d<n^2$, and determine all $n$ for which $d\mid n^2$.
\end{problem}

\begin{problem}{2002}{5}
Determine all functions $f\colon\R\to\R$ satisfying
\[
    (f(x)+f(y))(f(u)+f(v))=f(xu-yv)+f(xv+yu)
\]
for all $x, y, u, v\in\R$.
\end{problem}

\begin{problem}{2002}{6}
Let $n>2$ circles of radius $1$ in the plane have centres $O_1, \dots, O_n$. Suppose that no line meets more than two of the circles. Prove that
\[
    \sum_{1\leq i<j\leq n}\frac{1}{O_iO_j}\leq\frac{(n-1)\pi}{4}.
\]
\end{problem}
