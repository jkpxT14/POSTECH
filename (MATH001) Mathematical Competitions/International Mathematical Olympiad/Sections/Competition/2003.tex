\competitionyear{2003}

\begin{problem}{2003}{1}
Let $S=\{1, 2, \dots, 1000000\}$. Prove that every $101$-element subset $A$ of $S$ admits $100$ distinct elements $x_1, \dots, x_{100}\in S$ such that the sets $\{a+x_i\mid a\in A\}$ are pairwise disjoint.
\end{problem}

\begin{problem}{2003}{2}
Determine all pairs $(m, n)$ of positive integers for which
\[
    \frac{m^2}{2mn^2-n^3+1}
\]
is a positive integer.
\end{problem}

\begin{problem}{2003}{3}
Suppose that, in a convex hexagon, the distance between the midpoints of each pair of opposite sides equals $\frac{\sqrt3}{2}$ times the sum of their lengths. Prove that all six interior angles have equal measure.
\end{problem}

\begin{problem}{2003}{4}
Let $ABCD$ be a cyclic quadrilateral. Denote the feet of the perpendiculars from $D$ to the lines $AB$, $BC$, and $CA$ by $P$, $Q$, and $R$, respectively. Prove that the angle bisectors of $\angle ABC$ and $\angle CDA$ meet on the line $AC$ if and only if $RP=RQ$.
\end{problem}

\begin{problem}{2003}{5}
Let $n>2$ and let $x_1\leq x_2\leq\cdots\leq x_n$ be real numbers. Prove that
\[
    \left(\sum_{i,j=1}^n |x_i-x_j|\right)^2
    \leq\frac{2}{3}(n^2-1)\sum_{i,j=1}^n(x_i-x_j)^2.
\]
Prove also that equality holds precisely when $(x_i)_{i=1}^n$ is an arithmetic progression.
\end{problem}

\begin{problem}{2003}{6}
Prove that for every prime $p$ there is a prime $q$ such that $q\nmid n^p-p$ for every positive integer $n$.
\end{problem}
