\competitionyear{2005}

\begin{problem}{2005}{1}
On the sides of an equilateral triangle $ABC$, choose $A_1, A_2$ on $BC$, $B_1, B_2$ on $CA$, and $C_1, C_2$ on $AB$, so that $A_1A_2B_1B_2C_1C_2$ is a convex hexagon with all side lengths equal. Prove that the lines $A_1B_2$, $B_1C_2$, and $C_1A_2$ are concurrent.
\end{problem}

\begin{problem}{2005}{2}
Let $(a_n)_{n\geq1}$ be a sequence of integers containing infinitely many positive terms and infinitely many negative terms. Suppose that for every positive integer $n$, the first $n$ terms represent distinct residue classes modulo $n$. Prove that each integer occurs in the sequence exactly once.
\end{problem}

\begin{problem}{2005}{3}
For positive real numbers $x, y, z$ satisfying $xyz\geq1$, prove that
\[
    \frac{x^5-x^2}{x^5+y^2+z^2}+\frac{y^5-y^2}{x^2+y^5+z^2}
    +\frac{z^5-z^2}{x^2+y^2+z^5}\geq0.
\]
\end{problem}

\begin{problem}{2005}{4}
Determine all positive integers that are relatively prime to every term of the sequence $(a_n)_{n\geq1}$ defined by
\[
    a_n\coloneqq2^n+3^n+6^n-1.
\]
\end{problem}

\begin{problem}{2005}{5}
Fix a convex quadrilateral $ABCD$ with $BC=DA$ and with the sides $BC$ and $DA$ not parallel. Let $E$ and $F$ vary on the sides $BC$ and $DA$, respectively, subject to $BE=DF$. The lines $AC$ and $BD$ intersect at $P$; let $Q$ be the intersection of $BD$ and $EF$, and $R$ the intersection of $EF$ and $AC$. Prove that the circumcircles of triangles $PQR$, as $E$ and $F$ vary, share a point distinct from $P$.
\end{problem}

\begin{problem}{2005}{6}
A competition consists of six problems. For every pair of problems, more than $\frac25$ of the contestants solve both, and nobody solves all six. Prove that at least two contestants solve exactly five problems.
\end{problem}
