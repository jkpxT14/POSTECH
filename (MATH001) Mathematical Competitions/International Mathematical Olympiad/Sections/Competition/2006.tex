\competitionyear{2006}

\begin{problem}{2006}{1}
Let $I$ be the incentre of triangle $ABC$. An interior point $P$ satisfies
\[
    m\angle PBA+m\angle PCA=m\angle PBC+m\angle PCB.
\]
Prove that $AP\geq AI$, with equality if and only if $P=I$.
\end{problem}

\begin{problem}{2006}{2}
Let $P$ be a regular $2006$-gon. Call a diagonal \emph{good} when the two boundary paths between its endpoints each consist of an odd number of sides; also call every side of $P$ good. Divide $P$ into triangles using $2003$ diagonals, no two of which meet inside $P$. Determine the largest possible number of isosceles triangles in such a division that have two good sides.
\end{problem}

\begin{problem}{2006}{3}
Find the least real constant $M$ for which
\[
    \left|ab(a^2-b^2)+bc(b^2-c^2)+ca(c^2-a^2)\right|
    \leq M(a^2+b^2+c^2)^2
\]
for all $a, b, c\in\R$.
\end{problem}

\begin{problem}{2006}{4}
Determine all integer pairs $(x, y)$ satisfying
\[
    1+2^x+2^{2x+1}=y^2.
\]
\end{problem}

\begin{problem}{2006}{5}
Let $P$ be a polynomial with integer coefficients and degree $n>1$, and let $k$ be a positive integer. Define $Q\coloneqq P^{\circ k}$. Prove that at most $n$ integers $t$ satisfy $Q(t)=t$.
\end{problem}

\begin{problem}{2006}{6}
For every side $b$ of a convex polygon $P$, assign the largest area of a triangle contained in $P$ and having $b$ as one of its sides. Prove that the sum of these assigned areas is at least twice the area of $P$.
\end{problem}
