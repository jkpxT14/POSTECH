\competitionyear{2007}

\begin{problem}{2007}{1}
Given real numbers $a_1, \dots, a_n$, define, for $1\leq i\leq n$,
\[
    d_i\coloneqq\max\{a_j\mid1\leq j\leq i\}-\min\{a_j\mid i\leq j\leq n\},
\]
and let $d\coloneqq\max\{d_i\mid1\leq i\leq n\}$.
\begin{problemsubparts}
    \item Prove that whenever $x_1\leq\cdots\leq x_n$ are real numbers,
    \[
        \max\{|x_i-a_i|\mid1\leq i\leq n\}\geq\frac d2.\tag{*}
    \]
    \item Prove that real numbers $x_1\leq\cdots\leq x_n$ can be chosen to attain equality in $(*)$.
\end{problemsubparts}
\end{problem}

\begin{problem}{2007}{2}
Let $ABCD$ be a parallelogram and let $E$ be such that $BCED$ is a cyclic quadrilateral. A line $\ell$ through $A$ meets the interior of $\overline{DC}$ at $F$ and the line $BC$ at $G$. Suppose $EF=EG=EC$. Prove that $\ell$ bisects $\angle DAB$.
\end{problem}

\begin{problem}{2007}{3}
Among the competitors in a mathematics contest, friendship is mutual. A group is called a \emph{clique} if every pair of distinct members are friends; groups with fewer than two members are included. Its size is its number of members. Suppose that the maximum size of a clique among all competitors is even. Prove that the competitors can be divided between two rooms so that the maximum clique size in each room is the same.
\end{problem}

\begin{problem}{2007}{4}
In triangle $ABC$, let the bisector of $\angle BCA$ meet the circumcircle again at $R$, the perpendicular bisector of $\overline{BC}$ at $P$, and the perpendicular bisector of $\overline{AC}$ at $Q$. Let $K$ and $L$ be the midpoints of $BC$ and $AC$, respectively. Prove that $[RPK]=[RQL]$.
\end{problem}

\begin{problem}{2007}{5}
Let $a, b$ be positive integers satisfying
\[
    4ab-1\mid(4a^2-1)^2.
\]
Prove that $a=b$.
\end{problem}

\begin{problem}{2007}{6}
For a positive integer $n$, consider the following set of $(n+1)^3-1$ points in $\R^3$:
\[
    S=\{(x, y, z)\mid x, y, z\in\{0, 1, \dots, n\},\ x+y+z>0\}.
\]
Find the minimum number of planes whose union contains $S$ but excludes $(0, 0, 0)$.
\end{problem}
