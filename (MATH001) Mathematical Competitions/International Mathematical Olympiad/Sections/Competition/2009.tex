\competitionyear{2009}

\begin{problem}{2009}{1}
Let $n$ be a positive integer, and let $a_1, \dots, a_k$ be distinct elements of $\{1, \dots, n\}$, where $k\geq2$. Suppose that $n\mid a_i(a_{i+1}-1)$ for $i=1, \dots, k-1$. Prove that $n\nmid a_k(a_1-1)$.
\end{problem}

\begin{problem}{2009}{2}
Let $ABC$ be a triangle with circumcentre $O$, and let $P$ and $Q$ lie in the interiors of sides $CA$ and $AB$, respectively. Let $K$, $L$, and $M$ be the midpoints of $BP$, $CQ$, and $PQ$, respectively. If the line $PQ$ is tangent to the circle $(KLM)$, prove that $OP=OQ$.
\end{problem}

\begin{problem}{2009}{3}
Let $(s_n)_{n\geq1}$ be a strictly increasing sequence of positive integers. Suppose that both subsequences $(s_{s_n})_{n\geq1}$ and $(s_{s_n+1})_{n\geq1}$ are arithmetic progressions. Prove that $(s_n)_{n\geq1}$ is an arithmetic progression.
\end{problem}

\begin{problem}{2009}{4}
Let $ABC$ be a triangle with $AB=AC$. The angle bisectors of $\angle CAB$ and $\angle ABC$ meet the sides $BC$ and $CA$ at $D$ and $E$, respectively. Let $K$ be the incentre of triangle $ADC$. Suppose $m\angle BEK=45^\circ$. Determine all possible values of $m\angle CAB$.
\end{problem}

\begin{problem}{2009}{5}
Determine all functions $f\colon\N\to\N$ for which, for every $a, b\in\N$, the three numbers $a$, $f(b)$, and $f(b+f(a)-1)$ are the side lengths of a nondegenerate triangle (one whose vertices are not collinear).
\end{problem}

\begin{problem}{2009}{6}
Let $a_1, \dots, a_n$ be distinct positive integers, and let $M$ be a set of $n-1$ positive integers not containing $s=a_1+\cdots+a_n$. A grasshopper starts at $0$ on the real axis and makes $n$ jumps to the right, using each length $a_1, \dots, a_n$ exactly once. Prove that the lengths can be ordered so that none of its landing points belongs to $M$.
\end{problem}
