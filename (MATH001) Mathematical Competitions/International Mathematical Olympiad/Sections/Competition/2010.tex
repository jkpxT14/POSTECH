\competitionyear{2010}

\begin{problem}{2010}{1}
Determine all functions $f\colon\R\to\R$ for which
\[
    f(\lfloor x\rfloor y)=f(x)\lfloor f(y)\rfloor
\]
for every $x, y\in\R$.
\end{problem}

\begin{problem}{2010}{2}
Let $I$ be the incentre of triangle $ABC$, and let $\Gamma$ be the circumcircle of $ABC$. The line $AI$ meets $\Gamma$ again at $D$. Choose $E$ on the arc $\widehat{BDC}$ and $F$ on the side $BC$ such that
\[
    \angle BAF=\angle CAE<\frac12\angle BAC.
\]
Let $G$ be the midpoint of $IF$. Prove that the lines $DG$ and $EI$ intersect at a point of $\Gamma$.
\end{problem}

\begin{problem}{2010}{3}
Find all functions $g\colon\N\to\N$ such that
\[
    (g(m)+n)(m+g(n))
\]
is a perfect square for every $m, n\in\N$.
\end{problem}

\begin{problem}{2010}{4}
Let $P$ lie inside triangle $ABC$. The lines $AP$, $BP$, and $CP$ meet the circumcircle $\Gamma$ of $ABC$ again at $K$, $L$, and $M$, respectively. The tangent to $\Gamma$ at $C$ meets $AB$ at $S$. If $SC=SP$, prove that $MK=ML$.
\end{problem}

\begin{problem}{2010}{5}
Initially, each of six boxes $B_1, B_2, \dots, B_6$ contains one coin. The following operations are permitted:
\begin{problemchoices}
    \item Choose a nonempty box $B_j$, where $1\le j\le5$. Remove one coin from $B_j$ and place two coins in $B_{j+1}$.
    \item Choose a nonempty box $B_k$, where $1\le k\le4$. Remove one coin from $B_k$, then exchange the contents of $B_{k+1}$ and $B_{k+2}$, which may be empty.
\end{problemchoices}
Determine whether a finite sequence of these operations can leave $B_1, B_2, B_3, B_4, B_5$ empty and leave exactly $2010^{(2010^{2010})}$ coins in $B_6$.
\end{problem}

\begin{problem}{2010}{6}
Let $(a_n)_{n\ge1}$ be a sequence of positive real numbers. Suppose that there is a positive integer $s$ such that
\[
    a_n=\max\{a_k+a_{n-k}\mid1\le k\le n-1\}
\]
for every $n>s$. Prove that there exist positive integers $\ell, N$ with $\ell\le s$ such that
\[
    a_n=a_\ell+a_{n-\ell}
\]
for all $n\ge N$.
\end{problem}
