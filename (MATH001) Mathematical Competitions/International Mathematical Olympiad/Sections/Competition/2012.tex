\competitionyear{2012}

\begin{problem}{2012}{1}
In triangle $ABC$, let $J$ be the centre of the excircle opposite $A$. This excircle touches $BC$ at $M$, and the lines $AB$ and $AC$ at $K$ and $L$, respectively. Let the lines $LM$ and $BJ$ intersect at $F$, and the lines $KM$ and $CJ$ at $G$. Let the lines $AF$ and $BC$ intersect at $S$, and the lines $AG$ and $BC$ at $T$. Prove that $M$ is the midpoint of $ST$.

The excircle opposite $A$ is tangent to the segment $BC$, the ray $AB$ beyond $B$, and the ray $AC$ beyond $C$.
\end{problem}

\begin{problem}{2012}{2}
Let $n\ge3$ be an integer and let $a_2, a_3, \dots, a_n$ be positive real numbers with $a_2a_3\cdots a_n=1$. Prove that
\[
    (1+a_2)^2(1+a_3)^3\cdots(1+a_n)^n>n^n.
\]
\end{problem}

\begin{problem}{2012}{3}
In the \emph{liar's guessing game}, two players $A$ and $B$ know positive integers $k$ and $n$. Player $A$ secretly chooses integers $x, N$ with $1\le x\le N$ and tells $B$ the true value of $N$, but not $x$. Player $B$ may repeatedly name any set $S$ of positive integers and ask whether $x\in S$, including repeating a previous question. Player $A$ answers each question immediately with yes or no; lying is allowed, provided that among every $k+1$ consecutive answers at least one is truthful.

After any number of questions, $B$ chooses a set $X$ of at most $n$ positive integers. Player $B$ wins exactly when $x\in X$. Prove that:
\begin{problemsubparts}
    \item If $n\ge2^k$, then $B$ can guarantee a win.
    \item For all sufficiently large $k$, some integer $n\ge1.99^k$ does not allow $B$ to guarantee a win.
\end{problemsubparts}
\end{problem}

\begin{problem}{2012}{4}
Determine all functions $f\colon\Z\to\Z$ satisfying
\[
    f(a)^2+f(b)^2+f(c)^2
    =2f(a)f(b)+2f(b)f(c)+2f(c)f(a)
\]
for every triple of integers $a, b, c$ with $a+b+c=0$.
\end{problem}

\begin{problem}{2012}{5}
Let $ABC$ be a right triangle with $\angle BCA=90^\circ$, and let $D$ be the foot of the altitude from $C$. Choose $X$ in the interior of $CD$. Let $K$ be on the segment $AX$ with $BK=BC$, and $L$ on the segment $BX$ with $AL=AC$. If the lines $AL$ and $BK$ meet at $M$, prove that $MK=ML$.
\end{problem}

\begin{problem}{2012}{6}
Determine all positive integers $n$ for which some nonnegative integers $a_1, \dots, a_n$ satisfy
\[
    \frac1{2^{a_1}}+\frac1{2^{a_2}}+\cdots+\frac1{2^{a_n}}
    =\frac1{3^{a_1}}+\frac2{3^{a_2}}+\cdots+\frac n{3^{a_n}}=1.
\]
\end{problem}
