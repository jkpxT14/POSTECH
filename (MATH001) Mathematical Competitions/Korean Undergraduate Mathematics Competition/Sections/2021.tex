\competitionyear{2021}

\begin{problem}{2021}{2}{1}
Evaluate
\[
    \lim_{x\to0}
    \frac{e^{x^2}-e^{-x^2}}{1-\cos x}.
\]
\end{problem}

\begin{problem}{2021}{2}{2}
Solve the differential equation
\[
    2y y''+2(y')^2-8y y'+3y^2=0,
    \qquad
    y(0)=2,
    \qquad
    y'(0)=3.
\]
\end{problem}

\begin{problem}{2021}{2}{3}
Determine all integers $n\ge2$ and all real numbers
$\alpha_1,\dots,\alpha_n,\beta_1,\dots,\beta_n$ for which the matrix
\[
    \begin{pmatrix}
        \cos(\alpha_1-\beta_1) & \cos(\alpha_1-\beta_2) & \cdots & \cos(\alpha_1-\beta_n)\\
        \cos(\alpha_2-\beta_1) & \cos(\alpha_2-\beta_2) & \cdots & \cos(\alpha_2-\beta_n)\\
        \vdots & \vdots & \ddots & \vdots\\
        \cos(\alpha_n-\beta_1) & \cos(\alpha_n-\beta_2) & \cdots & \cos(\alpha_n-\beta_n)
    \end{pmatrix}
\]
is invertible.
\end{problem}

\begin{problem}{2021}{2}{4}
Let
\[
    D=\{(x,y)\in\R^2\mid x>0,\ y>0,\ 1-x-y>0\},
\]
and define $f\colon D\to\R$ by
\[
    f(x,y)
    =4\bigl(x^2+y^2+(1-x-y)^2\bigr)
    -x\ln x-y\ln y-(1-x-y)\ln(1-x-y).
\]
Find the number of critical points of $f$.
\end{problem}

\begin{problem}{2021}{2}{5}
Evaluate
\[
    \int_0^1\int_0^1\int_0^1
    \lfloor x+y+z\rfloor\,\dd x\,\dd y\,\dd z.
\]
Here $\lfloor t\rfloor$ denotes the greatest integer not exceeding $t$.
\end{problem}

\begin{problem}{2021}{2}{6}
Let $\mathbf A=(a_{ij})$ be an invertible $n\times n$ matrix satisfying, for every integer $k$ with $1\le k\le n$,
\[
    \sum_{i=1}^n a_{ik}
    =
    \sum_{j=1}^n a_{kj}
    =2021.
\]
If $\mathbf B=(b_{ij})=\mathbf A^{-1}$, prove that, for every integer $k$ with $1\le k\le n$,
\[
    \sum_{i=1}^n b_{ik}
    =
    \sum_{j=1}^n b_{kj}
    =\frac1{2021}.
\]
\end{problem}

\begin{problem}{2021}{2}{7}
For real numbers $a,b$, let
\[
    \mathbf A=
    \begin{pmatrix}
        a&1&1&1\\
        1&b&1&1\\
        1&1&a&1\\
        1&1&1&b
    \end{pmatrix}.
\]
Prove that $\mathbf A$ is positive definite if and only if $a>1$ and $b>1$.
\end{problem}

\begin{problem}{2021}{2}{8}
Let $f\colon\R\to\R$ be twice differentiable and satisfy
\[
    \lim_{x\to\infty}f(x)=0.
\]
Prove that
\[
    \int_0^\infty x^4 f(x)^4\,\dd x
    \le
    2\left(\int_0^\infty x^2f(x)^2\,\dd x\right)^{3/2}
    \left(\int_0^\infty x^2f'(x)^2\,\dd x\right)^{1/2}.
\]
\end{problem}

\begin{problem}{2021}{1}{2}
Let $M_n(\R)$ be the real vector space of all $n\times n$ real matrices. For $\mathbf A\in M_n(\R)$, define the linear map
\[
    \Phi_{\mathbf A}\colon M_n(\R)\to M_n(\R),
    \qquad
    \Phi_{\mathbf A}(\mathbf X)=\mathbf A\mathbf X-\mathbf X\mathbf A.
\]
Find $\det(\Phi_{\mathbf A})$ and $\tr(\Phi_{\mathbf A})$.
\end{problem}

\begin{problem}{2021}{1}{4}
Let $\mathbf A=(a_{ij})$ be an $n\times n$ real symmetric matrix, where $n\ge2$. Suppose that
\[
    a_{ij}\le0
    \qquad\text{for all }i\ne j,
\]
and that every eigenvalue of $\mathbf A$ is nonnegative.
\begin{problemsubparts}
    \item Prove that
    \[
        \ker\mathbf A
        =
        \{\mathbf v\in\R^n\mid \mathbf v\cdot(\mathbf A\mathbf v)=0\}.
    \]
    \item Suppose that for every proper partition $I,J$ of $\{1,2,\dots,n\}$, there exist $i\in I$ and $j\in J$ such that $a_{ij}\ne0$. Prove that
    \[
        \dim\ker\mathbf A\le1.
    \]
\end{problemsubparts}
Here a proper partition $I,J$ means that $I$ and $J$ are nonempty,
\[
    I\cup J=\{1,2,\dots,n\},
    \qquad
    I\cap J=\varnothing.
\]
\end{problem}

\begin{problem}{2021}{1}{6}
Let
\[
    S=\{\mathbf A_1,\mathbf A_2,\dots,\mathbf A_k\}
\]
be a set of $k$ distinct invertible $n\times n$ matrices, where $n,k\ge2$. Suppose that for every $\mathbf A_i\in S$,
\[
    S=\{\mathbf A_1\mathbf A_i,\mathbf A_2\mathbf A_i,\dots,\mathbf A_k\mathbf A_i\}.
\]
Prove that there exists $\mathbf A_t\in S$ such that
\[
    (\mathbf A_1+\mathbf A_2+\cdots+\mathbf A_k)-\mathbf A_t
\]
is invertible.
\end{problem}

\begin{problem}{2021}{1}{7}
Let $\mathbf A$ and $\mathbf B$ be $3\times3$ orthogonal matrices. Suppose that each of $\mathbf A$ and $\mathbf B$ has two eigenvectors with eigenvalue $-1$ and at least three eigenvectors with eigenvalue $1$ in the finite set
\[
    \{(x,y,z)\in\R^3\mid x,y,z\in\Z,\ x^2+y^2+z^2\le2\}.
\]
Prove that at least one of
\[
    (\mathbf A\mathbf B)^2,
    \qquad
    (\mathbf A\mathbf B)^3,
    \qquad
    (\mathbf A\mathbf B)^4
\]
is the identity matrix $\mathbf I_3$.
\end{problem}
