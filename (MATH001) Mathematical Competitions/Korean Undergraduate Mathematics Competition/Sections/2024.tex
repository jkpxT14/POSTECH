\competitionyear{2024}

\begin{problem}{2024}{2}{1}
Let
\[
    \mathbf A=
    \begin{pmatrix}
        0&1&1\\
        -1&0&1\\
        -1&-1&0
    \end{pmatrix}.
\]
Find real numbers $a,b,c$ such that
\[
    \mathbf A^5=a\mathbf A^2+b\mathbf A+c\mathbf I,
\]
where $\mathbf I$ is the $3\times3$ identity matrix.
\end{problem}

\begin{problem}{2024}{2}{2}
Solve the differential equation
\[
    \frac{\dd y}{\dd x}
    =-\frac{x+y-2}{x-y},
    \qquad
    y(1)=0.
\]
\end{problem}

\begin{problem}{2024}{2}{3}
For $a>0$, define the solid
\[
    V=
    \left\{
        (x,y,z)\in\R^3\mid
        x^2+y^2+(z-1)^2\le1,
        \quad
        z\ge\sqrt{ax^2+ay^2}
    \right\}.
\]
If the volume of $V$ is
\[
    \frac{7\pi}{12},
\]
find $a$.
\end{problem}

\begin{problem}{2024}{2}{4}
Find the real number $N$ satisfying
\[
    \sum_{k=1}^{2024}
    (2025-k)\,2024^k\,2025^{2024-k}
    =2024^N.
\]
\end{problem}

\begin{problem}{2024}{2}{5}
Evaluate the improper integral
\[
    \int_0^\infty
    \frac{1}{(x^{2024}+1)(x^2+1)}\,\dd x.
\]
\end{problem}

\begin{problem}{2024}{2}{6}
Let $\mathbf A$ and $\mathbf B$ be $n\times n$ real matrices, where $n\ge2$, and suppose that
\[
    \rk(\mathbf B)=1.
\]
Prove that
\[
    \det(\mathbf A+\mathbf B)\det(\mathbf A-\mathbf B)
    \le
    \det(\mathbf A)^2.
\]
\end{problem}

\begin{problem}{2024}{2}{7}
Let $D$ be a bounded closed convex region in the coordinate plane, and let $\beta$ be the boundary of $D$. Suppose that:
\begin{problemclauses}
    \item $\beta$ is a smooth simple closed curve.
    \item For every point $O$ on $\beta$, the circle of radius $1$ centered at $O$ meets $\beta$ at exactly two points.
    \item For every point $O$ on $\beta$, the tangent line to $\beta$ at $O$ does not meet $\beta$ at any point other than $O$.
\end{problemclauses}
Let $M$ be the midpoint of a rod $PQ$ of length $1$. Move the rod through one full turn while keeping both endpoints $P,Q$ on $\beta$ at all times; that is, move $P$ and $Q$ continuously once around $\beta$. Suppose that the locus of $M$ is a simple closed curve $\gamma$. Prove that the area of the region between $\beta$ and $\gamma$ is
\[
    \frac\pi4.
\]
\end{problem}

\begin{problem}{2024}{1}{5}
Let $f\colon\R\to\R$ be twice differentiable, with $f''$ continuous. Suppose that for every real $x$,
\[
    f'(x)>0
    \qquad\text{and}\qquad
    f''(x)<0.
\]
Prove that for every $t>0$,
\[
    \left|
        \int_0^t\cos(f(x))\,\dd x
    \right|
    \le
    \frac{2}{f'(t)}.
\]
\end{problem}

\begin{problem}{2024}{1}{6}
Let $n\ge3$, and let real numbers $a_1,a_2,\dots,a_n$ satisfy
\[
    a_1+a_2+\cdots+a_n=0,
    \qquad
    a_1^2+a_2^2+\cdots+a_n^2=1.
\]
Find the maximum value of
\[
    \left(\sum_{k=1}^{n-1}a_ka_{k+1}\right)+a_na_1.
\]
\end{problem}
