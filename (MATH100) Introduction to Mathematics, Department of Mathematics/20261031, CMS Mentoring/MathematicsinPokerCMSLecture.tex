\begin{frame}{Flop — 승리 확률의 하한}
 \vspace{.11cm}
 \begin{center}{\large 남은 두 장 중 적어도 한 장이 하트일 확률을 구하시오.}\end{center}
 \vspace{.16cm}
 \begin{overlayarea}{\textwidth}{3.12cm}\centering
 \only<2>{\normalsize 가능한 카드는 45장이고, 그중 하트는 9장이다.\[\Prob(\text{하트가 나오지 않음})=\frac{36}{45}\cdot\frac{35}{44}.\]}
 \only<3->{\[\begin{aligned}
 \Prob(\text{플러시 완성})
 &=1-\frac{36}{45}\cdot\frac{35}{44}\\[5pt]
 &=\important{\frac4{11}\approx36.36\%}.
 \end{aligned}\]}
 \end{overlayarea}
 \vfill\begin{center}\subnote{두 상대 패에는 하트가 없다. 플러시가 완성되면 A가 반드시 승리한다. 따라서 $4/11$은 승리 확률의 하한이다.}\end{center}
\end{frame}

  \bfseries 리버의 행동 & \bfseries 총 팟 & \bfseries Call 비용 & \bfseries 승리 순이익 \\ \hline
  B가 베팅함 & 100점 & 20점 & $+80$점 \\
  B가 베팅하지 않음 & 80점 & 10점 & $+70$점
 \end{tabular}
 \end{center}
 \vspace{.24cm}
 \begin{overlayarea}{\textwidth}{1.00cm}\centering
  \only<2->{\large 승리 시 최소 $+70$점, 패배 시 최대 $20$점 손실이다.}
 \end{overlayarea}
\end{frame}

\begin{frame}{Turn — 두 번째 선택}
 \vspace{.00cm}
 \begin{center}{\large 현재 팟 60점 · B의 베팅 10점 · Call 비용 10점}\end{center}
 \vspace{.23cm}
 \begin{center}{\normalsize 리버에서 B가 베팅하면 한 번 더 Call하는 전략을 비교한다.}\end{center}
 \vspace{.16cm}
 \begin{columns}[c,onlytextwidth]
  \begin{column}{.49\textwidth}\centering 승리 시 추가 순이익\par\vspace{.12cm}{\Large$\geq+70$점}\end{column}
  \begin{column}{.49\textwidth}\centering 패배 시 추가 순이익\par\vspace{.12cm}{\Large$\geq-20$점}\end{column}
 \end{columns}
 \begin{overlayarea}{\textwidth}{2.18cm}\centering
  \only<2>{\[\Exp[X_{\mathrm T}]\geq70p-20(1-p).\]}
  \only<3->{\[\Exp[X_{\mathrm T}]\geq70\cdot\frac{39}{110}-20\cdot\frac{71}{110}=\important{\frac{131}{11}>0}.\]}
 \end{overlayarea}
 \only<4->{\result{결론: Turn에서 Call}}
\end{frame}

  \only<3->{\normalsize 리버에서는 더 이상 카드가 공개되지 않는다. 베팅한 상대의 패를 다시 평가해야 한다.}
 \end{overlayarea}
\end{frame}

\begin{frame}{River — 레인지 갱신}
 \vspace{.07cm}
 \begin{center}{\large B가 베팅했다는 정보를 이용한다.}\end{center}
 \vspace{.18cm}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{.49\textwidth}\centering
   {\normalsize 가능성 I의 베팅 확률: $1$}\par\vspace{.17cm}
   \only<2->{\[\frac45\cdot1=\frac45\]}
  \end{column}
  \begin{column}{.49\textwidth}\centering
   {\normalsize 가능성 II의 베팅 확률: $\frac23$}\par\vspace{.17cm}
   \only<2->{\[\frac15\cdot\frac23=\frac2{15}\]}
  \end{column}
 \end{columns}
 \vspace{.15cm}
 \begin{overlayarea}{\textwidth}{2.24cm}\centering
  \only<3>{\[\Prob(\text{II}\mid\text{Bet})=\frac{\frac2{15}}{\frac45+\frac2{15}}.\]}
  \only<4->{\[\Prob(\text{II}\mid\text{Bet})=\important{\frac17},\qquad
                 \Prob(\text{I}\mid\text{Bet})=\frac67.\]}
 \end{overlayarea}
\end{frame}

\begin{frame}{River — 승리 가능성과 손익}
 \vspace{.20cm}
 \begin{center}{\large 최종 보드: 6$\heartsuit$\,7$\heartsuit$\,Q$\clubsuit$\,8$\spadesuit$\,2$\diamondsuit$}\end{center}
 \vspace{.20cm}
 \begin{columns}[T,onlytextwidth]
  \begin{column}{.49\textwidth}\centering
   {\large 가능성 I: $\frac67$}\par\vspace{.12cm}
   \card[.82]{10}{D}\cgap\card[.82]{9}{S}\par\vspace{.15cm}
   {\normalsize B 승리 (스트레이트)}
  \end{column}
  \begin{column}{.49\textwidth}\centering
   {\large 가능성 II: $\frac17$}\par\vspace{.12cm}
   \card[.82]{K}{D}\cgap\card[.82]{J}{S}\par\vspace{.15cm}
   {\normalsize A 승리 (A 하이 카드)}
  \end{column}
 \end{columns}
 \vfill\begin{center}{\large 따라서 $\displaystyle \Prob(\text{A 승리}\mid\text{Bet})=\important{\frac17}$.}\end{center}
\end{frame}

\begin{frame}{River — 세 번째 선택}
 \vspace{.00cm}
 \begin{center}{\large 현재 팟 80점 · B의 베팅 10점 · Call 비용 10점}\end{center}
 \vspace{.20cm}
 \begin{center}{\normalsize 최종 팟 $100$점, Call 비용 $10$점: 승리 시 $100-10=+90$점, 패배 시 $-10$점}\end{center}
 \begin{overlayarea}{\textwidth}{2.76cm}\centering
  \only<2>{\[\Exp[X_{\mathrm R}]=90p-10(1-p).\]}
  \only<3>{\[\Exp[X_{\mathrm R}]>0\quad\Longleftrightarrow\quad p>\frac1{10}.\]}
  \only<4->{\[\Exp[X_{\mathrm R}]=90\cdot\frac17-10\cdot\frac67=\important{\frac{30}{7}>0}.\]}
 \end{overlayarea}
 \only<5->{\result{결론: River에서 Call}}
\end{frame}
