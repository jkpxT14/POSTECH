% Flop: count only sufficient winning cases; no brute-force table.
% With both private hands and the flop known, there are C(45,2)=990
% equally likely unordered turn/river pairs under either H=h_1 or H=h_2.
% Exactly 360 contain at least one heart. For h_1, exactly 13 of those
% lose to a full house or four-of-a-kind:
% J-heart with (three 8s, three 9s, J-diamond, three remaining Ks): 10;
% K-heart with (J-diamond, K-club, K-spade): 3.
% For h_2 there are zero exceptions. Hence P(W)>=874/2475.
\begin{frame}{플랍 — 승리 확률의 하한}
 \vspace{.12cm}
 \begin{center}{\large 적어도 플러시가 완성되고 상대가 역전하지 못하면 승리한다.}\end{center}
 \begin{overlayarea}{\textwidth}{3.38cm}\centering
  \only<2>{\[\begin{aligned}
    \Prob(\text{하트가 한 장 이상})
       &=1-\frac{36}{45}\cdot\frac{35}{44}\\[5pt]
       &=\frac4{11}=\frac{360}{990}.
   \end{aligned}\]}
  \only<3>{\normalsize 가능성 I에서는 그중 $13$가지 조합에서 B가 풀하우스 또는 포카드를 만든다.\par\vspace{.20cm}
    $J\heartsuit$ 포함: $3+3+1+3=10$가지\par\vspace{.13cm}
    $J\heartsuit$ 없이 $K\heartsuit$ 포함: $1+2=3$가지\par\vspace{.15cm}
    {\small 가능성 II에서는 하트가 나오면 항상 A가 승리한다.}}
  \only<4->{\[\begin{aligned}
    \Prob(W)&\geq\frac45\cdot\frac{360-13}{990}
                     +\frac15\cdot\frac{360}{990}\\[6pt]
            &=\redans{\frac{874}{2475}\approx35.31\%}.
   \end{aligned}\]}
 \end{overlayarea}
 \vfill\begin{center}\smallnote{$W$: 마지막 카드까지 공개했을 때 A가 승리하는 사건. 확실한 승리 경우만 센 하한이다.}\end{center}
\end{frame}

 \begin{overlayarea}{\textwidth}{1.73cm}\centering
  \only<2>{\large A가 패배하는 리버 카드는 무엇인가?}
  \only<3>{\large K$\diamondsuit$, K$\spadesuit$ 및 하트가 아닌 10 세 장.}

\begin{frame}{턴 — 승리 확률}

 \begin{center}{\large 두 경우의 승리 확률을 레인지의 비율로 가중평균한다.}\end{center}

    &=\redans{\frac{111}{220}\approx50.45\%}.
  \end{aligned}\]}
 \end{overlayarea}
 \vfill\begin{center}\smallnote{$p$: 리버까지 공개했을 때 A가 승리할 확률. 이 단계에서는 무승부가 없다.}\end{center}
\end{frame}

\begin{frame}{턴 — 두 번째 선택}
