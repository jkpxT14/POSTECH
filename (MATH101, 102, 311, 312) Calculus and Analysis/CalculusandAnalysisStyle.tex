  \else

  \fi
}

% Compact polar grid used only when a source problem contains polar-graph
% choices. The geometry mirrors the original exam while using the book's line
% weights and typography.
\newcommand{\polargrid}{%
  \foreach \r in {1,2,3,4}{\draw[gray!38,line width=0.25pt] (0,0) circle (\r);}%
  \foreach \a in {0,30,...,330}{\draw[gray!32,line width=0.25pt] (0,0) -- (\a:4);}%
  \foreach \r in {1,2,3,4}{\node[font=\tiny,fill=white,inner sep=0.25pt] at (0:\r) {\r};}%
  \foreach \a in {0,30,...,330}{\node[font=\tiny] at (\a:4.42) {\a};}%
}
