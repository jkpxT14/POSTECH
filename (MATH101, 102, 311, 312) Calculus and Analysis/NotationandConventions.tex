\preliminarypart{Notation and Conventions}

General mathematical notation and writing follow \textit{(MATH000) Mathematical Language}. The conventions below record only notation specific to calculus and analysis used in this book.

\notationheading{Single-Variable Calculus}

The differential $\dd$ is upright, and $\varepsilon$ is the default small positive quantity in analysis.

\begin{notationtable}
$\displaystyle\lim_{x\to a} f(x)$ & limit as $x$ approaches $a$ \\
$\min A$ & minimum of $A$ \\
$\max A$ & maximum of $A$ \\
$\inf A$ & infimum of $A$ \\
$\sup A$ & supremum of $A$ \\
$f'(a)$ & first derivative of $f$ at $a$ \\
$f^{(n)}(a)$ & $n$th derivative of $f$ at $a$ \\
$\displaystyle\frac{\dd y}{\dd x}$ & derivative of $y$ with respect to $x$ \\
$\displaystyle\int_a^b f(x)\,\dd x$ & definite integral over $[a,b]$ \\
$\ln x,\ \log_e x$ & logarithm to base $e$ \\
$\lg x,\ \log_{10}x$ & logarithm to base $10$ \\
$\lb x,\ \log_2x$ & logarithm to base $2$ \\
$\log_a x$ & logarithm to base $a$ \\
$\arcsin x$ & principal inverse sine \\
$\arccos x$ & principal inverse cosine \\
$\arctan x$ & principal inverse tangent \\
$\arcsec x$ & principal inverse secant \\
$\arccsc x$ & principal inverse cosecant \\
$\arccot x$ & principal inverse cotangent \\
$\Gamma(z)$ & Gamma function \\
$B(x,y)$ & Beta function \\
$\zeta(s)$ & Riemann zeta function \\
\end{notationtable}

\notationheading{Series and Convergence}

\begin{notationtable}
$\displaystyle\sum_{k=1}^{n} a_k$ & finite sum \\
$\displaystyle\sum_{k=1}^{\infty} a_k$ & infinite series \\
$\displaystyle\prod_{k=1}^{n} a_k$ & finite product \\
$x_n\to x$ & convergence of a sequence \\
$a_n\nearrow L$ & monotone increasing convergence to $L$ \\
$a_n\searrow L$ & monotone decreasing convergence to $L$ \\
$\limsup a_n$ & limit superior \\
$\liminf a_n$ & limit inferior \\
$f_n\to f,\ f_n\ptwlabelto f$ & pointwise convergence \\
$f_n\rightrightarrows f,\ f_n\uniflabelto f$ & uniform convergence \\
$f_n\aeto f$ & convergence almost everywhere \\
$C^k(I)$ & $k$ times continuously differentiable functions on $I$ \\
$C^\infty(I)$ & smooth functions on $I$ \\
$\|f\|_\infty$ & supremum norm \\
$B_r(x)$ & open ball of radius $r$ centered at $x$ \\
$\overline{B}_r(x)$ & closed ball of radius $r$ centered at $x$ \\
\end{notationtable}

\notationheading{Multivariable Calculus}

\begin{notationtable}
$\displaystyle\frac{\partial f}{\partial x_i},\ \partial_i f$ & partial derivative with respect to $x_i$ \\
$D\mathbf{F}(\mathbf{a})$ & total derivative; Jacobian matrix in coordinates \\
$\Jac(\mathbf{F})$ & Jacobian determinant \\
$\Hess f$ & Hessian matrix of $f$ \\
$\nabla f$ & gradient of $f$ \\
$\nabla\cdot\mathbf{F}$ & divergence of $\mathbf{F}$ \\
$\nabla\times\mathbf{F}$ & curl of $\mathbf{F}$ \\
$\Delta f$ & Laplacian of $f$ \\
$\displaystyle\iint_D f\,\dd A$ & double integral over $D$ \\
$\displaystyle\iiint_\Omega f\,\dd V$ & triple integral over $\Omega$ \\
$\mathbf{r}(t)$ & parametrized curve \\
$\dd s$ & element of arc length \\
$\displaystyle\int_C \mathbf{F}\cdot\dd\mathbf{r}$ & line integral along $C$ \\
$\displaystyle\iint_S \mathbf{F}\cdot\mathbf{n}\,\dd S$ & flux integral through $S$ \\
\end{notationtable}
