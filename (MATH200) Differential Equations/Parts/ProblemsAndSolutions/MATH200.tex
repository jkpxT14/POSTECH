\[

\end{problemsubparts}
\end{problem}

\begin{problem}[2021 Spring \KAIST{} Differential Equations and Applications]
Parabolas have the reflective property that whenever an incident ray parallel to the axis of a parabola is reflected on the parabola, then it will pass through the focus; the following figure depicts a case of the parabola $y^2=4px$ $(p>0)$.

\begin{center}
\begin{tikzpicture}[x=1.08cm,y=1.08cm,>=Latex]
  \def\p{0.55}
  \def\yP{1.55}
  \pgfmathsetmacro{\xP}{\yP*\yP/(4*\p)}
  \draw[->,line width=0.45pt] (-0.25,0) -- (3.55,0) node[right] {$x$};
  \draw[->,line width=0.45pt] (0,-2.15) -- (0,2.75) node[above] {$y$};
  \draw[KAISTBlue,line width=0.9pt,domain=-2.0:2.25,samples=100,smooth,variable=\t]
    plot ({\t*\t/(4*\p)},{\t});
  \fill (\xP,\yP) circle (1.2pt) node[above left=1pt] {$P(x_0,y_0)$};
  \fill (\p,0) circle (1.2pt) node[below=2pt] {$F(p,0)$};
  \draw[black!55,dashed,->,line width=0.65pt] (3.25,\yP) -- (\xP+0.04,\yP) node[midway,above=2pt] {$L$};
  \draw[POSTECHRed,->,line width=0.75pt] (\xP,\yP) -- (\p+0.03,0.05);
  \draw[black!45,line width=0.6pt] ({\xP-0.55},{\yP-0.37}) -- ({\xP+1.05},{\yP+0.70});
  \node[KAISTBlue] at (2.25,-1.55) {$y^2=4px$};
\end{tikzpicture}
\end{center}

If we model such a reflective property as a differential equation, then we obtain the first-order nonlinear differential equation
\[
y(y')^2-2(p-x)y'-y=0.
\]
More precisely, a solution $y=f(x)$ of the above differential equation is intended to satisfy the following: if an incident ray $L$ parallel to the $x$-axis is reflected at the point $P(x_0,y_0)$ on the graph of $y=f(x)$, then it passes through the point $F(p,0)$.
\begin{problemsubparts}
\item Verify that the parabola $y^2=4px$ satisfies the given differential equation.
\item Find a one-parameter family of solutions of the given differential equation. \textit{(Hint. Use the substitution $w=y^2$. Then differentiate the modified equation with respect to the independent variable $x$.)}
\item Characterize a common geometric property of the one-parameter family of solutions you obtained.
\end{problemsubparts}
\end{problem}
