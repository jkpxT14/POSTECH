\begin{problem}{1.1}
Let $X=\R$ and let $R=\varnothing$ be the void relation on $X$. Determine whether $R$ is reflexive, symmetric, and transitive, and hence whether $R$ is an equivalence relation. Justify each answer.
\end{problem}

\begin{solution}
The relation $R$ is not reflexive, since $(x,x)\notin R$ for every $x\in X$.

The relation $R$ is symmetric: the implication $xRy\Rightarrow yRx$ is vacuously true because $xRy$ never occurs. Likewise, $R$ is transitive because the premise $xRy\land yRz$ never occurs, so
\[
    (xRy\land yRz)\Rightarrow xRz
\]
is vacuously true.

Thus $R$ is symmetric and transitive but not reflexive. Therefore $R$ is not an equivalence relation.
\end{solution}

\begin{problem}{1.2}
For $n\in\Nzero$, prove that
\[
    \sum_{i=0}^{n}\binom{n}{i}^2
    =\binom{2n}{n}.
\]
\end{problem}

\begin{solution}[1]
Let $U$ and $V$ be disjoint sets with $|U|=|V|=n$. Count the $n$-element subsets of $U\cup V$ according to the number $i$ of selected elements from $U$. If exactly $i$ elements are chosen from $U$, then $n-i$ elements must be chosen from $V$, giving
\[
    \binom{n}{i}\binom{n}{n-i}
    =\binom{n}{i}^2
\]
choices. Summing over $i=0,\ldots,n$ counts all $n$-element subsets of the $2n$-element set $U\cup V$. Hence
\[
    \sum_{i=0}^{n}\binom{n}{i}^2
    =\binom{2n}{n}.
\]
\end{solution}

\begin{solution}[2]
By symmetry of binomial coefficients and Vandermonde's identity,
\[
    \sum_{i=0}^{n}\binom{n}{i}^2
    =\sum_{i=0}^{n}\binom{n}{i}\binom{n}{n-i}
    =\binom{2n}{n}.
\]
\end{solution}

\begin{problem}{2.1}
A positive integer $a$ is called \emph{square-free} if it is not divisible by any perfect square other than $1$. For example, $1$, $7$, and $30$ are square-free, whereas $4$ and $63$ are not. Count the square-free positive integers less than or equal to $100$.
\end{problem}

\begin{solution}
A positive integer $a\leq100$ is square-free if and only if none of $4$, $9$, $25$, and $49$ divides $a$. If one of these prime squares divides $a$, then $a$ is not square-free. Conversely, if $a$ is not square-free, then $b^2\mid a$ for some integer $b\geq2$. Choose a prime $p\mid b$. Since $p^2\mid a\leq100$, we have $p\leq10$, so $p\in\{2,3,5,7\}$.

For $p\in\{2,3,5,7\}$, define
\[
    A_p=\{a\in[100]\mid p^2\mid a\}.
\]
The sizes of these sets are
\[
    |A_2|=\left\lfloor\frac{100}{4}\right\rfloor=25,
    \quad |A_3|=\left\lfloor\frac{100}{9}\right\rfloor=11,
    \quad |A_5|=4,
    \quad |A_7|=2.
\]
The only nonempty pairwise intersections are
\[
    |A_2\cap A_3|=\left\lfloor\frac{100}{36}\right\rfloor=2,
    \qquad
    |A_2\cap A_5|=\left\lfloor\frac{100}{100}\right\rfloor=1.
\]
All other pairwise intersections, and hence all intersections of three or more sets, are empty: the product of the corresponding distinct prime squares exceeds $100$. Therefore, by inclusion-exclusion, the number of square-free positive integers at most $100$ is
\[
    100-|A_2\cup A_3\cup A_5\cup A_7|
    =100-(25+11+4+2)+(2+1)
    =61.
\]
\end{solution}

\begin{problem}{2.2}
Let $n,k\in\N$. Count the sequences $(a_1,\ldots,a_k)$ satisfying
\[
    \{a_1,\ldots,a_k\}\subseteq[n],
    \qquad a_1<\cdots<a_k,
    \qquad |a_i-a_j|\geq2\quad(i\neq j).
\]
\end{problem}

\begin{solution}
The conditions imply $a_{i+1}\geq a_i+2$. Since $a_1\geq1$, it follows that $a_k\geq2k-1$. Thus no such sequence exists if $n<2k-1$.

Suppose $n\geq2k-1$. For each admissible sequence, define
\[
    b_i=a_i-(i-1),\qquad 1\leq i\leq k.
\]
Then
\[
    1\leq b_1<b_2<\cdots<b_k\leq n-k+1.
\]
Conversely, given any strictly increasing sequence $1\leq b_1<\cdots<b_k\leq n-k+1$, the formula $a_i=b_i+(i-1)$ produces an admissible sequence in $[n]$. These constructions are inverse to each other, so the desired sequences are in bijection with the $k$-element subsets of $[n-k+1]$. Hence their number is
\[
    \begin{cases}
        \displaystyle\binom{n-k+1}{k}, & n\geq2k-1,\\[5pt]
        0, & n<2k-1.
    \end{cases}
\]
\end{solution}



\begin{problem}{3.1}
Give an example of a graph $G=(V,E)$ satisfying both of the following conditions:
\begin{problemchoices}
\item $|V|\geq3$ and $|E|\geq3$.
\item For any $e_1,e_2\in E$, the graphs $(V,E\setminus\{e_1\})$ and $(V,E\setminus\{e_2\})$ are isomorphic.
\end{problemchoices}
Prove that your example satisfies these conditions.
\end{problem}

\begin{solution}
Each of the following graphs satisfies both conditions.
\begin{itemize}[leftmargin=2em,itemsep=0.35em,topsep=0.4em,parsep=0pt]
\item \emph{Cycle graphs.} For $n\geq3$, the cycle $C_n$ has $n$ vertices and $n$ edges. Deleting any edge leaves the path $P_n$, so all single-edge deletions are isomorphic.
\item \emph{Complete graphs.} For $n\geq3$, the graph $K_n$ has $n$ vertices and $\binom{n}{2}\geq3$ edges. Given edges $e_1=\{a,b\}$ and $e_2=\{c,d\}$, choose a permutation of the vertices sending $a$ to $c$ and $b$ to $d$. It induces an isomorphism $K_n-e_1\cong K_n-e_2$.
\item \emph{Star graphs.} For $n\geq4$, the star $K_{1,n-1}$ has $n$ vertices and $n-1\geq3$ edges. Deleting any edge leaves $K_{1,n-2}$ together with one isolated vertex, independently of the deleted edge.
\end{itemize}
These examples are \emph{edge-transitive}: for any two edges, an automorphism (an isomorphism from the graph to itself) maps one to the other. More generally, edge-transitivity guarantees the required edge-deletion property.
\end{solution}

\begin{problem}{3.2}
Let $V=\{1,\ldots,1000\}$ and let $E_1,E_2\subseteq V\times V$ be defined by
\[
\begin{aligned}
    (u,v)\in E_1 &\iff |u-v|\text{ is odd},\\
    (u,v)\in E_2 &\iff |u-v|\text{ is even}.
\end{aligned}
\]
Prove or disprove that the graphs $G_1=(V,E_1)$ and $G_2=(V,E_2)$ are isomorphic.
\end{problem}

\begin{solution}[1]
In $G_1$, vertices of opposite parity are adjacent, and any two distinct vertices of the same parity are joined by a path of length $2$ through a vertex of opposite parity. Thus $G_1$ is connected. In $G_2$, every edge joins vertices of the same parity, so there is no path between vertices $1$ and $2$. Hence $G_2$ is disconnected. Since connectedness is preserved by graph isomorphisms, $G_1\not\cong G_2$.
\end{solution}

\begin{solution}[2]
Every vertex $v\in V$ has exactly $500$ neighbors distinct from $v$ in $G_1$, namely the vertices of opposite parity. In $G_2$, it has exactly $499$ such neighbors, namely the other vertices of the same parity. An isomorphism preserves the number of neighbors distinct from each vertex, so $G_1\not\cong G_2$. Counting only distinct neighbors avoids any ambiguity from the loops $(v,v)\in E_2$ in the given definition.
\end{solution}
