\begin{exercise}{1.1.21}
Let $G$ be a finite group and let $x$ be an element of $G$ of order $n$. Prove that if $n$ is odd, then $x=(x^2)^k$ for some integer $k$.
\end{exercise}

\begin{solution}
Since $n$ is odd, $k=(n+1)/2$ is an integer. As $x^n=1_G$,
\[
 (x^2)^k=x^{n+1}=x^nx=x.
\]
Thus $x$ is a power of $x^2$.
\end{solution}

\begin{exercise}{1.1.25}
Prove that if $x^2=1_G$ for every $x\in G$, then $G$ is abelian.
\end{exercise}

\begin{solution}
The hypothesis gives $x^{-1}=x$ for every $x\in G$. In particular, for any $a,b\in G$,
\[
 ab=(ab)^{-1}=b^{-1}a^{-1}=ba.
\]
Therefore every two elements commute and $G$ is abelian.
\end{solution}

\begin{exercise}{1.1.31}
Prove that every finite group of even order has an element of order $2$.
\end{exercise}

\begin{hint}
Let $t(G)=\{g\in G\mid g\neq g^{-1}\}$. Pair each member with its inverse, then consider the remaining elements.
\end{hint}

\begin{solution}
Let $T=\{g\in G\mid g\ne g^{-1}\}$. The elements of $T$ can be paired as $\{g,g^{-1}\}$, with distinct members, so $|T|$ is even. Since $|G|$ is even, $G\setminus T$ also has even cardinality. It contains $1_G$, hence contains another element $x\ne1_G$. By the definition of $T$ we have $x=x^{-1}$, so $x^2=1_G$. As $x$ is not the identity, its order is exactly $2$.
\end{solution}

\begin{exercise}{1.1.34}
If $x$ has infinite order in $G$, prove that $x^n$ for $n\in\Z$ are all distinct.
\end{exercise}

\begin{solution}
If $x^m=x^n$ for integers $m\ne n$, multiplication by $x^{-\min(m,n)}$ yields $x^{|m-n|}=1_G$. This is a positive power of $x$, contradicting its infinite order. Hence $m=n$.
\end{solution}

\begin{exercise}{1.2.3}
Use the presentation
\[
D_n=\langle r,s\mid r^n=s^2=1,\ rs=sr^{-1}\rangle
\]
to prove that every element not in $\langle r\rangle$ has order $2$. Deduce that $D_n$ is generated by $s$ and $sr$, both of order $2$.
\end{exercise}

\begin{notation}
Under MATH000, $D_n$ denotes the symmetries of a regular $n$-gon (order $2n$); the textbook also uses the alternative notation $D_{2n}$.
\end{notation}

\begin{solution}
Every element outside $\langle r\rangle$ has the form $sr^i$. From $r^is=sr^{-i}$,
\[
 (sr^i)^2=sr^isr^i=s(sr^{-i})r^i=1_G.
\]
Such an element is not the identity, so its order is $2$. In particular $s$ and $sr$ have order $2$. Since $s(sr)=r$, the subgroup generated by $s$ and $sr$ contains both standard generators $r,s$, and therefore equals $D_n$.
\end{solution}

\begin{exercise}{1.2.6}
Let $x,y\in G$ have order $2$, and set $t=xy$. Prove $tx=xt^{-1}$. If $n=|xy|<\infty$, show that $x,t$ satisfy the dihedral relations.
\end{exercise}

\begin{solution}
Since $x^{-1}=x$ and $y^{-1}=y$, we have $t^{-1}=yx$. Therefore
\[
 tx=(xy)x=x(yx)=xt^{-1}.
\]
If $|t|=n<\infty$, then $t^n=x^2=1_G$ and $tx=xt^{-1}$. These are precisely the relations satisfied by $r,s$ in the usual dihedral presentation, with $r=t$ and $s=x$.
\end{solution}

\begin{exercise}{1.2.7}
Show that $\langle a,b\mid a^2=b^2=(ab)^n=1\rangle$ is a presentation of $D_n$, under $a=s$ and $b=sr$.
\end{exercise}

\begin{hint}
Show in both directions that the generators in one presentation can be expressed through the other and satisfy the required relations.
\end{hint}

\begin{solution}
Starting with the standard generators $r,s$ of $D_n$, put $a=s$ and $b=sr$. Then $a^2=1$, while
\[
 b^2=srsr=r^{-1}r=1,
 \qquad ab=s(sr)=r,
\]
so $(ab)^n=1$. Moreover, $s=a$ and $r=ab$, so $a,b$ generate $D_n$.

Conversely, in the group given by the displayed presentation, define $s=a$ and $r=ab$. Then $s^2=1$, $r^n=1$, and
\[
 rs=(ab)a=aba=a(ba)=sr^{-1},
\]
because $a^{-1}=a$ and $b^{-1}=b$. The changes of generators $a=s$, $b=sr$ and $s=a$, $r=ab$ are inverse to each other. Thus the two presentations describe the same group.
\end{solution}

\begin{exercise}{1.2.15}
Find a set of generators and relations for the additive group $\Z/n\Z$.
\end{exercise}

\begin{solution}
The class $[1]_n$ generates the additive group, since every class equals $k[1]_n$ for an integer $k$. Its order is exactly $n$, so the defining relation is $n[1]_n=[0]_n$. Thus a presentation in multiplicative notation is
\[
 \Z/n\Z\cong\langle x\mid x^n=1\rangle.
\]
Every word in the single generator reduces to a power $x^k$, and the relation identifies precisely exponents congruent modulo $n$, so no additional relations are needed.
\end{solution}

\begin{exercise}{1.3.5}
Find the order of $(1\ 12\ 8\ 10\ 4)(2\ 13)(5\ 11\ 7)(6\ 9)$.
\end{exercise}

\begin{solution}
The four cycles are disjoint and have lengths $5,2,3,2$. Their powers become the identity simultaneously exactly when the exponent is divisible by all four lengths. Hence the order is
\[
 \operatorname{lcm}(5,2,3,2)=\boxed{30}.
\]
\end{solution}

\begin{exercise}{1.3.13}
Show that an element has order $2$ in $S_n$ if and only if its disjoint cycle decomposition is a nonempty product of commuting $2$-cycles.
\end{exercise}

\begin{solution}
If $\sigma$ has order $2$, then $\sigma^2=1$. The length of every nontrivial disjoint cycle must divide $2$; hence each has length $2$. At least one occurs because $\sigma\ne1$. Conversely, a nonempty product of disjoint transpositions has square $1$ because its factors commute and each has square $1$. It is not the identity, so its order is exactly $2$.
\end{solution}

\begin{exercise}{1.3.14}
Let $p$ be prime. Show that an element has order $p$ in $S_n$ if and only if its disjoint cycle decomposition is a nonempty product of $p$-cycles. Give a counterexample for composite $p$.
\end{exercise}

\begin{solution}
Suppose $|\sigma|=p$. Then $\sigma^p=1$, so every cycle in the disjoint cycle decomposition must have length dividing $p$. Since $p$ is prime, each nontrivial cycle has length $p$. At least one such cycle occurs because $\sigma\ne1$.

Conversely, a nonempty product of disjoint $p$-cycles has $p$th power equal to the identity, since the cycles commute. It is not the identity, so its order is exactly $p$.

For a composite order the conclusion need not hold: $\sigma=(1\ 2)(3\ 4\ 5\ 6)$ has order $\operatorname{lcm}(2,4)=4$, but its decomposition contains a $2$-cycle, not only $4$-cycles.
\end{solution}

\begin{exercise}{1.3.15}
Prove that the order of a permutation equals the least common multiple of its disjoint cycle lengths.
\end{exercise}

\begin{hint}
Use the fact that disjoint cycles commute and that an $m$-cycle has order $m$.
\end{hint}

\begin{solution}
Write $\sigma=c_1\cdots c_r$ as a product of disjoint nontrivial cycles of lengths $m_1,\ldots,m_r$. Disjoint cycles commute, and a power $\sigma^k$ is the identity precisely when each $c_i^k$ is the identity: on the support of $c_i$, all other cycles act trivially. Since $c_i^k=1$ if and only if $m_i\mid k$, the least positive such $k$ is
\[
 |\sigma|=\operatorname{lcm}(m_1,\ldots,m_r).
\]
For the identity permutation, which has no nontrivial cycles, its order is $1$ (equivalently, include its $1$-cycles).
\end{solution}

\begin{exercise}{1.3.18}
Find all positive integers $n$ for which $S_5$ contains an element of order $n$.
\end{exercise}

\begin{hint}
Apply the least-common-multiple formula from Exercise 1.3.15 to all partitions of $5$.
\end{hint}

\begin{solution}
A cycle decomposition partitions the five symbols into disjoint cycles. Apart from fixed points, the possible cycle-length patterns are $2$, $2+2$, $3$, $3+2$, $4$, and $5$. The corresponding least common multiples are $2,2,3,6,4,5$. Together with the identity (order $1$), the possible orders are
\[
 \boxed{1,2,3,4,5,6}.
\]
Every listed value is realized by the corresponding cycle pattern, and no other pattern is possible.
\end{solution}

\begin{exercise}{1.6.4}
Prove that the multiplicative groups $\R\setminus\{0\}$ and $\C\setminus\{0\}$ are not isomorphic.
\end{exercise}

\begin{solution}
The complex number $i$ has multiplicative order $4$. An isomorphism preserves element orders. But a real number satisfying $x^4=1$ must be $\pm1$, of orders $1$ and $2$, respectively. Thus the real multiplicative group has no element of order $4$, and the groups cannot be isomorphic.
\end{solution}

\begin{exercise}{1.6.14}
For a homomorphism $\varphi:G\to H$, show that $\ker\varphi=\{g\in G\mid\varphi(g)=1_H\}$ is a subgroup of $G$, and that $\varphi$ is injective if and only if $\ker\varphi=\{1_G\}$.
\end{exercise}

\begin{solution}
We have $\varphi(1_G)=1_H$, so $1_G\in\ker\varphi$. If $a,b\in\ker\varphi$, then
\[
 \varphi(ab^{-1})=\varphi(a)\varphi(b)^{-1}=1_H,
\]
so $ab^{-1}\in\ker\varphi$. Thus $\ker\varphi\le G$ by the subgroup criterion.

If $\varphi$ is injective, then $\varphi(g)=1_H=\varphi(1_G)$ implies $g=1_G$. Conversely, if $\ker\varphi=\{1_G\}$ and $\varphi(a)=\varphi(b)$, then $\varphi(ab^{-1})=1_H$, so $ab^{-1}=1_G$ and $a=b$.
\end{solution}

\begin{exercise}{1.6.19}
Let $G=\{z\in\C\mid z^n=1\text{ for some }n\in\N\}$. For a fixed integer $k>1$, prove that $\varphi:G\to G$, $z\mapsto z^k$, is a surjective homomorphism but not an isomorphism.
\end{exercise}

\begin{solution}
Since complex multiplication is commutative, $(zw)^k=z^kw^k$, so $\varphi$ is a homomorphism. Given $w\in G$, choose a complex $k$th root $z$ of $w$. As $w^n=1$ for some $n\ge1$, we have $z^{kn}=1$, so $z\in G$ and $\varphi(z)=w$. Thus $\varphi$ is surjective.

However, for $\zeta=e^{2\pi i/k}$ we have $\zeta\ne1$, $\zeta\in G$, and $\varphi(\zeta)=1=\varphi(1)$. Therefore $\varphi$ is not injective and hence not an isomorphism.
\end{solution}

\begin{exercise}{1.6.22}
Let $A$ be an abelian group and $k\in\Z$. Prove that $a\mapsto a^k$ is a homomorphism $A\to A$. Show that for $k=-1$ it is an automorphism.
\end{exercise}

\begin{solution}
For commuting $a,b\in A$ and any integer $k$, the exponent laws give $(ab)^k=a^kb^k$. Hence $\varphi_k(ab)=\varphi_k(a)\varphi_k(b)$. For $k=-1$, the map sends $a$ to $a^{-1}$, and applying it twice returns $a$. Thus $\varphi_{-1}^{-1}=\varphi_{-1}$, proving that it is an automorphism.
\end{solution}

\begin{exercise}{1.6.25}
Let $n$ be a positive integer, let $r,s$ be the usual generators of $D_n$, and let $\theta=2\pi/n$.
\begin{exercisesubparts}
\item Prove that
\[
\begin{pmatrix}
\cos\theta&-\sin\theta\\
\sin\theta&\cos\theta
\end{pmatrix}
\]
is the matrix of the counterclockwise rotation of the $xy$-plane about the origin by $\theta$ radians.
\item Prove that the map $\varphi\colon D_n\to\GL_2(\R)$ defined on generators by
\[
\varphi(r)=\begin{pmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{pmatrix},
\qquad
\varphi(s)=\begin{pmatrix}0&1\\1&0\end{pmatrix}
\]
extends to a homomorphism.
\item Prove that the homomorphism $\varphi$ in part (b) is injective.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
Put $\mathbf R=\begin{pmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{pmatrix}$ and $\mathbf S=\begin{pmatrix}0&1\\1&0\end{pmatrix}$.
\begin{exercisesubparts}
\item Rotation through $\theta$ maps the standard basis vectors to $(\cos\theta,\sin\theta)^T$ and $(-\sin\theta,\cos\theta)^T$. These vectors form the columns of $\mathbf R$, giving the claimed matrix.
\item The matrices are invertible, with $\mathbf R^n=\mathbf I$ and $\mathbf S^2=\mathbf I$. Direct multiplication yields
\[
\mathbf R\mathbf S=\begin{pmatrix}-\sin\theta&\cos\theta\\\cos\theta&\sin\theta\end{pmatrix}=\mathbf S\mathbf R^{-1}.
\]
They therefore satisfy the defining relations of $D_n$. The presentation gives a homomorphism extending the prescribed values on $r,s$.
\item Every element of $D_n$ has the form $r^j$ or $sr^j$ for $0\le j<n$. The equality $\mathbf R^j=\mathbf I$ occurs for this range only when $j=0$, since $\mathbf R$ rotates by $2\pi/n$. On the other hand,
\[
\det(\mathbf S\mathbf R^j)=(-1)(1)=-1\ne\det\mathbf I.
\]
Hence no $sr^j$ lies in the kernel. Thus the kernel is $\{1_G\}$, so the homomorphism is injective.
\end{exercisesubparts}
\end{solution}
