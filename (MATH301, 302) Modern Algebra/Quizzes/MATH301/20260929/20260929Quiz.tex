\begin{exercise}{1.7.1}
Let $F$ be a field. Show that $F^\times=F\setminus\{0\}$ acts on $F$ by $g\cdot a=ga$. State which field axioms are used.
\end{exercise}

\begin{solution}
For $g,h\in F^\times$ and $a\in F$, associativity of multiplication gives
\[
 g\cdot(h\cdot a)=g(ha)=(gh)a=(gh)\cdot a.
\]
The multiplicative identity satisfies $1\cdot a=a$. Thus the two action axioms hold. They use precisely the associativity and identity axioms for multiplication. The other field axioms ensure that nonzero elements form the group $F^\times$: in particular, multiplication is closed on nonzero elements and every nonzero element has a multiplicative inverse.
\end{solution}

\begin{exercise}{1.7.16}
Let $G$ act on itself by conjugation, $g\cdot a=gag^{-1}$. Show that this defines a left group action.
\end{exercise}

\begin{solution}
For $g,h,a\in G$,
\[
 g\cdot(h\cdot a)=g(hah^{-1})g^{-1}=(gh)a(gh)^{-1}=(gh)\cdot a.
\]
Moreover $1_G\cdot a=1_Ga1_G^{-1}=a$. Both defining axioms are satisfied, so conjugation is a left action of $G$ on itself.
\end{solution}

\begin{exercise}{1.7.17}
For $g\in G$, prove that $c_g:G\to G$, $x\mapsto gxg^{-1}$, is an automorphism. Deduce that $|x|=|gxg^{-1}|$ and $|A|=|gAg^{-1}|$ for any $A\subseteq G$.
\end{exercise}

\begin{solution}
For $x,y\in G$,
\[
 c_g(xy)=gxyg^{-1}=(gxg^{-1})(gyg^{-1})=c_g(x)c_g(y).
\]
Thus $c_g$ is a homomorphism. Since $c_{g^{-1}}\circ c_g=c_g\circ c_{g^{-1}}=\id_G$, it is bijective, hence an automorphism. An isomorphism preserves element orders because $c_g(x)^m=1$ if and only if $x^m=1$. Its restriction $A\to gAg^{-1}$ is also a bijection, proving $|A|=|gAg^{-1}|$.
\end{solution}

\begin{exercise}{2.1.2}
For each subset below, prove that it is not a subgroup of the indicated group.
\begin{exercisesubparts}
\item The transpositions in $S_n$, $n\ge3$.
\item The reflections in $D_n$, $n\ge3$.
\item If $n>1$ is composite and $G$ has an element of order $n$, the set $\{x\in G:|x|=n\}\cup\{1_G\}$.
\item The odd integers together with $0$ in $(\Z,+)$.
\item The set $\{x\in\R:x^2\in\Q\}$ in $(\R,+)$.
\end{exercisesubparts}
\end{exercise}

\begin{notation}
As in MATH000, $D_n$ is the symmetry group of a regular $n$-gon, of order $2n$ (written $D_{2n}$ in the alternative convention of Dummit--Foote).
\end{notation}

\begin{solution}
\begin{exercisesubparts}
\item The identity permutation is not a transposition; hence the set does not contain the identity.
\item The identity is a rotation, not a reflection, so the set does not contain the identity.
\item Take $x\in G$ of order $n$, and write $n=ab$ with $1<a,b<n$. Then $x$ belongs to the set, but $|x^a|=b$, so $x^a$ does not. Any subgroup containing $x$ would contain all its powers.
\item We have $1$ in the set but $1+1=2$ is not, so it is not closed under addition.
\item Both $1$ and $\sqrt2$ belong to the set, since their squares are rational, but $(1+\sqrt2)^2=3+2\sqrt2$ is irrational. The set is not closed under addition.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{2.1.3}
Show that $H=\{1,r^2,s,sr^2\}$ and $K=\{1,r^2,sr,sr^3\}$ are subgroups of $D_4=\langle r,s\mid r^4=s^2=1,\ srs=r^{-1}\rangle$.
\end{exercise}

\begin{solution}
Both $r^2$ and $s$ are involutions and commute, since $sr^2s^{-1}=r^{-2}=r^2$. Therefore
\[
 H=\langle r^2,s\rangle=\{1,r^2,s,sr^2\},
\]
which is a subgroup. Similarly, $sr$ has order $2$ and commutes with $r^2$. Thus
\[
 K=\langle r^2,sr\rangle=\{1,r^2,sr,r^2sr\}=\{1,r^2,sr,sr^3\},
\]
which is also a subgroup. In each case the displayed elements exhaust the subgroup generated by two commuting involutions.
\end{solution}

\begin{exercise}{2.1.6}
Let $G$ be abelian. Prove that its elements of finite order form a subgroup (its torsion subgroup). Give a nonabelian example where finite-order elements are not a subgroup.
\end{exercise}

\begin{solution}
Let $T=\{g\in G:|g|<\infty\}$. Certainly $1_G\in T$. If $x,y\in T$ have orders $m,n$, then commutativity gives
\[
 (xy^{-1})^{mn}=x^{mn}y^{-mn}=1_G.
\]
Hence $xy^{-1}\in T$, so $T\le G$.

For a counterexample take
\[
\mathbf A=\begin{pmatrix}0&1\\1&0\end{pmatrix},\qquad
\mathbf B=\begin{pmatrix}0&2\\1/2&0\end{pmatrix}\in\GL_2(\R).
\]
Both matrices have order $2$, whereas $\mathbf A\mathbf B=\operatorname{diag}(1/2,2)$ has infinite order: its $k$th power is $\operatorname{diag}(2^{-k},2^k)\ne\mathbf I$ for every $k>0$. Thus finite-order elements are not necessarily closed under multiplication in a nonabelian group.
\end{solution}

\begin{exercise}{2.1.13}
Let $H\le(\Q,+)$ and suppose $1/x\in H$ for every nonzero $x\in H$. Prove that $H=\{0\}$ or $H=\Q$.
\end{exercise}

\begin{solution}
Suppose $H\ne\{0\}$ and choose $x=a/b\in H$ with nonzero $a\in\Z$, $b>0$, and $\gcd(a,b)=1$. Then $1/x=b/a\in H$. Since $H$ is an additive subgroup, it contains the integers $bx=a$ and $a(1/x)=b$. By B\'ezout's identity there exist integers $u,v$ such that $ua+vb=1$, so $1\in H$. Hence $\Z\subseteq H$.

For any positive integer $q$, the assumption applied to $q\in H$ yields $1/q\in H$. Multiplication by integers then gives $p/q\in H$ for all $p\in\Z$, so $H=\Q$.
\end{solution}

\begin{exercise}{2.3.1}
Find all subgroups of $C_{45}=\langle x\rangle$, give a generator for each, and describe their containments.
\end{exercise}

\begin{notation}
$C_n$ denotes the abstract cyclic group of order $n$ (called $Z_n$ in Dummit--Foote). The additive residue-class group is written $\Z/n\Z$ under MATH000.
\end{notation}

\begin{solution}
A cyclic group of order $45$ has exactly one subgroup of each order dividing $45$, namely $H_d=\langle x^{45/d}\rangle$. The six divisors and subgroups are
\[
\begin{array}{c@{\qquad}l}
\toprule
 d & H_d\\
\midrule
 1 & \{1_G\}=\langle x^{45}\rangle\\
 3 & \langle x^{15}\rangle\\
 5 & \langle x^9\rangle\\
 9 & \langle x^5\rangle\\
 15 & \langle x^3\rangle\\
 45 & \langle x\rangle\\
\bottomrule
\end{array}
\]
For any divisors $d,e$ of $45$, uniqueness of cyclic subgroups shows $H_d\le H_e$ if and only if $d\mid e$. This rule gives all containments; for instance, $H_3\le H_9,H_{15}$, while $H_5$ and $H_9$ are incomparable.
\end{solution}

\begin{exercise}{2.3.8}
Let $C_{48}=\langle x\rangle$. For which integers $a$ does $[1]_{48}\mapsto x^a$ extend to an isomorphism $\Z/48\Z\to C_{48}$?
\end{exercise}

\begin{solution}
The proposed map extends uniquely to $\varphi_a([k]_{48})=x^{ak}$. It is well defined: changing $k$ by a multiple of $48$ changes the exponent by a multiple of $48$, and $x^{48}=1$. It is a homomorphism by the exponent laws. It is bijective exactly when its image contains the generator $x$, equivalently when $x^a$ generates $C_{48}$. Since
\[
 |x^a|=\frac{48}{\gcd(a,48)},
\]
this occurs if and only if $\gcd(a,48)=1$. Modulo $48$ the solutions are
\[
\boxed{1,5,7,11,13,17,19,23,25,29,31,35,37,41,43,47}.
\]
\end{solution}

\begin{exercise}{2.3.11}
Find all cyclic subgroups of $D_4$ and exhibit a proper noncyclic subgroup.
\end{exercise}

\begin{solution}
Write $D_4=\{1,r,r^2,r^3,s,sr,sr^2,sr^3\}$. The rotations have orders $1,4,2,4$, respectively; every reflection $sr^j$ has order $2$. Thus the distinct cyclic subgroups are
\[
\{1\},\quad\langle r\rangle,\quad\langle r^2\rangle,\quad\langle s\rangle,\quad\langle sr\rangle,\quad\langle sr^2\rangle,\quad\langle sr^3\rangle.
\]
The proper subgroup $H=\langle r^2,s\rangle=\{1,r^2,s,sr^2\}$ is not cyclic: all three of its nonidentity elements have order $2$, so none generates its four elements.
\end{solution}

\begin{exercise}{2.3.12}
Prove that $(\Z/2\Z)\times(\Z/2\Z)$, $(\Z/2\Z)\times\Z$, and $\Z\times\Z$ are not cyclic.
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item Every nonidentity element of $(\Z/2\Z)^2$ has order $2$, whereas a cyclic group of order $4$ would have an element of order $4$.
\item If $(\Z/2\Z)\times\Z=\langle(a,m)\rangle$, then generating $(0,1)$ forces $km=1$ for some $k\in\Z$, hence $m=\pm1$. But every nonzero multiple of $(a,m)$ has nonzero second coordinate, so it cannot generate the nonzero element $(1,0)$.
\item If $\Z^2=\langle(a,b)\rangle$, then generating $(1,0)$ forces $ka=1$ and $kb=0$ for some integer $k$. Thus $k=\pm1$ and $b=0$. All multiples of $(a,b)$ have second coordinate zero, so $(0,1)$ cannot be generated.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{2.3.19}
Let $H$ be a group and $h\in H$. Prove that there is a unique homomorphism $\varphi:\Z\to H$ with $\varphi(1)=h$.
\end{exercise}

\begin{solution}
Define $\varphi(n)=h^n$ for $n\in\Z$. Then $\varphi(m+n)=h^{m+n}=h^mh^n=\varphi(m)\varphi(n)$ and $\varphi(1)=h$. Any homomorphism $\psi:\Z\to H$ with $\psi(1)=h$ must satisfy $\psi(n)=\psi(1)^n=h^n$ for positive, negative, and zero integers $n$. Therefore $\psi=\varphi$, proving uniqueness.
\end{solution}

\begin{exercise}{2.3.21}
Let $p$ be an odd prime and $n\ge1$. Use the binomial theorem to show
\[
 (1+p)^{p^{n-1}}\equiv1\pmod{p^n}
\]
and, for $n\ge2$,
\[
 (1+p)^{p^{n-2}}\not\equiv1\pmod{p^n}.
\]
Deduce that $1+p$ has order $p^{n-1}$ in $(\Z/p^n\Z)^\times$.
\end{exercise}

\begin{hint}
Prove the stronger congruence modulo $p^{m+2}$ by induction on $m$.
\end{hint}

\begin{solution}
We first show by induction on $m\ge0$ that
\[
 (1+p)^{p^m}\equiv1+p^{m+1}\pmod{p^{m+2}}.
\]
This is immediate for $m=0$. Suppose $(1+p)^{p^m}=1+u$, where $u=p^{m+1}+cp^{m+2}$. The binomial theorem gives
\[
(1+u)^p=1+pu+\sum_{j=2}^{p-1}\binom pj u^j+u^p.
\]
Here $pu\equiv p^{m+2}\pmod{p^{m+3}}$. For $2\le j\le p-1$, the coefficient $\binom pj$ is divisible by $p$, so $\binom pj u^j$ is divisible by $p^{1+j(m+1)}$, hence by $p^{m+3}$. Also $u^p$ is divisible by $p^{p(m+1)}$, which is a multiple of $p^{m+3}$ because $p\ge3$. The induction follows.

Taking $m=n-1$ yields $(1+p)^{p^{n-1}}\equiv1\pmod{p^n}$. When $n\ge2$, taking $m=n-2$ yields
\[
(1+p)^{p^{n-2}}\equiv1+p^{n-1}\not\equiv1\pmod{p^n}.
\]
Therefore the order divides $p^{n-1}$ but not $p^{n-2}$, and so equals $p^{n-1}$. If $n=1$, the element is $1$ modulo $p$ and its order is $1=p^0$.
\end{solution}

\begin{exercise}{2.3.24}
Let $G$ be finite and $x\in G$.
\begin{exercisesubparts}
\item If $g\in N_G(\langle x\rangle)$, prove that $gxg^{-1}=x^a$ for some $a\in\Z$.
\item Prove the converse.
\end{exercisesubparts}
\end{exercise}

\begin{hint}
The conjugation map is a bijection; in a finite group, a subgroup contained in another subgroup of the same order must equal it.
\end{hint}

\begin{solution}
\begin{exercisesubparts}
\item Since $g\langle x\rangle g^{-1}=\langle x\rangle$, the conjugate $gxg^{-1}$ belongs to the cyclic group $\langle x\rangle$; hence it equals $x^a$ for some integer $a$.
\item Suppose $gxg^{-1}=x^a$. Then
\[
 gx^kg^{-1}=(gxg^{-1})^k=x^{ak}\in\langle x\rangle
\]
for every integer $k$, showing $g\langle x\rangle g^{-1}\subseteq\langle x\rangle$. Conjugation by $g$ is a bijection, so the two finite sets have the same cardinality $|x|$. Hence containment is equality and $g\in N_G(\langle x\rangle)$.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{2.3.26}
For a cyclic group $C_n$ and each integer $a$, define $\sigma_a(x)=x^a$ for $x\in C_n$.
\begin{exercisesubparts}
\item Show that $\sigma_a\in\Aut(C_n)$ if and only if $\gcd(a,n)=1$.
\item Show that $\sigma_a=\sigma_b$ if and only if $a\equiv b\pmod n$.
\item Show that every automorphism is $\sigma_a$ for some $a$.
\item Show $\sigma_a\circ\sigma_b=\sigma_{ab}$ and deduce $(\Z/n\Z)^\times\cong\Aut(C_n)$, an abelian group of order $\varphi(n)$.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
Write $C_n=\langle z\rangle$. Because $C_n$ is abelian, the power map $\sigma_a$ is a homomorphism.
\begin{exercisesubparts}
\item Its image is $\langle z^a\rangle$; it is surjective, hence an automorphism since $C_n$ is finite, exactly when $z^a$ has order $n$. This is equivalent to $\gcd(a,n)=1$.
\item Two homomorphisms from a cyclic group agree precisely when they agree on its generator. Thus $\sigma_a=\sigma_b$ if and only if $z^a=z^b$, or $n\mid(a-b)$.
\item If $\tau\in\Aut(C_n)$, the generator $z$ maps to another generator $z^a$. For all integers $k$, $\tau(z^k)=\tau(z)^k=z^{ak}=\sigma_a(z^k)$, so $\tau=\sigma_a$.
\item For every $x\in C_n$,
\[
 (\sigma_a\circ\sigma_b)(x)=\sigma_a(x^b)=x^{ab}=\sigma_{ab}(x).
\]
The assignment $[a]_n\mapsto\sigma_a$ from $(\Z/n\Z)^\times$ is well defined by (b), maps to automorphisms by (a), is surjective by (c), injective by (b), and preserves multiplication by the displayed identity. Hence it is an isomorphism. Because $(\Z/n\Z)^\times$ is abelian and has $\varphi(n)$ elements, so does $\Aut(C_n)$.
\end{exercisesubparts}
\end{solution}
