\begin{exercise}{1.7.18}
Let $H$ act on a set $A$. Define $a\sim b$ if $a=h\cdot b$ for some $h\in H$. Prove that this is an equivalence relation and that the equivalence classes (orbits) partition $A$.
\end{exercise}

\begin{solution}
For $a\in A$, $a=1_H\cdot a$, proving reflexivity. If $a=h\cdot b$, then $b=h^{-1}\cdot a$, proving symmetry. If $a=h\cdot b$ and $b=k\cdot c$, then $a=(hk)\cdot c$, proving transitivity. Thus $\sim$ is an equivalence relation. Its class containing $a$ is
\[
 [a]=\{h\cdot a:h\in H\}=H\cdot a,
\]
exactly the orbit of $a$. Equivalence classes cover the set and are pairwise equal or disjoint, hence the distinct orbits partition $A$.
\end{solution}

\begin{exercise}{2.2.4}
For $S_3$, $D_4$, and $Q_8$, determine the centralizer of each element and the center of each group. Explain how Lagrange's theorem helps.
\end{exercise}

\begin{solution}
We compute $C_G(x)=\{g\in G:gx=xg\}$.

In $S_3$, a transposition commutes with itself and the identity, but not with either other transposition; the two nonidentity $3$-cycles are powers of one another. Accordingly,
\[
\begin{array}{c@{\qquad}l}\toprule
x&C_{S_3}(x)\\\midrule
1&S_3\\
(12),(13),(23)&\langle x\rangle\\
(123),(132)&\langle(123)\rangle\\\bottomrule\end{array}
\]
and $Z(S_3)=\{1\}$.

Write $D_4=\langle r,s\mid r^4=s^2=1,\ srs=r^{-1}\rangle$. We have $r^2\in Z(D_4)$ and $r^i(sr^j)=(sr^j)r^i$ precisely when $r^{2i}=1$. Two reflections $sr^i,sr^j$ commute if and only if $r^{2(i-j)}=1$, equivalently $i\equiv j\pmod2$. Thus
\[
\begin{array}{c@{\qquad}l}\toprule
x&C_{D_4}(x)\\\midrule
1,r^2&D_4\\
r,r^3&\langle r\rangle\\
s,sr^2&\langle s,r^2\rangle\\
sr,sr^3&\langle sr,r^2\rangle\\\bottomrule\end{array}
\]
and $Z(D_4)=\{1,r^2\}$.

In $Q_8=\{\pm1,\pm i,\pm j,\pm k\}$, the elements $\pm1$ are central. Distinct quaternion units among $i,j,k$ anticommute, so an element $\pm i$ commutes precisely with $\langle i\rangle$ (and similarly for $j,k$):
\[
\begin{array}{c@{\qquad}l}\toprule
x&C_{Q_8}(x)\\\midrule
\pm1&Q_8\\
\pm i&\langle i\rangle\\
\pm j&\langle j\rangle\\
\pm k&\langle k\rangle\\\bottomrule\end{array}
\]
Therefore $Z(Q_8)=\{\pm1\}$.

Lagrange's theorem restricts $|C_G(x)|$, since $\langle x\rangle\le C_G(x)\le G$. For example, a transposition in $S_3$ has order $2$, so its centralizer has order $2$ or $6$; a single noncommuting element rules out $6$.
\end{solution}

\begin{exercise}{2.2.6}
Let $H\le G$.
\begin{exercisesubparts}
\item Show $H\le N_G(H)$; give a counterexample for a subset $H$ which is not a subgroup.
\item Show $H\le C_G(H)$ if and only if $H$ is abelian.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item For $h\in H$, conjugation by $h$ maps $H$ to itself, and its inverse is conjugation by $h^{-1}\in H$. Hence $hHh^{-1}=H$, giving $H\le N_G(H)$.

For a non-subgroup subset take $H=\{(12),(13)\}\subseteq S_3$. Conjugation by $(12)$ sends $(13)$ to $(23)$, so $(12)H(12)^{-1}\ne H$. Thus $(12)\in H$ does not lie in $N_{S_3}(H)$.
\item The inclusion $H\le C_G(H)$ means exactly that every $h\in H$ commutes with every element of $H$, which is the definition that $H$ is abelian.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{2.4.3}
If $H$ is an abelian subgroup of $G$, show that $\langle H,Z(G)\rangle$ is abelian. Give an example where $\langle H,C_G(H)\rangle$ is not abelian.
\end{exercise}

\begin{solution}
Every element of $Z(G)$ commutes with every element of $G$, and the elements of $H$ commute with each other. Consequently, every element of $\langle H,Z(G)\rangle$ can be written $hz$ with $h\in H$, $z\in Z(G)$. For two such elements,
\[
(h_1z_1)(h_2z_2)=h_1h_2z_1z_2=h_2h_1z_2z_1=(h_2z_2)(h_1z_1),
\]
proving this subgroup is abelian.

For an example let $G=D_4$ and $H=Z(D_4)=\langle r^2\rangle$. Then $H$ is abelian, while $C_G(H)=G$, so $\langle H,C_G(H)\rangle=G$ is nonabelian.
\end{solution}

\begin{exercise}{2.4.7}
Prove that $\langle(12),(13)(24)\rangle\le S_4$ is isomorphic to $D_4$ (of order $8$).
\end{exercise}

\begin{solution}
Set $a=(12)$, $b=(13)(24)$, $r=ab$, and $s=a$. Both $a,b$ are involutions, and $r=ab$ is a $4$-cycle, so $|r|=4$. Moreover,
\[
 s^2=1,\qquad srs=a(ab)a=ba=(ab)^{-1}=r^{-1}.
\]
The relations reduce every word in $r,s$ to one of $r^j$ or $sr^j$ ($0\le j<4$), giving at most eight elements. The first four are distinct; the other four are distinct and do not coincide with them because $s$ is a transposition while the elements of $\langle r\rangle$ are the identity, a double transposition, or $4$-cycles. Thus there are exactly eight elements. Sending the standard generators of $D_4$ to $r,s$ defines a surjective homomorphism between two groups of order $8$, hence an isomorphism.
\end{solution}

\begin{exercise}{2.4.14}
A group is finitely generated if it has a finite generating set.
\begin{exercisesubparts}
\item Show every finite group is finitely generated.
\item Show $\Z$ is finitely generated.
\item Show every finitely generated subgroup of $(\Q,+)$ is cyclic.
\item Deduce that $(\Q,+)$ is not finitely generated.
\end{exercisesubparts}
\end{exercise}

\begin{hint}
Clear the denominators of the proposed finite generating set.
\end{hint}

\begin{solution}
\begin{exercisesubparts}
\item A finite group $H$ is generated by the finite set $H$ itself.
\item The additive group $\Z=\langle1\rangle$ is generated by one element.
\item If $H=\langle a_1/b_1,\ldots,a_t/b_t\rangle\le\Q$ with nonzero denominators $b_i>0$, let $k=b_1\cdots b_t$. Each $a_i/b_i=(a_i k/b_i)(1/k)$ is an integer multiple of $1/k$. Thus $H\le\langle1/k\rangle$, and every subgroup of a cyclic group is cyclic.
\item Were $\Q$ finitely generated, (c) would make it cyclic. But if $\Q=\langle q\rangle$ with $q\ne0$, then $q/2$ would have to be an integer multiple of $q$, forcing $1/2\in\Z$. The generator $q=0$ is plainly impossible. Hence $\Q$ is not finitely generated.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{2.4.15}
Exhibit a proper subgroup of $(\Q,+)$ that is not cyclic.
\end{exercise}

\begin{solution}
Consider $H=\{m/2^k:m\in\Z,\ k\ge0\}$. It is closed under subtraction because fractions with denominators powers of $2$ have a common denominator of this form. Thus $H\le\Q$, but $1/3\notin H$, so the subgroup is proper.

If $H=\langle q\rangle$ for a nonzero $q=m/2^k$, then $1/2^{k+1}\in H$ would have to equal $tq$ for some $t\in\Z$. This would give $2tm=1$, which is impossible. The zero element cannot generate $H$ either. Thus $H$ is not cyclic.
\end{solution}

\begin{exercise}{2.5.3}
Show that $\langle s,r^2\rangle\le D_4$ is isomorphic to $V_4$.
\end{exercise}

\begin{solution}
Since $r^2$ and $s$ are commuting, distinct involutions, they generate exactly
\[
 H=\{1,r^2,s,sr^2\}.
\]
The map $(\Z/2\Z)^2\to H$, $([i]_2,[j]_2)\mapsto r^{2i}s^j$, respects addition because the generators commute, and is a bijection between two four-element sets. Therefore $H\cong(\Z/2\Z)^2\cong V_4$.
\end{solution}

\begin{exercise}{3.1.1}
If $\varphi:G\to H$ is a homomorphism and $E\le H$, show $\varphi^{-1}(E)\le G$. If $E\trianglelefteq H$, prove $\varphi^{-1}(E)\trianglelefteq G$. Deduce $\ker\varphi\trianglelefteq G$.
\end{exercise}

\begin{solution}
Because $\varphi(1_G)=1_H\in E$, the preimage is nonempty. If $x,y\in\varphi^{-1}(E)$, then $\varphi(xy^{-1})=\varphi(x)\varphi(y)^{-1}\in E$, so $xy^{-1}\in\varphi^{-1}(E)$. Hence it is a subgroup.

If $E\trianglelefteq H$ and $g\in G$, $x\in\varphi^{-1}(E)$, then
\[
\varphi(gxg^{-1})=\varphi(g)\varphi(x)\varphi(g)^{-1}\in E.
\]
Thus conjugation by $g$ preserves $\varphi^{-1}(E)$ (equality follows by using $g^{-1}$), proving normality. Apply this to $E=\{1_H\}$ to obtain $\ker\varphi\trianglelefteq G$.
\end{solution}

\begin{exercise}{3.1.2}
Let $\varphi:G\to H$ be a homomorphism with kernel $K$. For $a,b\in\varphi(G)$ let $X=\varphi^{-1}(a)$ and $Y=\varphi^{-1}(b)$ be the fibers. If $u\in X$, $XY=Z$ in $G/K$, and $w\in Z$, prove that $w=uy$ for some $y\in Y$.
\end{exercise}

\begin{hint}
Consider $y=u^{-1}w$ and compute its image under $\varphi$.
\end{hint}

\begin{solution}
The fibers are precisely the cosets of $K=\ker\varphi$. Since $u\in X$, the coset $X$ equals $uK$. The equality $Z=XY$ implies
\[
 \varphi(w)=\varphi(u)\,b=ab.
\]
Set $y=u^{-1}w$. Then
\[
 \varphi(y)=\varphi(u)^{-1}\varphi(w)=a^{-1}(ab)=b,
\]
so $y\in\varphi^{-1}(b)=Y$, and $uy=w$ as required. This directly identifies the element of the second fiber rather than merely showing its existence.
\end{solution}

\begin{exercise}{3.1.7}
Define $\pi:(\R^2,+)\to(\R,+)$ by $\pi(x,y)=x+y$. Show it is a surjective homomorphism and describe its kernel and fibers geometrically.
\end{exercise}

\begin{solution}
For vectors $(x_1,y_1),(x_2,y_2)$, their images satisfy
\[
\pi((x_1,y_1)+(x_2,y_2))=x_1+x_2+y_1+y_2=\pi(x_1,y_1)+\pi(x_2,y_2).
\]
The map is surjective because $\pi(c,0)=c$ for every real $c$. Its kernel is
\[
\ker\pi=\{(t,-t):t\in\R\},
\]
the line $x+y=0$ through the origin of slope $-1$. For any $c\in\R$, the fiber $\pi^{-1}(c)=\{(x,y):x+y=c\}$ is the parallel line $y=-x+c$, which is a coset of the kernel.
\end{solution}
