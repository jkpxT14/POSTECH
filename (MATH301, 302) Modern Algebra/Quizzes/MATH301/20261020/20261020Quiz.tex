\begin{exercise}{3.1.14}
Consider the additive group $\Q/\Z$.
\begin{exercisesubparts}
\item Show that every coset has a unique representative $q\in\Q$ with $0\le q<1$.
\item Show every element has finite order, and orders are unbounded.
\item Show $\Q/\Z$ is the torsion subgroup of $\R/\Z$.
\item Show $\Q/\Z$ is isomorphic to the group of complex roots of unity.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item For $q\in\Q$, subtract its floor $m=\lfloor q\rfloor$. Then $q-m\in[0,1)\cap\Q$ and represents the same coset. If $u,v\in[0,1)$ represent the same coset, then $u-v\in\Z\cap(-1,1)=\{0\}$, so $u=v$.
\item For $q=a/b$ with $b>0$, the coset $q+\Z$ satisfies $b(q+\Z)=\Z$, hence has finite order. The coset $1/n+\Z$ has order exactly $n$ for each positive integer $n$, so the orders are unbounded.
\item An element $x+\Z\in\R/\Z$ has finite order precisely when $nx\in\Z$ for some positive integer $n$, equivalently $x\in\Q$. Hence the torsion subgroup of $\R/\Z$ is $\Q/\Z$.
\item Define $\varphi:\Q/\Z\to\C^\times$ by $\varphi(q+\Z)=e^{2\pi iq}$. It is well defined because $e^{2\pi i(q+m)}=e^{2\pi iq}$ for $m\in\Z$, and it is a homomorphism because exponentials convert sums into products. Its kernel is trivial since $e^{2\pi iq}=1$ exactly when $q\in\Z$. Every root of unity of order $n$ is $e^{2\pi ik/n}$ for some integer $k$, so $\varphi$ maps onto the group of all roots of unity. This gives the desired isomorphism.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{3.1.17}
Let $G=D_8=\langle r,s\mid r^8=s^2=1,\ rs=sr^{-1}\rangle$, the dihedral group of order $16$ (whose subgroup lattice appears in Section~2.5). The subgroup $\langle r^4\rangle=Z(G)$ is normal. Let $\overline G=G/\langle r^4\rangle$, and write $\bar r=r\langle r^4\rangle$, $\bar s=s\langle r^4\rangle$.
\begin{exercisesubparts}
\item Show $|\overline G|=8$.
\item List every element as $\bar s^a\bar r^b$.
\item Give the order of every element in (b).
\item Express $\overline{rs}$, $\overline{sr^{-2}s}$, and $\overline{s^{-1}r^{-1}sr}$ in this form.
\item Show $\overline H=\langle\bar s,\bar r^2\rangle$ is normal and isomorphic to $V_4$; identify its complete preimage in $G$.
\item Find $Z(\overline G)$ and identify $\overline G/Z(\overline G)$.
\end{exercisesubparts}
\end{exercise}

\begin{notation}
The $D_8$ appearing here is the dihedral group of order $16$ under the MATH000 convention. The textbook may index a dihedral group by its order, writing this group as $D_{16}$.
\end{notation}

\begin{solution}
The element $r^4$ is central: it commutes with $r$ and $sr^4s^{-1}=r^{-4}=r^4$. Hence $\langle r^4\rangle\trianglelefteq G$, and the quotient is defined.
\begin{exercisesubparts}
\item Since $|\langle r^4\rangle|=2$, Lagrange's theorem gives $|\overline G|=16/2=8$.
\item The relation $\bar r^4=1$ and the dihedral normal form give
\[
1,\bar r,\bar r^2,\bar r^3,\bar s,\bar s\bar r,\bar s\bar r^2,\bar s\bar r^3.
\]
These eight elements are distinct: the first four are distinct powers of $\bar r$, and the last four belong to the different coset $\bar s\langle\bar r\rangle$.
\item The orders in this order are
\[
 \boxed{1,4,2,4,2,2,2,2}.
\]
Indeed $|\bar r|=4$, so $|\bar r^2|=2$ and $|\bar r^3|=4$. Moreover, $(\bar s\bar r^j)^2=\bar s\bar r^j\bar s\bar r^j=\bar r^{-j}\bar r^j=1$, and none of these reflections is the identity.
\item Using $r^js=sr^{-j}$ and $s^2=1$, we find
\[
\overline{rs}=\bar s\bar r^3,\qquad
\overline{sr^{-2}s}=\bar r^2,\qquad
\overline{s^{-1}r^{-1}sr}=\bar r^2.
\]
Indeed, $rs=sr^{-1}$ gives
\[
rs=sr^{-1},\qquad
sr^{-2}s=r^2,\qquad
s^{-1}r^{-1}sr=sr^{-1}sr=r^2.
\]
In the quotient, $\bar r^{-1}=\bar r^3$, giving the stated normal forms.
\item We have
\[
\overline H=\{1,\bar r^2,\bar s,\bar s\bar r^2\}.
\]
It has index $2$, so it is normal. Since $\bar s$ and $\bar r^2$ are commuting involutions, its three nonidentity elements have order $2$, giving $\overline H\cong V_4$. Its full preimage under $G\to\overline G$ is $\langle r^2,s\rangle$; $r^2$ has order $4$, $s$ has order $2$, and $s r^2s^{-1}=r^{-2}$. It is therefore the dihedral group of order $8$, namely $D_4$ under MATH000.
\item An element $\bar r^j$ commutes with $\bar s$ precisely when $\bar r^{2j}=1$, so $j$ is even. No reflection can be central because it does not commute with $\bar r$ of order $4$. Thus $Z(\overline G)=\langle\bar r^2\rangle$. In the quotient by this center, the commuting images of $\bar r$ and $\bar s$ are two distinct involutions; the quotient has order $4$, so
\[
\overline G/Z(\overline G)\cong V_4.
\]
\end{exercisesubparts}
\end{solution}

\begin{exercise}{3.1.22}
\begin{exercisesubparts}
\item Show that if $H,K\trianglelefteq G$, then $H\cap K\trianglelefteq G$.
\item Show that any nonempty intersection of normal subgroups of $G$ is normal, even when the collection is infinite.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item The intersection is a subgroup. For $g\in G$,
\[
 g(H\cap K)g^{-1}=(gHg^{-1})\cap(gKg^{-1})=H\cap K,
\]
so it is normal.
\item Let $(N_i)_{i\in I}$ be a nonempty family of normal subgroups. Its intersection $N=\bigcap_{i\in I}N_i$ is a subgroup, and for every $g\in G$,
\[
 gNg^{-1}=\bigcap_{i\in I}gN_ig^{-1}=\bigcap_{i\in I}N_i=N.
\]
Thus $N\trianglelefteq G$ regardless of the size of the index set.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{3.1.24}
Prove that if $N\trianglelefteq G$ and $H\le G$, then $N\cap H\trianglelefteq H$.
\end{exercise}

\begin{solution}
The intersection is a subgroup of $H$. For $h\in H$ and $x\in N\cap H$, normality of $N$ gives $hxh^{-1}\in N$, while closure of $H$ gives $hxh^{-1}\in H$. Thus $h(N\cap H)h^{-1}\subseteq N\cap H$. Repeating with $h^{-1}$ gives the reverse inclusion, so $N\cap H\trianglelefteq H$.
\end{solution}

\begin{exercise}{3.1.31}
If $H\le G$ and $N\trianglelefteq H$, show $H\le N_G(N)$. Deduce that $N_G(N)$ is the largest subgroup of $G$ in which $N$ is normal.
\end{exercise}

\begin{solution}
For $h\in H$, normality of $N$ in $H$ says $hNh^{-1}=N$. Thus every $h\in H$ belongs to $N_G(N)$ and $H\le N_G(N)$. Conversely, by definition $N$ is normal in $N_G(N)$, since every element of the normalizer fixes $N$ by conjugation. Consequently $N_G(N)$ is itself one of the eligible subgroups and contains every other one, making it the largest (and their join).
\end{solution}

\begin{exercise}{3.1.33}
Find all normal subgroups of $D_4$ and determine the isomorphism type of the corresponding quotients.
\end{exercise}

\begin{hint}
Use the subgroup lattice in Section 2.5, together with the fact that subgroups of index $2$ are normal.
\end{hint}

\begin{solution}
Write $D_4=\langle r,s\mid r^4=s^2=1,\ srs=r^{-1}\rangle$. By Lagrange's theorem a proper nontrivial normal subgroup has order $2$ or $4$.

If a normal subgroup $N$ has order $2$, its unique nonidentity element is fixed by every conjugation (since an automorphism of $C_2$ is trivial), and therefore lies in $Z(D_4)=\{1,r^2\}$. Hence $N=\langle r^2\rangle$. The five subgroups of order $2$ therefore contribute only this one normal subgroup.

Every subgroup of order $4$ has index $2$ and is normal. The only elements of order $4$ in $D_4$ are $r$ and $r^3$, so a cyclic subgroup of order $4$ must be $\langle r\rangle$. If a subgroup of order $4$ is not cyclic, it is isomorphic to $V_4$ and hence contains three involutions. Since the only involutions are $r^2$ and the four reflections, it must contain $r^2$ and a reflection. Reflections differing by a factor of $r^2$ generate the same subgroup together with $r^2$; the even and odd reflections give precisely $\langle r^2,s\rangle$ and $\langle r^2,sr\rangle$. Thus these are all subgroups of order $4$. Adding the trivial group and the whole group gives all normal subgroups. The quotients are
\[
\begin{array}{c@{\qquad}c}\toprule
N&D_4/N\\\midrule
\{1\}&D_4\\
\langle r^2\rangle&V_4\\
\langle r\rangle&C_2\\
\langle r^2,s\rangle&C_2\\
\langle r^2,sr\rangle&C_2\\
D_4&\{1\}\\\bottomrule\end{array}
\]
The three index-$2$ quotients have order $2$. For the central quotient, the images of $r$ and $s$ are distinct commuting involutions, so it is $V_4$.
\end{solution}

\begin{exercise}{3.1.34}
Let $D_n=\langle r,s\mid r^n=s^2=1,\ rs=sr^{-1}\rangle$, $n\ge3$, and let $k$ be a positive divisor of $n$.
\begin{exercisesubparts}
\item Prove $\langle r^k\rangle\trianglelefteq D_n$.
\item Show $D_n/\langle r^k\rangle$ is dihedral of order $2k$.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item Conjugation by the generators preserves $\langle r^k\rangle$, because $r(r^k)r^{-1}=r^k$ and $s(r^k)s^{-1}=r^{-k}$. Thus $\langle r^k\rangle$ is normal in $D_n$.
\item Let $\bar r=r\langle r^k\rangle$ and $\bar s=s\langle r^k\rangle$. These satisfy
\[
\bar r^k=\bar s^2=1,\qquad \bar s\bar r\bar s^{-1}=\bar r^{-1}.
\]
Since $k\mid n$, the subgroup $\langle r^k\rangle$ has order $n/k$, so the quotient has order $2k$. Its elements are the $2k$ distinct cosets $\bar r^j$ and $\bar s\bar r^j$ ($0\le j<k$), giving the dihedral presentation. For $k\ge3$ the quotient is $D_k$ in MATH000; for $k=1,2$ the degenerate cases are $C_2$ and $V_4$. 
\end{exercisesubparts}
\end{solution}

\begin{exercise}{3.1.40}
Let $N\trianglelefteq G$ and $\bar G=G/N$. Show that $\bar x$ and $\bar y$ commute if and only if $[x,y]=x^{-1}y^{-1}xy\in N$.
\end{exercise}

\begin{solution}
The images commute if and only if
\[
\bar x^{-1}\bar y^{-1}\bar x\bar y=1_{G/N}.
\]
The left-hand side equals $\overline{x^{-1}y^{-1}xy}=\overline{[x,y]}$, which equals the identity coset precisely when $[x,y]\in N$.
\end{solution}

\begin{exercise}{3.1.41}
Prove that $[G,G]=\langle[x,y]:x,y\in G\rangle$ is a normal subgroup of $G$ and that $G/[G,G]$ is abelian.
\end{exercise}

\begin{solution}
Conjugation takes a commutator to a commutator:
\[
 g[x,y]g^{-1}=[gxg^{-1},gyg^{-1}].
\]
Consequently conjugation by any $g\in G$ preserves the subgroup generated by all commutators, so $[G,G]\trianglelefteq G$. Every commutator belongs to this subgroup. By Exercise 3.1.40 the images of any two elements of $G$ therefore commute in the quotient $G/[G,G]$, which is abelian.
\end{solution}

\begin{exercise}{3.2.4}
Show that if $|G|=pq$ for primes $p,q$ (not necessarily distinct), then either $G$ is abelian or $Z(G)=\{1_G\}$.
\end{exercise}

\begin{hint}
Recall that if $G/Z(G)$ is cyclic, then $G$ is abelian.
\end{hint}

\begin{solution}
Suppose $Z(G)$ is nontrivial. If it equals $G$, the group is abelian. Otherwise its order is a proper divisor of $pq$ larger than $1$, necessarily $p$ or $q$, so the quotient $G/Z(G)$ has prime order and is cyclic. If $G/Z(G)=\langle xZ(G)\rangle$, every element of $G$ can be written $x^az$ with $z\in Z(G)$. For any $a,b$ and central $z,w$, the elements $x^az$ and $x^bw$ commute. Hence $G$ is abelian. Contrapositively, if $G$ is not abelian its center is trivial.
\end{solution}

\begin{exercise}{3.2.5}
For $H\le G$ and $g\in G$:
\begin{exercisesubparts}
\item Show $gHg^{-1}$ is a subgroup of the same order as $H$.
\item If $H$ is the unique subgroup of order $n$, show $H\trianglelefteq G$.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item Conjugation $c_g(x)=gxg^{-1}$ is an automorphism of $G$. Its restriction to $H$ is a bijection onto $gHg^{-1}$, and the image is a subgroup. Hence $|gHg^{-1}|=|H|$ even if $H$ is infinite.
\item For each $g\in G$, $gHg^{-1}$ is a subgroup of order $n$. Uniqueness forces $gHg^{-1}=H$. This is exactly the definition of $H\trianglelefteq G$.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{3.2.8}
If $H,K\le G$ are finite subgroups of relatively prime orders, prove $H\cap K=\{1_G\}$.
\end{exercise}

\begin{solution}
The group $H\cap K$ is a subgroup of both $H$ and $K$. By Lagrange's theorem, its order divides both $|H|$ and $|K|$. Since these numbers are coprime, $|H\cap K|=1$, so $H\cap K=\{1_G\}$.
\end{solution}

\begin{exercise}{3.2.11}
For $H\le K\le G$, prove $[G:H]=[G:K][K:H]$, allowing infinite groups and infinite indices.
\end{exercise}

\begin{solution}
Choose representatives $g_i$ ($i\in I$) for the left $K$-cosets in $G$, and representatives $k_j$ ($j\in J$) for the left $H$-cosets in $K$. Every $g\in G$ belongs to exactly one coset $g_iK$; within it $g_i^{-1}g$ belongs to exactly one coset $k_jH$. Thus the cosets $g_ik_jH$ cover $G$. If $g_ik_jH=g_{i'}k_{j'}H$, passing to $K$-cosets gives $i=i'$, and cancellation then gives $j=j'$. Consequently the left $H$-cosets in $G$ are in bijection with $I\times J$, so
\[
[G:H]=|I\times J|=|I|\,|J|=[G:K][K:H].
\]
This is a cardinal equality, also valid for infinite indices.
\end{solution}

\begin{exercise}{3.2.14}
Prove that $S_4$ has no normal subgroup of order $8$ and no normal subgroup of order $3$.
\end{exercise}

\begin{solution}
Suppose $N\trianglelefteq S_4$ has order $8$. Then $S_4/N$ has order $3$. Every transposition has order $2$, so its image in this quotient has order dividing both $2$ and $3$, hence is the identity. Since transpositions generate $S_4$, the quotient map would be trivial, contradicting the quotient's order $3$.

If $N\trianglelefteq S_4$ has order $3$, it contains a $3$-cycle. All eight $3$-cycles in $S_4$ are conjugate, so normality forces $N$ to contain all eight. Adding the identity gives at least nine elements, contradicting $|N|=3$.
\end{solution}

\begin{exercise}{3.2.16}
Use Lagrange's theorem in $(\Z/p\Z)^\times$ to prove Fermat's little theorem: $a^p\equiv a\pmod p$ for every integer $a$, where $p$ is prime.
\end{exercise}

\begin{solution}
If $p\mid a$, both sides are zero modulo $p$. Otherwise $[a]_p\in(\Z/p\Z)^\times$, a group of order $p-1$. Lagrange's theorem gives $[a]_p^{p-1}=[1]_p$. Thus $a^{p-1}\equiv1\pmod p$. Multiplying by $a$ yields $a^p\equiv a\pmod p$, as required.
\end{solution}

\begin{exercise}{3.2.20}
If $A$ is an abelian normal subgroup of $G$ and $B$ is any subgroup, prove $A\cap B\trianglelefteq AB$.
\end{exercise}

\begin{solution}
Because $A\trianglelefteq G$, the product $AB$ is a subgroup of $G$. Put $N=A\cap B$. If $b\in B$, then $bNb^{-1}\subseteq A$ by normality of $A$ and $bNb^{-1}\subseteq B$ by closure of $B$, so $B$ normalizes $N$ (equality using $b^{-1}$). Every $a\in A$ centralizes $N$ because $A$ is abelian. Thus both $A$ and $B$ normalize $N$, hence their product $AB$ does. Since $N\le AB$, it follows that $N\trianglelefteq AB$.
\end{solution}

\begin{exercise}{3.2.21}
Prove that $(\Q,+)$ has no proper subgroups of finite index; deduce the same for $\Q/\Z$.
\end{exercise}

\begin{hint}
The additive group $\Q$ is divisible: for any $x\in\Q$ and positive integer $m$, there is $y$ with $my=x$.
\end{hint}

\begin{solution}
Let $H\le\Q$ and $[\Q:H]=m<\infty$. As $\Q$ is abelian, $\Q/H$ is a group of order $m$, and Lagrange's theorem gives $mq\in H$ for every $q\in\Q$. Given any $x\in\Q$, apply this to $q=x/m$ to obtain $x\in H$. Thus $H=\Q$.

If $K\le\Q/\Z$ has finite index, its inverse image under the quotient map $\pi:\Q\to\Q/\Z$ is a subgroup of $\Q$ with the same finite index. It must equal $\Q$, so $K=\Q/\Z$.
\end{solution}

\begin{exercise}{3.3.3}
If $H\trianglelefteq G$ has prime index $p$, prove that for every subgroup $K\le G$ either $K\le H$, or $G=HK$ and $[K:K\cap H]=p$.
\end{exercise}

\begin{solution}
Because $H$ is normal, $HK$ is a subgroup containing $H$. The index formula gives
\[
 [G:H]=[G:HK][HK:H]=p.
\]
As $p$ is prime, either $HK=H$, so $K\le H$, or $HK=G$. In the latter case, the Second Isomorphism Theorem gives
\[
 K/(K\cap H)\cong HK/H=G/H,
\]
whence $[K:K\cap H]=|G/H|=p$.
\end{solution}

\begin{exercise}{3.3.4}
Suppose $C\trianglelefteq A$ and $D\trianglelefteq B$. Prove $C\times D\trianglelefteq A\times B$ and $(A\times B)/(C\times D)\cong(A/C)\times(B/D)$.
\end{exercise}

\begin{solution}
Conjugation acts coordinatewise, so
\[
(a,b)(C\times D)(a,b)^{-1}=(aCa^{-1})\times(bDb^{-1})=C\times D.
\]
Thus the product subgroup is normal. Define $\varphi:A\times B\to(A/C)\times(B/D)$ by $\varphi(a,b)=(aC,bD)$. The map is a surjective homomorphism whose kernel is $C\times D$, and the First Isomorphism Theorem yields the required isomorphism.
\end{solution}

\begin{exercise}{3.5.3}
Prove that $S_n$ is generated by the adjacent transpositions $s_i=(i\ i+1)$, $1\le i<n$.
\end{exercise}

\begin{hint}
A conjugate of a transposition is another transposition: $\sigma(a\ b)\sigma^{-1}=(\sigma(a)\ \sigma(b))$.
\end{hint}

\begin{solution}
Every permutation is a product of transpositions, so it is enough to express an arbitrary transposition through adjacent ones. If $1\le i<j\le n$, then
\[
(i\ j)=s_i s_{i+1}\cdots s_{j-2}s_{j-1}s_{j-2}\cdots s_{i+1}s_i.
\]
To see this, put $w=s_i\cdots s_{j-2}$. Then $w$ fixes $j$ and sends $j-1$ to $i$ (under right-to-left composition), so $w s_{j-1}w^{-1}$ interchanges $i$ and $j$. The displayed word is exactly $w s_{j-1}w^{-1}$. Thus every transposition, and hence every permutation, is generated by the adjacent transpositions.
\end{solution}

\begin{exercise}{3.5.4}
Show that $S_n=\langle(12),(1\ 2\ \cdots\ n)\rangle$ for all $n\ge2$.
\end{exercise}

\begin{solution}
Put $s=(12)$ and $c=(1\ 2\ \cdots\ n)$. For $1\le i<n$,
\[
 c^{i-1}sc^{-(i-1)}=(i\ i+1).
\]
Thus $\langle s,c\rangle$ contains every adjacent transposition. By Exercise 3.5.3, these generate the whole $S_n$, proving the claim.
\end{solution}

\begin{exercise}{5.1.1}
Prove $Z(G_1\times\cdots\times G_n)=Z(G_1)\times\cdots\times Z(G_n)$. Deduce the product is abelian if and only if every factor is abelian.
\end{exercise}

\begin{solution}
Multiplication in a direct product is coordinatewise. An element $(g_1,\ldots,g_n)$ commutes with every $(h_1,\ldots,h_n)$ precisely when $g_ih_i=h_ig_i$ for every $i$ and every $h_i\in G_i$. This means $g_i\in Z(G_i)$ for each coordinate, proving the formula. The product is abelian exactly when its center is the whole product, which by this formula happens exactly when $Z(G_i)=G_i$ for all $i$.
\end{solution}

\begin{exercise}{5.1.14}
For $G=A_1\times\cdots\times A_n$ and $B_i\trianglelefteq A_i$, prove $B_1\times\cdots\times B_n\trianglelefteq G$ and
\[
 G/(B_1\times\cdots\times B_n)\cong(A_1/B_1)\times\cdots\times(A_n/B_n).
\]
\end{exercise}

\begin{solution}
Conjugation in the direct product is coordinatewise. Thus
\[
(a_i)(\prod_{i=1}^n B_i)(a_i)^{-1}=\prod_{i=1}^n a_iB_ia_i^{-1}=\prod_{i=1}^n B_i,
\]
so $\prod_iB_i$ is normal. The map
\[
\varphi:G\to\prod_{i=1}^n(A_i/B_i),\qquad(a_i)\mapsto(a_iB_i)
\]
is a surjective homomorphism: any cosets have representatives in their respective factors. Its kernel is precisely $\prod_iB_i$. The First Isomorphism Theorem proves the result.
\end{solution}

\begin{exercise}{5.2.2}
For each of the following orders, give the invariant-factor lists for all abelian groups of that order:
\begin{exercisesubparts}
\item $270$;
\item $9801$;
\item $320$;
\item $105$;
\item $44100$.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
We use decreasing invariant factors: $(n_1,\ldots,n_t)$ denotes $C_{n_1}\times\cdots\times C_{n_t}$ with $n_{i+1}\mid n_i$. The classification theorem says to partition the exponents in the prime factorization, combine the corresponding prime powers in aligned columns, and omit factors equal to $1$. We obtain:
\begin{exercisesubparts}
\item $270=2\cdot3^3\cdot5$. Since $3$ has partitions $3$, $2+1$, and $1+1+1$, there are three types:
\[
(270),\qquad(90,3),\qquad(30,3,3).
\]
\item $9801=3^4\cdot11^2$. There are $p(4)p(2)=5\cdot2=10$ types:
\[
\begin{gathered}
(9801)\qquad(3267,3)\qquad(1089,9)\\
(1089,3,3)\qquad(891,11)\qquad(363,3,3,3)\\
(297,33)\qquad(99,99)\qquad(99,33,3)\\
(33,33,3,3).
\end{gathered}
\]
\item $320=2^6\cdot5$. The exponent $6$ has $p(6)=11$ partitions, hence eleven types:
\[
\begin{gathered}
(320)\qquad(160,2)\qquad(80,4)\\
(80,2,2)\qquad(40,8)\qquad(40,4,2)\\
(40,2,2,2)\qquad(20,4,4)\qquad(20,4,2,2)\\
(20,2,2,2,2)\qquad(10,2,2,2,2,2).
\end{gathered}
\]
\item $105=3\cdot5\cdot7$ has only one type because every exponent is $1$:
\[
 (105).
\]
\item $44100=2^2\cdot3^2\cdot5^2\cdot7^2$. Each exponent $2$ has two partitions, so there are $2^4=16$ types:
\[
\begin{gathered}
(44100)\qquad(22050,2)\qquad(14700,3)\\
(8820,5)\qquad(7350,6)\qquad(6300,7)\\
(4410,10)\qquad(3150,14)\qquad(2940,15)\\
(2100,21)\qquad(1470,30)\qquad(1260,35)\\
(1050,42)\qquad(630,70)\qquad(420,105)\\
(210,210).
\end{gathered}
\]
\end{exercisesubparts}
Every displayed list has product equal to the prescribed order and successive factors dividing preceding ones. The classification theorem gives uniqueness of the invariant-factor type; the partition counts $3,10,11,1,16$ match the displayed lists, so none is missing.
\end{solution}

\begin{exercise}{5.2.5}
If a finite abelian group $G$ has invariant factors $(n_1,n_2,\ldots,n_r)$ with $n_{i+1}\mid n_i$, prove $G$ contains an element of order $m$ if and only if $m\mid n_1$. Deduce that the exponent of $G$ is $n_1$.
\end{exercise}

\begin{solution}
Write $G\cong C_{n_1}\times\cdots\times C_{n_r}$. The order of an element is the least common multiple of the orders of its coordinates. Since the order of the $i$th coordinate divides $n_i$ and $n_i\mid n_1$, every element has order dividing $n_1$. Conversely, whenever $m\mid n_1$, the first cyclic factor contains an element of order $m$; adjoining identity elements in the remaining coordinates gives an element of order $m$ in $G$. In particular, the first factor supplies an element of order $n_1$, and every element order divides $n_1$. Hence the exponent (least common multiple of all element orders) is $n_1$.
\end{solution}

\begin{exercise}{5.2.9}
Let $A=C_{60}\times C_{45}\times C_{12}\times C_{36}$. Find the number of elements of order $2$ and the number of subgroups of index $2$ in $A$.
\end{exercise}

\begin{solution}
In a cyclic group of even order, the equation $x^2=1$ has exactly two solutions; in an odd-order cyclic group it has only one. Exactly three factors of $A$ have even order. Hence $2^3=8$ elements satisfy $x^2=1$, and excluding the identity leaves $\boxed{7}$ elements of order exactly $2$.

Every subgroup of index $2$ is normal and is the kernel of a surjective homomorphism $A\to C_2$. Such a homomorphism is determined by its values on the four cyclic generators: each even-order factor has two choices for the image, while the odd-order factor has only the identity choice. Hence there are $2^3-1=7$ nontrivial homomorphisms, all surjective. Two surjections $A\to C_2$ with the same kernel agree, since their induced maps $A/K\to C_2$ differ by an automorphism of $C_2$, and that group has only the identity automorphism. Thus there are $\boxed{7}$ distinct index-$2$ subgroups.
\end{solution}
