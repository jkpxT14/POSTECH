\begin{exercise}{4.5.13}
Prove that a group of order $56$ has a normal Sylow $p$-subgroup for some prime $p$ dividing its order.
\end{exercise}

\begin{solution}
Since $56=2^3\cdot7$, Sylow's theorem gives $n_7\mid8$ and $n_7\equiv1\pmod7$, so $n_7=1$ or $8$. If $n_7=1$, the Sylow $7$-subgroup is normal.

If $n_7=8$, the eight subgroups of order $7$ intersect pairwise only in the identity, because their order is prime. They therefore contain $8(7-1)=48$ distinct nonidentity elements, all of order $7$. The other eight elements consist of the identity and seven elements not of order $7$. Any Sylow $2$-subgroup has order $8$ and cannot contain an element of order $7$, so it is exactly this eight-element set. Consequently all Sylow $2$-subgroups coincide: the unique Sylow $2$-subgroup is normal.
\end{solution}

\begin{exercise}{4.5.16}
Let $|G|=pqr$, where $p<q<r$ are primes. Prove that $G$ has a normal Sylow subgroup for at least one of $p,q,r$.
\end{exercise}

\begin{solution}
Sylow's theorem gives $n_r\mid pq$ and $n_r\equiv1\pmod r$. Because neither $p$ nor $q$ is congruent to $1$ modulo $r$, $n_r=1$ or $pq$. If $n_r=1$, the Sylow $r$-subgroup is normal.

Suppose $n_r=pq$. Its $pq$ distinct subgroups of order $r$ have pairwise trivial intersections and contain $pq(r-1)$ nonidentity elements, leaving only $pq-1$ nonidentity elements outside them. Also $n_q\mid pr$ and $n_q\equiv1\pmod q$. The possible divisors of $pr$ are $1,p,r,pr$; since $p<q$, the value $p$ is not $1$ modulo $q$. Thus if $n_q>1$, necessarily $n_q\ge r$. The distinct Sylow $q$-subgroups would then contribute at least $r(q-1)$ nonidentity elements, disjoint from the elements of order $r$. But
\[
 r(q-1)\ge rp>pq-1,
\]
since $q-1\ge p$ and $r>q$. This is impossible. Therefore $n_q=1$, so a Sylow $q$-subgroup is normal.
\end{solution}

\begin{exercise}{4.5.17}
Prove that every group $G$ of order $105$ has a normal Sylow $5$-subgroup and a normal Sylow $7$-subgroup.
\end{exercise}

\begin{solution}
Since $105=3\cdot5\cdot7$, Sylow's theorem gives $n_5\in\{1,21\}$ and $n_7\in\{1,15\}$. If both were greater than $1$, the distinct subgroups of orders $5$ and $7$ would contain $21\cdot4+15\cdot6=174$ distinct nonidentity elements. This exceeds $104=|G|-1$, so at least one of the two Sylow subgroups is normal.

First suppose $P_7\trianglelefteq G$. Then $G/P_7$ has order $15$. The number of its Sylow $5$-subgroups divides $3$ and is $1$ modulo $5$, hence is $1$. Its unique Sylow $5$-subgroup has a normal preimage $K\trianglelefteq G$ of order $35$. In $K$, the number of Sylow $5$-subgroups divides $7$ and is $1$ modulo $5$, hence is $1$. This unique subgroup of order $5$ is invariant under every conjugation by $G$, because $G$ normalizes $K$ and conjugation preserves subgroup orders; thus it is normal in $G$.

Conversely, if $P_5\trianglelefteq G$, then $G/P_5$ has order $21$. Its number of Sylow $7$-subgroups divides $3$ and is $1$ modulo $7$, so equals $1$. Its preimage is again a normal subgroup $K$ of order $35$. The Sylow $7$-subgroup of $K$ is unique (its number divides $5$ and is $1$ modulo $7$). It is consequently invariant under conjugation by all elements of $G$ and normal in $G$. Hence both Sylow subgroups of $G$ are normal.
\end{solution}

\begin{supplementaryexercise}
Let $G$ be a group with $|G|=30$. Determine whether $G$ can be simple. Justify your answer.
\end{supplementaryexercise}

\begin{solution}
Suppose $G$ were simple. Since $|G|=2\cdot3\cdot5$, Sylow's theorem yields
\[
 n_5\mid6,\quad n_5\equiv1\pmod5
 \qquad\Longrightarrow\qquad n_5\in\{1,6\},
\]
and similarly $n_3\in\{1,10\}$. Simplicity rules out a unique normal Sylow subgroup, so $n_5=6$ and $n_3=10$. Distinct subgroups of prime orders have trivial pairwise intersections. Hence the Sylow $5$-subgroups contribute $6\cdot4=24$ distinct elements of order $5$, and Sylow $3$-subgroups contribute $10\cdot2=20$ distinct elements of order $3$. Counting the identity gives at least $1+24+20=45$ elements, contradicting $|G|=30$. Therefore no group of order $30$ is simple.
\end{solution}

\begin{exercise}{5.1.18}
Give examples of groups with the following properties.
\begin{exercisesubparts}
\item An infinite group in which every element has order $1$ or $2$.
\item An infinite group in which all elements have finite order, with an element of order $n$ for every positive $n$.
\item A group with an infinite-order element and an element of order $2$.
\item A group containing an isomorphic copy of every finite group.
\item A nontrivial group $G$ with $G\cong G\times G$.
\end{exercisesubparts}
\end{exercise}

\begin{solution}
\begin{exercisesubparts}
\item Let $G=\bigoplus_{n\ge1}C_2$. This is infinite, and every element has finite support. Squaring coordinatewise gives the identity, so every nonidentity element has order $2$.
\item Take $G=\bigoplus_{n\ge1}C_n$. Any element has only finitely many nonidentity coordinates, each of finite order, so its order is the least common multiple of finitely many finite numbers. The generator of the $n$th factor gives an element of order $n$.
\item Take $G=\Z\times C_2$. The element $(1,1_{C_2})$ has infinite order, while $(0,t)$ has order $2$ for the nonidentity $t\in C_2$.
\item Take $G=\bigcup_{n\ge1}S_n$, with $S_n\le S_{n+1}$ via permutations fixing $n+1$. This union is a group. Cayley's theorem embeds each finite group $H$ in $S_{|H|}$, hence in $G$.
\item Take again $G=\bigoplus_{n\ge1}C_2$. Splitting the coordinates into odd and even indices gives a group isomorphism $G\cong G\times G$; $G$ is certainly nontrivial.
\end{exercisesubparts}
\end{solution}

\begin{exercise}{5.2.10}
Let $A$ be a free abelian group of rank $n$ and let $k>0$. Show that $A/kA$ is isomorphic to the direct product of $n$ copies of $\Z/k\Z$, where $kA=\{ka:a\in A\}$.
\end{exercise}

\begin{hint}
Choose a basis of $A$ and reduce each integer coefficient modulo $k$.
\end{hint}

\begin{solution}
Choose a basis $a_1,\ldots,a_n$ of $A$. Every $a\in A$ has a unique representation $a=m_1a_1+\cdots+m_na_n$ with integers $m_i$. Define
\[
\begin{aligned}
\varphi:A&\longrightarrow(\Z/k\Z)^n,\\
\sum_{i=1}^n m_i a_i&\longmapsto([m_1]_k,\ldots,[m_n]_k).
\end{aligned}
\]
Uniqueness of the coefficients makes the map well defined, and addition of coefficients shows it is a homomorphism. It is surjective because every tuple of residue classes has integer representatives. Moreover,
\[
\varphi(a)=0\iff k\mid m_i\text{ for every }i\iff a\in kA,
\]
so $\ker\varphi=kA$. By the First Isomorphism Theorem,
\[
A/kA\cong(\Z/k\Z)^n.
\]
\end{solution}
