\begin{problem}{1.1}
Let $G$ be a finite group. Suppose that $G$ has no element of order $2$. Prove that $|G|$ is odd.
\end{problem}

\begin{solution}
For every $g\in G\setminus\{1_G\}$, we have $g\neq g^{-1}$; otherwise $g^2=1_G$, so $g$ would have order $2$. Hence the elements of $G\setminus\{1_G\}$ occur in disjoint pairs
\[
    \{g,g^{-1}\}.
\]
Thus $|G|-1$ is even, and therefore $|G|$ is odd.
\end{solution}

\begin{problem}{1.2}
Let $\sigma\in S_n$ have order $6$. Prove that if the cycle decomposition of $\sigma$ contains no $6$-cycle, then it must contain both a $2$-cycle and a $3$-cycle.
\end{problem}

\begin{solution}
Write the disjoint cycle decomposition of $\sigma$ as
\[
    \sigma=c_1c_2\cdots c_r,
\]
where the nontrivial cycles $c_i$ have lengths $m_i$. Then
\[
    |\sigma|=\lcmop(m_1,\ldots,m_r)=6.
\]
Hence each $m_i$ divides $6$. Since there is no $6$-cycle, every $m_i$ is $2$ or $3$. If there were no $2$-cycle, then $|\sigma|$ would divide $3$; if there were no $3$-cycle, then $|\sigma|$ would divide $2$. Both contradict $|\sigma|=6$. Therefore the cycle decomposition contains both a $2$-cycle and a $3$-cycle.
\end{solution}

\begin{problem}{2.1}
Let $G$ be a group and let $\mathcal S$ be the set of all subgroups of $G$. Show that
\[
    g\cdot H=gHg^{-1},
\]
for $g\in G$ and $H\in\mathcal S$, defines a left group action of $G$ on $\mathcal S$.
\end{problem}

\begin{solution}
First, the proposed map is well defined. For $g\in G$ and $H\le G$, we have
\[
    1_G=g1_Gg^{-1}\in gHg^{-1}.
\]
If $gh_1g^{-1},gh_2g^{-1}\in gHg^{-1}$, then
\[
    (gh_1g^{-1})(gh_2g^{-1})^{-1}
    =g(h_1h_2^{-1})g^{-1}\in gHg^{-1},
\]
since $h_1h_2^{-1}\in H$. By the Subgroup Criterion, $gHg^{-1}\le G$.

Now let $g_1,g_2\in G$ and $H\in\mathcal S$. Then
\[
\begin{aligned}
    g_1\cdot(g_2\cdot H)
    &=g_1(g_2Hg_2^{-1})g_1^{-1}\\
    &=(g_1g_2)H(g_1g_2)^{-1}\\
    &=(g_1g_2)\cdot H,
\end{aligned}
\]
and
\[
    1_G\cdot H=1_GH1_G^{-1}=H.
\]
Thus $g\cdot H=gHg^{-1}$ defines a left group action of $G$ on $\mathcal S$.
\end{solution}

\begin{problem}{2.2}
Let $C_{12}=\langle x\rangle$ and $C_{18}=\langle y\rangle$ be cyclic groups of orders $12$ and $18$, respectively. Find, with proof, the number of homomorphisms
\[
    \varphi\colon C_{12}\to C_{18}.
\]
\end{problem}

\begin{notation}
Following MATH000, $C_n$ denotes an abstract cyclic group of order $n$. The original quiz uses $Z_n$ for the same object.
\end{notation}

\begin{solution}
A homomorphism $\varphi\colon C_{12}\to C_{18}$ is determined by the image of the generator $x$. Write
\[
    \varphi(x)=y^a,
    \qquad 0\le a<18.
\]
Since $x^{12}=1$, we must have
\[
    1=\varphi(x^{12})=y^{12a},
\]
so
\[
    18\mid12a
    \quad\Longleftrightarrow\quad
    3\mid a.
\]
Thus the only possible values are
\[
    a=0,3,6,9,12,15.
\]

Conversely, for each such $a$, define
\[
    \varphi_a(x^k)=y^{ak}.
\]
If $x^k=x^\ell$, then $12\mid(k-\ell)$; since $18\mid12a$, it follows that $18\mid a(k-\ell)$, so $y^{ak}=y^{a\ell}$. Hence $\varphi_a$ is well defined, and
\[
    \varphi_a(x^{k+\ell})
    =y^{a(k+\ell)}
    =y^{ak}y^{a\ell}
    =\varphi_a(x^k)\varphi_a(x^\ell),
\]
so $\varphi_a$ is a homomorphism. These six homomorphisms are distinct because they send $x$ to distinct elements of $C_{18}$. Therefore the number of homomorphisms is
\[
    \boxed{6}.
\]
\end{solution}

\begin{problem}{3.1}
Let $G$ act on a nonempty set $A$. For $a\in A$, define the orbit of $a$ by
\[
    G\cdot a=\{g\cdot a\mid g\in G\}.
\]
Prove that if $b\in G\cdot a$, then $G\cdot b=G\cdot a$. Deduce that the distinct orbits form a partition of $A$.
\end{problem}

\begin{solution}
Suppose that $b\in G\cdot a$, so $b=g\cdot a$ for some $g\in G$. Then
\[
\begin{aligned}
    G\cdot b
    &=\{h\cdot(g\cdot a)\mid h\in G\}\\
    &=\{(hg)\cdot a\mid h\in G\}\\
    &=G\cdot a,
\end{aligned}
\]
since $\{hg\mid h\in G\}=G$.

Every $a\in A$ lies in $G\cdot a$, so the orbits cover $A$. If two orbits $G\cdot a$ and $G\cdot b$ intersect, choose $c\in(G\cdot a)\cap(G\cdot b)$. By the first part,
\[
    G\cdot a=G\cdot c=G\cdot b.
\]
Hence any two distinct orbits are disjoint, and the distinct orbits form a partition of $A$.
\end{solution}

\begin{problem}{3.2}
Let $G$ be a group and let $\varphi\in\Aut(G)$. Define
\[
    \operatorname{Fix}(\varphi)=\{g\in G\mid\varphi(g)=g\}.
\]
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.25em,topsep=0.4em]
    \item Prove that $\operatorname{Fix}(\varphi)$ is a subgroup of $G$.
    \item Let $G=\langle x\rangle$ be cyclic of order $12$, and let $\varphi$ be the automorphism determined by $\varphi(x)=x^5$. Determine $\operatorname{Fix}(\varphi)$.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.45em,topsep=0.2em]
    \item Since $\varphi(1_G)=1_G$, the set $\operatorname{Fix}(\varphi)$ is nonempty. If $a,b\in\operatorname{Fix}(\varphi)$, then
    \[
        \varphi(ab^{-1})=\varphi(a)\varphi(b)^{-1}=ab^{-1}.
    \]
    Thus $ab^{-1}\in\operatorname{Fix}(\varphi)$, so $\operatorname{Fix}(\varphi)\le G$ by the Subgroup Criterion.

    \item Every element of $G$ has the form $x^k$, and
    \[
    \begin{aligned}
        x^k\in\operatorname{Fix}(\varphi)
        &\iff x^{5k}=x^k\\
        &\iff x^{4k}=1_G\\
        &\iff 12\mid4k\\
        &\iff 3\mid k.
    \end{aligned}
    \]
    Therefore
    \[
        \operatorname{Fix}(\varphi)
        =\langle x^3\rangle
        =\{1_G,x^3,x^6,x^9\}.
    \]
\end{enumerate}
\end{solution}

\begin{problem}{4.1}
Let $G$ be a group and let $H\le G$. Prove that $H$ is abelian if and only if
\[
    \langle H,C_G(H)\rangle=C_G(H).
\]
\end{problem}

\begin{solution}
By definition, $H$ is abelian if and only if every element of $H$ commutes with every element of $H$, which is equivalent to $H\le C_G(H)$. Since $C_G(H)$ is a subgroup of $G$, we also have
\[
    H\le C_G(H)
    \quad\Longleftrightarrow\quad
    \langle H,C_G(H)\rangle=C_G(H).
\]
Combining these equivalences proves the claim.
\end{solution}

\begin{problem}{4.2}
Let $\varphi\colon G\to H$ be a group homomorphism. Determine whether each of the following statements is true or false. For each false statement, give a counterexample.
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.25em,topsep=0.4em]
    \item $\varphi(1_G)=1_H$.
    \item If $A\trianglelefteq G$, then $\varphi(A)\trianglelefteq H$.
    \item If $E\le H$, then $\varphi^{-1}(E)\le G$.
    \item If $E\trianglelefteq H$, then $\varphi^{-1}(E)\trianglelefteq G$.
    \item If $x\in G$ has finite order, then $|x|$ divides $|\varphi(x)|$.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.55em,topsep=0.2em]
    \item \textbf{True.} Since $\varphi(1_G)^2=\varphi(1_G)$, cancellation in $H$ gives $\varphi(1_G)=1_H$.

    \item \textbf{False.} Define $\varphi\colon\Z/2\Z\to S_3$ by $\varphi([k]_2)=(12)^k$ and take $A=\Z/2\Z$. Although $A\trianglelefteq\Z/2\Z$, its image $\langle(12)\rangle$ is not normal in $S_3$, since
    \[
        (13)\langle(12)\rangle(13)^{-1}
        =\langle(23)\rangle\neq\langle(12)\rangle.
    \]
    The image of a normal subgroup is normal in $\varphi(G)$, but need not be normal in $H$.

    \item \textbf{True.} The set $\varphi^{-1}(E)$ contains $1_G$. If $x,y\in\varphi^{-1}(E)$, then
    \[
        \varphi(xy^{-1})=\varphi(x)\varphi(y)^{-1}\in E,
    \]
    so $xy^{-1}\in\varphi^{-1}(E)$. Thus $\varphi^{-1}(E)\le G$.

    \item \textbf{True.} By (c), $\varphi^{-1}(E)\le G$. For $g\in G$ and $x\in\varphi^{-1}(E)$,
    \[
        \varphi(gxg^{-1})
        =\varphi(g)\varphi(x)\varphi(g)^{-1}\in E
    \]
    because $E\trianglelefteq H$. Hence $gxg^{-1}\in\varphi^{-1}(E)$, proving normality.

    \item \textbf{False.} The reduction homomorphism
    \[
        \varphi\colon\Z/4\Z\to\Z/2\Z,
        \qquad [k]_4\mapsto[k]_2
    \]
    sends the element $[1]_4$ of order $4$ to $[1]_2$ of order $2$, and $4\nmid2$. In general, the correct divisibility is $|\varphi(x)|\mid|x|$ for an element $x$ of finite order.
\end{enumerate}
\end{solution}


\begin{problem}{5.1}
Let $G$ be a group, and define
\[
G'=\langle x^{-1}y^{-1}xy\mid x,y\in G\rangle.
\]
($G'$ is called the commutator (or derived) subgroup of $G$.)
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.2em,topsep=0.35em]
\item Prove that $G'$ is a normal subgroup of $G$.
\item Let $H\le G$. Prove that
\[
G'\subseteq H\quad\Longleftrightarrow\quad H\trianglelefteq G\text{ and }G/H\text{ is abelian}.
\]
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.55em,topsep=0.2em]
\item For any $g,x,y\in G$,
\[
g[x,y]g^{-1}=[gxg^{-1},gyg^{-1}]\in G'.
\]
Conjugation by $g$ is an automorphism of $G$, so it sends the subgroup generated by these commutators into $G'$. Thus $gG'g^{-1}\subseteq G'$ for every $g\in G$. Applying the same inclusion to $g^{-1}$ gives the reverse inclusion. Hence $gG'g^{-1}=G'$ and $G'\trianglelefteq G$.
\item Suppose first that $G'\subseteq H$. For $g\in G$ and $h\in H$, the element
\[
ghg^{-1}h^{-1}=[g^{-1},h^{-1}]\in G'\subseteq H,
\]
so $ghg^{-1}\in H$. Hence $H\trianglelefteq G$. Moreover, since $[x,y]\in H$ for all $x,y\in G$, their images in $G/H$ commute, and $G/H$ is abelian.

Conversely, suppose $H\trianglelefteq G$ and $G/H$ is abelian. For all $x,y\in G$ we have $(xH)(yH)=(yH)(xH)$, so $[x,y]H=H$. Thus every commutator belongs to $H$, and therefore $G'\subseteq H$.
\end{enumerate}
\end{solution}

\begin{problem}{5.2}
Let $A$ and $B$ be groups, and let $C\trianglelefteq A$ and $D\trianglelefteq B$. Define
\[
\varphi\colon A\times B\longrightarrow(A/C)\times(B/D),
\qquad \varphi(a,b)=(aC,bD).
\]
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.2em,topsep=0.35em]
\item Prove that $\varphi$ is a surjective group homomorphism.
\item Determine $\ker\varphi$ and prove that
\[
(A\times B)/(C\times D)\cong(A/C)\times(B/D).
\]
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.55em,topsep=0.2em]
\item For $(a_1,b_1),(a_2,b_2)\in A\times B$,
\[
\begin{aligned}
\varphi((a_1,b_1)(a_2,b_2))
&=(a_1a_2C,b_1b_2D)\\
&=(a_1C,b_1D)(a_2C,b_2D)\\
&=\varphi(a_1,b_1)\varphi(a_2,b_2).
\end{aligned}
\]
Thus $\varphi$ is a homomorphism. Every $(aC,bD)$ is the image of $(a,b)$, so $\varphi$ is surjective.
\item The identity of $(A/C)\times(B/D)$ is $(C,D)$, hence
\[
\begin{aligned}
\ker\varphi
&=\{(a,b)\in A\times B\mid aC=C,\ bD=D\}\\
&=C\times D.
\end{aligned}
\]
By the First Isomorphism Theorem and surjectivity,
\[
(A\times B)/(C\times D)
=(A\times B)/\ker\varphi
\cong\im\varphi=(A/C)\times(B/D).
\]
\end{enumerate}
\end{solution}

\begin{problem}{6.1}
Suppose that $\varphi\colon G\to H$ is a surjective group homomorphism, where $G$ is a finite group. Let $K=\ker\varphi$. If $P$ is a Sylow $p$-subgroup of $G$, prove that $\varphi(P)$ is also a Sylow $p$-subgroup of $H$.
\end{problem}
\begin{solution}
Since $K\trianglelefteq G$, the product $PK$ is a subgroup of $G$. By the First Isomorphism Theorem,
\[
H\cong G/K,\qquad \varphi(P)\cong P/(P\cap K)\cong PK/K.
\]
Thus $|\varphi(P)|=|P|/|P\cap K|$ is a power of $p$. Moreover,
\[
[H:\varphi(P)]=[G:PK]
\]
divides $[G:P]$, since $P\le PK\le G$. As $P$ is Sylow in $G$, $p\nmid[G:P]$; hence $p\nmid[H:\varphi(P)]$. Therefore $\varphi(P)$ has $p$-power order and index prime to $p$, so it is a Sylow $p$-subgroup of $H$.
\end{solution}

\begin{problem}{6.2}
Suppose that $\varphi\colon G\to H$ is a surjective group homomorphism, where $G$ is a finite group. Let $K=\ker\varphi$. If $P_1$ and $P_2$ are Sylow $p$-subgroups of $G$ with $\varphi(P_1)=\varphi(P_2)$, prove that $P_1$ and $P_2$ are conjugate by an element of $K$.
\end{problem}
\begin{solution}
For any subgroup $P\le G$, one has $\varphi^{-1}(\varphi(P))=PK$. Since $\varphi(P_1)=\varphi(P_2)$, it follows that
\[
P_1K=P_2K=:L.
\]
Both $P_1$ and $P_2$ lie in $L$ and have the largest possible $p$-power order among subgroups of $G$; hence they are Sylow $p$-subgroups of $L$. By Sylow conjugacy in $L$, there is $x\in L$ with $P_2=xP_1x^{-1}$.

As $K\trianglelefteq G$, we have $L=P_1K=KP_1$. Write $x=ky$ with $k\in K$ and $y\in P_1$. Then
\[
P_2=(ky)P_1(ky)^{-1}=kP_1k^{-1},
\]
so the required conjugating element lies in $K$.
\end{solution}

\begin{problem}{7.1}
Let $R$ be a commutative ring with identity $1\neq0$, and let $a\in R$ satisfy $a^3=0$.
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.2em,topsep=0.35em]
\item Prove that $1-a$ is a unit in $R$, and find its inverse.
\item Let $u$ be a unit in $R$. Prove that $u-a$ is a unit in $R$.
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.55em,topsep=0.2em]
\item Since $a^3=0$,
\[
(1-a)(1+a+a^2)=1-a^3=1.
\]
Thus $1-a$ is a unit and $(1-a)^{-1}=1+a+a^2$.
\item As $R$ is commutative, $(u^{-1}a)^3=u^{-3}a^3=0$. Applying (a) to $u^{-1}a$ shows that $1-u^{-1}a$ is a unit. Since
\[
u-a=u(1-u^{-1}a),
\]
and the product of units is a unit, $u-a$ is a unit.
\end{enumerate}
\end{solution}

\begin{problem}{7.2}
Define
\[
I=\{f(x)\in\Z[x]\mid f(1)\equiv0\pmod3\}.
\]
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.2em,topsep=0.35em]
\item Prove that $I$ is an ideal of $\Z[x]$.
\item Prove that $\Z[x]/I\cong\Z/3\Z$ as rings.
\end{enumerate}
\end{problem}
\begin{solution}
\begin{enumerate}[label=(\alph*),leftmargin=2.5em,itemsep=0.55em,topsep=0.2em]
\item Clearly $0\in I$. For $f,g\in I$,
\[
(f-g)(1)=f(1)-g(1)\equiv0\pmod3,
\]
so $f-g\in I$. For $h\in\Z[x]$ and $f\in I$,
\[
(hf)(1)=h(1)f(1)\equiv0\pmod3,
\]
so $hf\in I$. Since $\Z[x]$ is commutative, $I$ is an ideal.
\item Define
\[
\psi\colon\Z[x]\longrightarrow\Z/3\Z,
\qquad \psi(f)=[f(1)]_3.
\]
Evaluation at $1$ and reduction modulo $3$ both preserve addition, multiplication, and the identity; thus $\psi$ is a ring homomorphism. Every class $[n]_3$ is $\psi(n)$, so $\psi$ is surjective. Finally,
\[
\ker\psi=\{f\in\Z[x]\mid f(1)\equiv0\pmod3\}=I.
\]
The First Isomorphism Theorem now gives $\Z[x]/I\cong\Z/3\Z$.
\end{enumerate}
\end{solution}
