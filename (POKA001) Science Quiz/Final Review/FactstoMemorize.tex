\begin{fact}[No Nobel Prize in Mathematics]
Nobel's will listed physics, chemistry, physiology or medicine, literature, and peace; mathematics was not included. The romantic-rivalry story is unsupported.
\end{fact}

\finalsection{Fields Medal and ICM}
\begin{fact}[Fields Basics]
ICM = \textit{International Congress of Mathematicians}; IMU = \textit{International Mathematical Union}. A candidate's 40th birthday must not occur before January 1 of the Congress year. First medals: \boxed{1936, Oslo}, Lars Ahlfors and Jesse Douglas.
\end{fact}

\begin{fact}[Fields Medal Design]
The medal bears Archimedes. A sphere and circumscribed cylinder are associated with Archimedes' celebrated $2:3$ volume ratio and his tomb tradition.
\end{fact}

\finalsection{Anecdotes with Quiz Value}
\begin{fact}[1729: Hardy--Ramanujan Number]
\[
\boxed{1729=1^3+12^3=9^3+10^3}.
\]
Smallest positive integer expressible as the sum of two positive cubes in two distinct ways.
\end{fact}

\begin{fact}[Dantzig's Homework]
George Dantzig copied two problems from the board believing they were homework; they were famous unsolved statistics problems, and he solved them. The story is often used as a recognition clue for Dantzig.
\end{fact}

\begin{fact}[Galois and the Duel]
Évariste Galois died at age 20 after a duel. His work connected polynomial equations with permutation groups and founded Galois theory.
\end{fact}

\begin{fact}[Hamilton and Brougham Bridge]
Hamilton famously carved the quaternion multiplication relation into Brougham Bridge in Dublin after the insight
\[
i^2=j^2=k^2=ijk=-1.
\]
\end{fact}

\begin{fact}[The Scottish Book Goose]
Per Enflo solved a longstanding basis problem recorded in the Scottish Book and received the promised live goose from Stanisław Mazur.
\end{fact}

\begin{fact}[Witten's Undergraduate Major]
Edward Witten majored in \boxed{history} as an undergraduate before becoming a central figure in mathematical physics and the first physicist to receive a Fields Medal.
\end{fact}

\begin{fact}[Perelman's Refusals]
Perelman declined the 2006 Fields Medal and later the Clay Millennium Prize money associated with the Poincaré conjecture.
\end{fact}

\begin{fact}[Birkar's Stolen Medal]
Caucher Birkar's 2018 Fields Medal was stolen shortly after the Rio ceremony; a replacement medal was later presented.
\end{fact}

\begin{fact}[Archimedes' Tomb]
Tradition says Archimedes wanted the sphere-and-cylinder figure marking his favorite result on his tomb. This image also resonates with the Fields Medal design.
\end{fact}
