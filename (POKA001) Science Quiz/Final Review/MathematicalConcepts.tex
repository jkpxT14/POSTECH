\finalchapter{1}{Set Theory and Logic}

\begin{formula}[논리적 동치]
\[
(P\Rightarrow Q)\Longleftrightarrow(\neg Q\Rightarrow\neg P),\qquad
\neg(P\Rightarrow Q)\Longleftrightarrow P\land\neg Q,
\]
\[
\neg(P\land Q)\Longleftrightarrow(\neg P)\lor(\neg Q),\qquad
\neg(P\lor Q)\Longleftrightarrow(\neg P)\land(\neg Q).
\]
또한
\[
\neg(\forall x\in A\,P(x))\Longleftrightarrow
\exists x\in A:\ \neg P(x).
\]
\end{formula}

\begin{theorem}[기수의 핵심 사실]
\[
|\N|=|\Z|=|\Q|=\aleph_0,\qquad
|(0,1)|=|\R|=\mathfrak c=2^{\aleph_0}>\aleph_0.
\]
유한집합 $A$에 대해 $|\Pow(A)|=2^{|A|}$. 소수의 집합은 $\N$의 부분집합이므로 가산이다.
\end{theorem}

\begin{theorem}[칸토어 정리와 슈뢰더--베른슈타인 정리]
모든 집합 $A$에 대해 $|A|<|\Pow(A)|$. 또한 $A\hookrightarrow B$와
$B\hookrightarrow A$가 모두 존재하면 $|A|=|B|$.
\end{theorem}

\begin{fact}[집합론 기초의 핵심 사실]
러셀의 역설은 순진한 집합론의 대표 모순이다. ZFC는 체르멜로--프렝켈 집합론(ZF)에 선택공리를 더한 체계이다.
ZF 위에서 선택공리, 초른의 보조정리, 정렬정리는 서로 동치이다.
연속체 가설(CH)은 $\mathfrak c=\aleph_1$이며, 괴델과 코언의 결과로 ZFC로부터 독립임이 알려져 있다
(일관성 가정 아래).
\end{fact}

\finalchapter{2}{Calculus and Analysis}
\finalsection{One-Variable Calculus}

\begin{formula}[산술합·등비급수·거듭제곱합]
\[
\sum_{k=1}^n k=\frac{n(n+1)}2,\quad
\sum_{k=1}^n k^2=\frac{n(n+1)(2n+1)}6,\quad
\sum_{k=1}^n k^3=\left(\frac{n(n+1)}2\right)^2,
\]
\[
\sum_{k=0}^{n-1}ar^k=a\frac{1-r^n}{1-r},\qquad
|r|<1\Rightarrow \sum_{k=0}^{\infty}ar^k=\frac{a}{1-r}.
\]
\end{formula}

\begin{formula}[점근식]
\[
n!\sim\sqrt{2\pi n}\left(\frac ne\right)^n,\qquad
H_n=\ln n+\gamma+o(1).
\]
\end{formula}

\begin{formula}[미분 공식: 실수하기 쉬운 항목]
\[
(\arcsin x)'=\frac1{\sqrt{1-x^2}},\qquad
(\arccos x)'=-\frac1{\sqrt{1-x^2}},\qquad
(\arctan x)'=\frac1{1+x^2},
\]
\[
(a^x)'=a^x\ln a,\qquad
(\log_a x)'=\frac1{x\ln a}.
\]
곱·몫·연쇄법칙은 별도 암기 항목이 아니라 즉시 쓰는 도구로 본다.
\end{formula}

\begin{formula}[테일러·매클로린 급수]
\[
e^x=\sum_{n=0}^{\infty}\frac{x^n}{n!},\qquad
\sin x=x-\frac{x^3}{3!}+\frac{x^5}{5!}-\cdots,
\]
\[
\cos x=1-\frac{x^2}{2!}+\frac{x^4}{4!}-\cdots,
\]
\[
\ln(1+x)=x-\frac{x^2}{2}+\frac{x^3}{3}-\cdots,\qquad
\ln(1-x)=-\sum_{n=1}^{\infty}\frac{x^n}{n},
\]
\[
(1+x)^\alpha=\sum_{n=0}^{\infty}\binom{\alpha}{n}x^n.
\]
\end{formula}

\begin{theorem}[미적분학의 기본정리]
연속함수 $f$에 대해
\[
\frac{\dd}{\dd x}\int_a^x f(t)\,\dd t=f(x),\qquad
\int_a^b f(x)\,\dd x=F(b)-F(a)\quad(F'=f).
\]
\end{theorem}

\begin{theorem}[롤의 정리와 평균값 정리]
적절한 연속성·미분가능성 가정 아래
\[
f(a)=f(b)\Longrightarrow\exists c\in(a,b):f'(c)=0,
\]
\[
\exists c\in(a,b):\quad f'(c)=\frac{f(b)-f(a)}{b-a}.
\]
\end{theorem}

\begin{formula}[부분적분과 치환적분]
\[
\int u\,\dd v=uv-\int v\,\dd u,
\qquad
\int_a^b f(g(t))g'(t)\,\dd t=\int_{g(a)}^{g(b)}f(x)\,\dd x.
\]
정적분 치환에서는 적분구간의 경계도 함께 바꾼다.
\end{formula}

\begin{formula}[역함수의 미분]
$y=f(x)$, $f'(x)\neq0$이면
\[
(f^{-1})'(y)=\frac1{f'(x)}.
\]
\end{formula}

\begin{formula}[자주 쓰는 부정적분]
\[
\int\frac{\dd x}{a^2+x^2}=\frac1a\arctan\frac xa+C,\qquad
\int\frac{\dd x}{\sqrt{a^2-x^2}}=\arcsin\frac xa+C,
\]
\[
\int\frac{\dd x}{x^2-a^2}=\frac1{2a}\ln\left|\frac{x-a}{x+a}\right|+C.
\]
\end{formula}

\begin{formula}[가우스 적분과 베타·감마 함수]
\[
\int_{-\infty}^{\infty}e^{-x^2}\,\dd x=\sqrt\pi,\qquad
\Gamma(s)=\int_0^\infty x^{s-1}e^{-x}\,\dd x,
\]
\[
B(p,q)=\int_0^1x^{p-1}(1-x)^{q-1}\,\dd x
=\frac{\Gamma(p)\Gamma(q)}{\Gamma(p+q)}.
\]
특히 $\Gamma(n+1)=n!$, $\Gamma(1/2)=\sqrt\pi$.
\end{formula}

\begin{formula}[아크탄젠트 망원합]
주값 범위에서 $1+uv>0$이면
\[
\arctan u-\arctan v=\arctan\left(\frac{u-v}{1+uv}\right).
\]
특히
\[
\arctan\left(\frac{a}{x(x+1)+a^2}\right)=\arctan\frac ax-\arctan\frac a{x+1}.
\]
\end{formula}

\begin{formula}[로그 적분]
$a>-1$, $m\in\Nzero$이면
\[
\int_0^1x^a(\ln x)^m\,\dd x=\frac{(-1)^m m!}{(a+1)^{m+1}}.
\]
따라서
\[
\int_0^1\frac{\ln x}{1-x}\,\dd x=-\frac{\pi^2}{6}.
\]
\end{formula}

\begin{formula}[바젤 급수와 교대조화급수]
\[
\sum_{n=1}^{\infty}\frac1{n^2}=\frac{\pi^2}{6},\qquad
1-\frac12+\frac13-\frac14+\cdots=\ln 2.
\]
따라서 홀수 제곱 역수합은 $\pi^2/8$.
\end{formula}

\begin{formula}[급수 판정과 월리스 곱]
\[
\sum_{n=1}^{\infty}\frac1{n^p}\text{ 는 }p>1\text{일 때, 그리고 그때에만 수렴한다.}
\]
비율판정법과 근판정법은 극한이 $<1$이면 절대수렴, $>1$이면 발산.
또한
\[
\frac{\pi}{2}=\prod_{n=1}^{\infty}\frac{(2n)^2}{(2n-1)(2n+1)}
\]
은 월리스 곱이다.
\end{formula}

\begin{formula}[자주 쓰는 무한급수]
\[
\sum_{n=1}^{\infty}\frac{x^n}{n}=-\ln(1-x)\qquad(|x|<1),
\]
\[
\sum_{n=0}^{\infty}(-1)^n\frac{x^{2n+1}}{2n+1}=\arctan x.
\]
특히 $x=1$에서 그레고리--라이프니츠 급수
\[
\frac\pi4=1-\frac13+\frac15-\frac17+\cdots.
\]
\end{formula}

\noindent\textbf{수열·급수를 만나면.} 먼저 표준급수인지, 망원합으로 소거되는지, 지배항 비교가 가능한지 본다. 계승은 스털링 공식, 지수·거듭제곱 비교는 로그, 양항급수는 비교·비율·근판정, 교대급수는 절대수렴 여부와 교대급수 판정을 확인한다. 급수와 무한과정의 역사적 단서는 오일러와 코시를 함께 떠올린다.

\begin{fact}[소포모어의 꿈 (Sophomore's Dream)]
특이하고 출제하기 좋은 항등식이다:
\[
\boxed{\displaystyle\int_0^1x^{-x}\,\dd x=\sum_{n=1}^{\infty}\frac1{n^n}}.
\]
\end{fact}

\begin{fact}[$e$, $\pi$와 열린 문제]
에르미트는 $e$의 초월성을, 린데만은 $\pi$의 초월성을 증명했다. 그러나
\[
e+\pi,\qquad e\pi
\]
각각이 무리수인지조차 알려져 있지 않다. 단, 둘이 동시에 대수적일 수는 없다.
\end{fact}

\begin{formula}[프룰라니 적분]
적절한 가정 아래
\[
\int_0^\infty\frac{f(ax)-f(bx)}x\,\dd x=(f(0)-f(\infty))\ln\frac ba.
\]
형태가 보이면 $f(0)$와 $f(\infty)$를 먼저 확인한다.
\end{formula}

\noindent\textbf{적분을 만나면.} 정적분은 먼저 구간과 피적분함수의 대칭을 본다. $x\mapsto a-x$, $x\mapsto1/x$, 홀짝성을 확인하고, 분자에 분모의 미분이 있는지와 표준 $\arctan$·가우스·베타·감마 형태인지 본다. 그다음 치환이나 부분적분을 고려한다. 처음부터 원시함수를 억지로 찾지 않는다. 베타·감마 함수가 보이면 오일러와 연결한다.

\begin{formula}[거듭제곱의 비교]
양수 $a,b$에 대해 $a^b$와 $b^a$는 $b\ln a$와 $a\ln b$를 비교한다.
함수 $\ln x/x$는 $x>e$에서 감소한다.
\end{formula}

\begin{formula}[$x^x$의 최솟값]
$y=x^x$에 로그를 취하면 $\ln y=x\ln x$. 따라서
\[
\boxed{\min_{x>0}x^x=e^{-1/e}}
\]
이고 최소점은 $x=e^{-1}$이다.
\end{formula}

\finalsection{Multivariable Calculus}

\begin{formula}[그래디언트·방향미분·헤시안]
\[
\nabla f=(f_{x_1},\dots,f_{x_n})^{\mathsf T},\qquad
D_{\mathbf u}f=\nabla f\cdot\mathbf u\quad(\|\mathbf u\|=1).
\]
임계점에서는 헤시안의 부호를 이용해 극값을 판정한다.
\end{formula}

\begin{definition}[야코비 행렬과 전미분]
${\mathbf F}:\R^n\to\R^m$이면
\[
D\mathbf F(\mathbf a)=\left(\frac{\partial F_i}{\partial x_j}(\mathbf a)\right).
\]
미분가능성은
\[
\mathbf F(\mathbf a+\mathbf h)=\mathbf F(\mathbf a)+D\mathbf F(\mathbf a)\mathbf h+\mathbf r(\mathbf h),
\qquad
\frac{\|\mathbf r(\mathbf h)\|}{\|\mathbf h\|}\to0
\]
인 최적 선형근사가 존재한다는 뜻이다.
\end{definition}

\begin{theorem}[다변수 연쇄법칙]
\[
D(\mathbf G\circ\mathbf F)(\mathbf a)=D\mathbf G(\mathbf F(\mathbf a))\,D\mathbf F(\mathbf a).
\]
야코비 행렬의 곱 순서를 바꾸지 않는다.
\end{theorem}

\begin{formula}[이계도함수 판정]
임계점에서 헤시안이 양의 정부호이면 엄밀한 국소최솟값, 음의 정부호이면 엄밀한 국소최댓값,
부정부호이면 안장점이다.
\end{formula}

\begin{theorem}[라그랑주 승수법]
제약식 $g(\mathbf x)=c$ 위의 정칙 극값에서는
\[
\boxed{\nabla f=\lambda\nabla g}.
\]
여러 제약식이면 제약식들의 그래디언트의 선형결합을 쓴다.
\end{theorem}

\begin{formula}[변수변환]
$\mathbf x=\mathbf F(\mathbf u)$에 대해
\[
\dd\mathbf x=|\Jac(\mathbf F)(\mathbf u)|\,\dd\mathbf u.
\]
특히
\[
\dd A=r\,\dd r\,\dd\theta,
\qquad
\dd V=\rho^2\sin\phi\,\dd\rho\,\dd\phi\,\dd\theta.
\]
\end{formula}

\begin{formula}[벡터해석 기본식]
\[
\nabla\times(\nabla f)=\mathbf0,\qquad
\nabla\cdot(\nabla\times\mathbf F)=0,\qquad
\Delta f=\nabla\cdot\nabla f.
\]
단일연결 영역에서 충분히 매끄러운 $\mathbf F$가 회전이 $0$이면 보존장이다.
\end{formula}

\begin{theorem}[그린·스토크스·발산 정리]
형태를 알아보는 것이 핵심이다.
\[
\oint_{\partial D}P\,\dd x+Q\,\dd y=\iint_D(Q_x-P_y)\,\dd A,
\]
\[
\int_{\partial S}\mathbf F\cdot\dd\mathbf r=\iint_S(\nabla\times\mathbf F)\cdot\mathbf n\,\dd S,
\]
\[
\iint_{\partial V}\mathbf F\cdot\mathbf n\,\dd S=\iiint_V\nabla\cdot\mathbf F\,\dd V.
\]
\end{theorem}

\finalsection{Real Analysis}

\begin{theorem}[균등수렴 극한 정리]
연속함수열 $f_n$이 $f$로 균등수렴하면 $f$도 연속이다.
연속성 보존에는 점별수렴이 아니라 균등수렴이 핵심이다.
\end{theorem}

\begin{fact}[국소 균등수렴]
함수열에서 극한과 미분·적분의 순서를 바꿀 때는 점별수렴만으로 판단하지 않는다.
국소 균등수렴, 도함수의 균등제어, 지배수렴정리 같은 추가 조건이 보이면 교환 가능성을 확인한다.
\end{fact}

\begin{theorem}[콤팩트성과 연결성]
$\R^n$에서 콤팩트는 닫히고 유계인 것과 동치이다(하이네--보렐 정리).
연속함수는 콤팩트 집합을 콤팩트 집합으로, 연결집합을 연결집합으로 보낸다.
따라서 $f:\R\to\Q$가 연속이면 $f$는 상수함수이다.
\end{theorem}

\begin{formula}[코시--슈바르츠 부등식과 삼각부등식]
\[
|\langle\mathbf u,\mathbf v\rangle|\le\|\mathbf u\|\,\|\mathbf v\|,
\qquad
\|\mathbf u+\mathbf v\|\le\|\mathbf u\|+\|\mathbf v\|.
\]
코시--슈바르츠 부등식에서 등호가 성립할 필요충분조건은 두 벡터가 선형종속인 것이다.
\end{formula}

\begin{fact}[칸토어 집합]
표준 칸토어 집합은 닫히고 유계이며, 완전하고 어디에서도 조밀하지 않으며, 비가산이고 르베그 측도가 $0$이다.
``구간을 계속 제거했는데도 비가산''이라는 대비를 기억한다.
\end{fact}

\begin{theorem}[무리수 회전의 조밀성]
$\alpha\notin\Q$이면 $\{n\alpha\}$는 $[0,1]$에서 조밀하다.
주기함수에 정수열을 대입하는 문제에서는 무리수 주기에 대한 나머지들의 조밀성을 의심한다.
\end{theorem}

\noindent\textbf{극한·함수열을 만나면.} 점별수렴만으로 극한과 미분·적분을 교환하지 않는다. 연속성 보존에는 균등수렴을, 적분과 극한의 교환에는 지배수렴형 조건을, 미분의 교환에는 도함수의 균등한 제어를 확인한다.\par\medskip

\finalsection{Complex Analysis}

\begin{formula}[오일러 공식과 복소로그]
\[
e^{i\theta}=\cos\theta+i\sin\theta,\qquad z=re^{i\theta}.
\]
주편각은
\[
\Arg z\in(-\pi,\pi]
\]
로 두고
\[
\Log z=\ln|z|+i\Arg z
\]
를 주로그로 정의한다.
\end{formula}

\begin{definition}[정칙함수]
$f(z)=u(x,y)+iv(x,y)$가 복소미분 가능하면 정칙(holomorphic)이라 한다. 충분히 매끄러운 경우 가장 먼저 확인할 조건은 코시--리만 방정식이다.
\end{definition}

\begin{formula}[코시--리만 방정식]
\[
\boxed{u_x=v_y,\qquad u_y=-v_x}.
\]
\end{formula}

\begin{theorem}[코시 적분공식]
$f$가 적절한 영역에서 정칙이고 $a$가 $C$ 내부에 있으면
\[
f(a)=\frac1{2\pi i}\oint_C\frac{f(z)}{z-a}\,\dd z,
\]
더 일반적으로
\[
f^{(n)}(a)=\frac{n!}{2\pi i}\oint_C\frac{f(z)}{(z-a)^{n+1}}\,\dd z.
\]
특히
\[
\oint_{|z|=1}\frac{\dd z}{z}=2\pi i.
\]
\end{theorem}

\begin{theorem}[유수정리]
\[
\oint_C f(z)\,\dd z=2\pi i\sum_{a_k\text{가 }C\text{ 내부}}\Res(f;a_k).
\]
단순극 $a$에서 $f=g/h$, $h(a)=0$, $h'(a)\neq0$이면
\[
\Res(f;a)=\frac{g(a)}{h'(a)}.
\]
\end{theorem}

\begin{theorem}[루셰 정리]
단순 폐곡선 $C$ 위에서 $|g|<|f|$이면 $f$와 $f+g$는 $C$ 내부에 같은 개수의 영점을 갖는다
(중복도 포함).
\end{theorem}

\begin{theorem}[리우빌 정리와 대수학의 기본정리]
유계 전해함수는 상수함수이다. 모든 비상수 복소다항식은 복소근을 갖는다.
\end{theorem}

\noindent\textbf{복소해석 문제를 만나면.} 먼저 특이점과 적분경로의 위치를 그린다. 단순한 $(z-a)^{-1}$ 꼴이면 코시 적분공식을, 유리함수의 폐곡선 적분이면 유수정리를, 영점의 개수를 묻고 지배항이 보이면 루셰 정리를 생각한다. 분지를 정해야 하는 $\log z$나 $z^\alpha$에서는 $\Arg z$의 범위를 먼저 고정한다. 코시와 리만을 복소해석의 중심 인물로 연결한다.\par\medskip

\finalchapter{3}{Linear Algebra}

\begin{formula}[계수·영공간과 부분공간]
$\mathbf A\in\F^{m\times n}$이면
\[
\rk(\mathbf A)+\dim\Null(\mathbf A)=n.
\]
부분공간 $U,V$에 대해
\[
\dim(U+V)=\dim U+\dim V-\dim(U\cap V),
\]
따라서
\[
\dim(U\cap V)\ge \dim U+\dim V-n.
\]
\end{formula}

\begin{formula}[행렬식의 기본 성질]
\leavevmode
\begin{itemize}
\item 행 교환: 행렬식의 부호가 바뀐다.
\item 한 행에 $c$를 곱하면 행렬식도 $c$배.
\item 한 행에 다른 행의 배를 더해도 행렬식은 변하지 않는다.
\item 삼각행렬에서는 $\det(\mathbf A)=\prod_i a_{ii}$.
\item $\det(\mathbf A\mathbf B)=\det(\mathbf A)\det(\mathbf B)$.
\end{itemize}
\end{formula}

\noindent\textbf{행렬식을 만나면.} 삼각·블록 구조, 행·열 연산, 선형종속, $\mathbf L\mathbf U$나 $\mathbf L\mathbf L^{\mathsf T}$ 같은 분해를 먼저 확인한다. 여인수 전개는 구조가 없을 때의 마지막 수단으로 둔다.

\begin{theorem}[가역성과 동치인 조건]
$n\times n$ 행렬 $\mathbf A$에 대해
\[
\det(\mathbf A)\ne0
\Longleftrightarrow\rk(\mathbf A)=n
\Longleftrightarrow\Null(\mathbf A)=\{\mathbf0\}
\Longleftrightarrow0\notin\spec(\mathbf A).
\]
또한 $\mathbf A\mathbf B=\mathbf I\Rightarrow\mathbf B\mathbf A=\mathbf I$.
\end{theorem}

\begin{formula}[특성다항식과 표준 연산]
\[
p_{\mathbf A}(\lambda)=\det(\lambda\mathbf I-\mathbf A).
\]
$\mathbf A\mathbf v=\lambda\mathbf v$이면
\[
\mathbf A^k\mathbf v=\lambda^k\mathbf v,
\qquad
p(\mathbf A)\mathbf v=p(\lambda)\mathbf v.
\]
$\mathbf A$가 가역이면 $\mathbf A^{-1}$의 대응 고유값은 $\lambda^{-1}$이고,
$\mathbf A+c\mathbf I$는 고유값을 $c$만큼 이동시킨다.
\end{formula}

\begin{formula}[대각합·행렬식·고유값]
고유값을 중복도 포함 $\lambda_1,\dots,\lambda_n$라 하면
\[
\tr(\mathbf A)=\sum_i\lambda_i,\qquad \det(\mathbf A)=\prod_i\lambda_i.
\]
$2\times2$에서는
\[
\lambda^2-\tr(\mathbf A)\lambda+\det(\mathbf A)=0.
\]
가역 $2\times2$ 행렬에 대해
\[
\tr(\mathbf A^{-1})=\frac{\tr(\mathbf A)}{\det(\mathbf A)}.
\]
\end{formula}

\noindent\textbf{고유값·고유벡터를 만나면.} 곧바로 $\det(\lambda\mathbf I-\mathbf A)$를 전개하지 않는다. 대각·삼각·대칭·계수 1·전부 1인 행렬·회전·멱등·멱영 같은 구조가 있는지, $\tr(\mathbf A)$와 $\det(\mathbf A)$만으로 충분한지, 이미 알려진 다항식 관계가 있는지부터 본다. 구조가 없을 때 특성다항식으로 돌아간다. 큰 거듭제곱은 케일리--해밀턴, 표준형은 조르당, 실대칭행렬은 스펙트럼 정리로 연결한다.

\begin{theorem}[$\mathbf A\mathbf B$와 $\mathbf B\mathbf A$]
$\mathbf A\mathbf B$와 $\mathbf B\mathbf A$는 $0$이 아닌 고유값을 중복도까지 공유한다.
정사각행렬 $\mathbf A,\mathbf B$에 대해서는 특성다항식도 같다.
\end{theorem}

\begin{fact}[계수 1 행렬과 전부 1인 행렬]
외적 $\mathbf u\mathbf v^{\mathsf T}$의 계수는 최대 $1$이다.
\[
\mathbf J_n=\ones\ones^{\mathsf T}
\]
의 고유값은 $n$ 한 개와 $0$ $n-1$개이다. 따라서 $\mathbf J_n-\mathbf I_n$의 고유값은
$n-1$ 한 개와 $-1$ $n-1$개이다.
\end{fact}

\begin{formula}[외적의 합]
\[
\mathbf A=\sum_{i=1}^k\mathbf u_i\mathbf v_i^{\mathsf T}
\quad\Longrightarrow\quad
\rk(\mathbf A)\le k.
\]
각 외적의 계수가 최대 $1$이라는 사실과 계수의 준가법성을 사용한다.
\end{formula}

\begin{fact}[멱등·멱영·반대칭 행렬]
$\mathbf A^2=\mathbf A$이면 고유값은 $0,1$뿐이고 $\tr(\mathbf A)=\rk(\mathbf A)$.
$\mathbf A^k=\mathbf0$이면 모든 고유값은 $0$.
실수 반대칭행렬의 계수는 짝수이며, 홀수 차수에서는 행렬식이 $0$이다.
\end{fact}

\begin{formula}[멱영 조르당 블록]
멱영 조르당 블록 $J_s(0)$에 대해
\[
\rk(J_s(0)^k)=\max(s-k,0).
\]
여러 블록이면 이를 합한다.
\end{formula}

\begin{theorem}[대각화와 스펙트럼 정리]
서로 다른 고유값에 대응하는 고유벡터는 선형독립이다.
실수 대칭행렬은 직교대각화 가능하다:
\[
\mathbf A=\mathbf Q\mathbf D\mathbf Q^{\mathsf T}.
\]
\end{theorem}

\begin{formula}[정사영과 특이값분해]
\[
\proj_{\mathbf u}\mathbf v=\frac{\mathbf v\cdot\mathbf u}{\mathbf u\cdot\mathbf u}\mathbf u.
\]
특이값분해(SVD):
\[
\mathbf A=\mathbf U\boldsymbol{\Sigma}\mathbf V^{\mathsf T},
\]
특이값은 $\mathbf A^{\mathsf T}\mathbf A$의 고유값의 음이 아닌 제곱근이다.
\end{formula}

\begin{fact}[$\R^3$의 회전]
$\mathbf R\in SO(3)$인 고유 회전은 회전축 방향의 실수 고유값 $1$을 갖고,
나머지는 보통 $e^{\pm i\theta}$. 따라서 $\det(\mathbf R)=1$.
\end{fact}

\begin{fact}[파스칼형 행렬식]
이항계수로 이루어진 파스칼형 행렬은 행·열 점화나 $\mathbf L\mathbf L^{\mathsf T}$형 삼각분해를 먼저 찾는다.
표준 파스칼 행렬의 행렬식은 $1$로 떨어지는 경우가 많다.
\end{fact}

\begin{fact}[LU 분해와 피벗]
일반적으로
\[
\mathbf P\mathbf A=\mathbf L\mathbf U
\]
형태를 사용한다. 행 순열이 바뀌면 피벗 값 자체는 달라질 수 있다.
\end{fact}

\begin{theorem}[케일리--해밀턴 정리]
모든 정사각행렬은 자신의 특성다항식을 만족한다:
\[
p_{\mathbf A}(\mathbf A)=\mathbf0.
\]
따라서 큰 거듭제곱 $\mathbf A^N$은 낮은 차수의 $\mathbf I,\mathbf A,\dots,\mathbf A^{n-1}$ 조합으로 줄일 수 있다.
특히 별도의 다항식 관계가 주어지면 그 관계로 $x^N$의 나머지를 취하는 것이 빠르다.
\end{theorem}

\begin{formula}[계수 부등식]
\[
\rk(\mathbf A\mathbf B)\le\min(\rk(\mathbf A),\rk(\mathbf B)),\qquad \rk(\mathbf A+\mathbf B)\le\rk(\mathbf A)+\rk(\mathbf B).
\]
$\mathbf A=\mathbf P\mathbf Q^{\mathsf T}$이고 $\mathbf Q^{\mathsf T}$가 전사이면 $\im(\mathbf A)=\im(\mathbf P)$.
\end{formula}

\noindent\textbf{행렬의 계수를 만나면.} 열공간·영공간과 계수--영공간 차원 정리를 먼저 보고, 외적의 합이나 $\mathbf P\mathbf Q$ 꼴의 분해가 보이면 그것을 이용한다. 계수 부등식으로 충분한지 확인한 뒤 필요할 때만 소거법을 쓴다.

\finalchapter{4}{Differential Equations}


\begin{formula}[일계 지수 성장·감쇠]
\[
y'=ay\quad\Longrightarrow\quad y=Ce^{ax}.
\]
$a>0$이면 성장, $a<0$이면 감쇠이다.
\end{formula}

\begin{formula}[변수분리형 미분방정식]
\[
y'=g(x)h(y)\quad\Longrightarrow\quad\int\frac{\dd y}{h(y)}=\int g(x)\,\dd x.
\]
로지스틱 방정식
\[
y'=ry\left(1-\frac yK\right)
\]
의 해는
\[
y(t)=\frac{K}{1+Ce^{-rt}}.
\]
\end{formula}

\begin{formula}[일계 선형 미분방정식]
\[
y'+p(x)y=q(x),\qquad
\mu(x)=e^{\int p(x)\,\dd x},
\]
\[
(\mu y)'=\mu q.
\]
\end{formula}

\begin{formula}[상수계수 선형 미분방정식]
\[
ay''+by'+cy=0
\]
에는 특성방정식 $ar^2+br+c=0$을 쓴다.
중근이면 $(C_1+C_2x)e^{rx}$, 복소근 $\alpha\pm i\beta$이면
$e^{\alpha x}(C_1\cos\beta x+C_2\sin\beta x)$.
\end{formula}

\begin{formula}[상수계수 연립미분방정식]
\[
\mathbf x'(t)=\mathbf A\mathbf x(t)
\quad\Longrightarrow\quad
\mathbf x(t)=e^{t\mathbf A}\mathbf x(0).
\]
대각화 가능하면 $e^{\lambda t}\mathbf v$ 꼴의 고유모드들의 선형결합으로 본다.
\end{formula}

\begin{fact}[세 기본 편미분방정식]
\[
\text{열방정식: }u_t=\kappa\Delta u,
\qquad
\text{파동방정식: }u_{tt}=c^2\Delta u,
\qquad
\text{라플라스방정식: }\Delta u=0.
\]
확산, 파동 전파, 조화함수·평형을 각각 대표한다.
\end{fact}

\begin{formula}[베셀 방정식]
\[
x^2y''+xy'+(x^2-\nu^2)y=0.
\]
표준해는 베셀함수 $J_\nu(x),Y_\nu(x)$. 원통대칭 편미분방정식에서 자주 나타난다.
\end{formula}

\noindent\textbf{미분방정식을 만나면.} 먼저 변수분리형인지, 일계 선형형인지, 상수계수 선형식인지 형태를 분류한다. 연립식은 $e^{t\mathbf A}$와 고유모드를 보고, 편미분방정식에서는 열방정식·파동방정식·라플라스방정식 중 어떤 구조인지부터 식별한다.

\finalchapter{5}{Combinatorics and Discrete Mathematics}

\begin{formula}[기본 계수법]
\[
{}_nP_r=\frac{n!}{(n-r)!},\qquad \binom nr=\frac{n!}{r!(n-r)!}.
\]
중복조합:
\[
x_1+\cdots+x_k=n,\ x_i\ge0\quad\Longrightarrow\quad\binom{n+k-1}{k-1}.
\]
\end{formula}

\begin{theorem}[이항정리]
\[
(x+y)^n=\sum_{k=0}^n\binom nkx^{n-k}y^k,
\]
\[
\sum_{k=0}^n\binom nk=2^n,
\qquad
\sum_{k=0}^n(-1)^k\binom nk=0\quad(n\ge1).
\]
\end{theorem}

\begin{formula}[격자경로]
$(0,0)$에서 $(m,n)$까지 오른쪽 또는 위쪽으로만 이동하면
\[
\binom{m+n}{m}
\]
개. 대각선 제약이 붙으면 반사원리 또는 카탈란 수를 먼저 본다.
\end{formula}

\begin{formula}[완전순열]
\[
D_n=n!\sum_{k=0}^n\frac{(-1)^k}{k!}
=\left\lfloor\frac{n!}{e}+\frac12\right\rfloor.
\]
\end{formula}

\begin{formula}[방데르몽 항등식과 중복조합]
\[
\sum_k\binom rk\binom s{n-k}=\binom{r+s}{n}.
\]
$x_1+\cdots+x_m=n$의 음이 아닌 정수해 수는 $\binom{n+m-1}{m-1}$,
양의 정수해 수는 $\binom{n-1}{m-1}$.
\end{formula}

\begin{theorem}[포함배제 원리]
\[
\left|\bigcup_{i=1}^nA_i\right|=\sum|A_i|-\sum|A_i\cap A_j|+\cdots.
\]
전사함수 개수는
\[
\#\operatorname{Surj}([m],[n])=\sum_{j=0}^n(-1)^j\binom nj(n-j)^m.
\]
\end{theorem}

\begin{theorem}[비둘기집 원리]
$N$개 물건을 $k$개 상자에 넣으면 어떤 상자에는 적어도 $\lceil N/k\rceil$개가 들어간다.
\end{theorem}

\begin{formula}[간격·블록 계수법]
$n$개 위치에서 서로 이웃하지 않게 $k$개를 고르는 방법:
\[
\binom{n-k+1}{k}.
\]
서로 다른 $n$개 중 지정된 $r$개가 연속하도록 배열:
\[
r!(n-r+1)!.
\]
\end{formula}

\begin{formula}[카탈란 수와 반사원리]
\[
C_n=\frac1{n+1}\binom{2n}{n}.
\]
격자경로가 $y=x$를 넘지 않는 문제, 올바른 괄호열, 이진트리 등에서 등장한다.
금지선을 처음 넘는 지점까지 반사하는 반사원리를 함께 기억한다.
\end{formula}

\begin{theorem}[홀의 결혼정리]
이분그래프에서 왼쪽 모든 정점을 서로 다른 오른쪽 정점과 매칭할 수 있을 필요충분조건은 모든 $S$에 대해
\[
\boxed{|N(S)|\ge|S|}.
\]
\end{theorem}

\begin{fact}[램지 수의 핵심값]
\[
\boxed{R(3,3)=6}.
\]
6명이 있으면 서로 모두 아는 3명 또는 서로 모두 모르는 3명이 반드시 존재한다.
\end{fact}

\begin{fact}[여그래프]
\[
\omega(G)=\alpha(\overline G),\qquad
\alpha(G)=\omega(\overline G).
\]
클리크와 독립집합은 여그래프에서 서로 뒤바뀐다.
\end{fact}

\begin{theorem}[번사이드 보조정리]
유한군 $G$가 집합 $X$에 작용하면 궤도의 개수는
\[
\frac1{|G|}\sum_{g\in G}|\operatorname{Fix}(g)|.
\]
회전·대칭으로 같은 것을 식별하는 색칠 문제에서 바로 쓴다.
\end{theorem}

\begin{formula}[생성함수와 계수추출]
\[
[x^n]f(x)=\frac{f^{(n)}(0)}{n!}.
\]
피보나치 수열의 생성함수:
\[
\frac1{1-x-x^2}=\sum_{n\ge0}F_{n+1}x^n.
\]
테일러 계수와 점화식을 연결하는 대표적인 형태이다.
\end{formula}

\begin{formula}[선형 점화식]
\[
a_n=c_1a_{n-1}+\cdots+c_ka_{n-k}
\]
에는 특성다항식을 쓴다. 피보나치 수열에서는 $r^2-r-1=0$이다.
\end{formula}

\begin{theorem}[분할정복 마스터 정리]
\[
T(n)=aT(n/b)+f(n),\qquad d=\log_ba.
\]
표준형에서는 $f(n)$과 $n^d$의 크기를 비교한다. 서로 다른 크기의 부분문제에는 그대로 적용하지 않는다.
\end{theorem}

\begin{formula}[그래프 기본식]
악수 보조정리:
\[
\sum_{v\in V}\deg(v)=2|E|.
\]
트리에서는 $|E|=|V|-1$. 완전그래프 $K_n$의 생성트리 수는 $n^{n-2}$ (케일리 공식).
여그래프에 대해서는
\[
|E(G)|+|E(\overline G)|=\binom n2,\qquad
\deg_G(v)+\deg_{\overline G}(v)=n-1.
\]
\end{formula}

\begin{theorem}[오일러 트레일 판정]
연결 그래프에서 모든 꼭짓점 차수가 짝수이면 오일러 회로가 존재한다.
정확히 두 꼭짓점만 홀수 차수이면 그 두 점을 끝점으로 하는 오일러 트레일이 존재한다.
\end{theorem}

\begin{theorem}[평면그래프의 오일러 공식]
연결 평면그래프의 평면 매입에서
\[
V-E+F=2.
\]
단순 평면그래프에서 $V\ge3$이면 $E\le3V-6$, 삼각형이 없으면 $E\le2V-4$.
\end{theorem}

\begin{formula}[모저의 원 문제]
원 위 $n$점을 모든 현으로 잇고 내부에서 세 현이 한 점에 만나지 않으면 최대 영역 수는
\[
\boxed{1+\binom n2+\binom n4}.
\]
첫 값은 $1,2,4,8,16,31,\ldots$로 $n=6$에서 $2^{n-1}$ 꼴이 깨진다.
\end{formula}

\begin{fact}[오일러 경로와 해밀턴 경로]
오일러 경로는 모든 \textbf{변}을 정확히 한 번, 해밀턴 경로는 모든 \textbf{꼭짓점}을 정확히 한 번 방문한다.
\end{fact}

\begin{formula}[하노이 탑]
\[
T(n)=2T(n-1)+1,\qquad T(n)=2^n-1.
\]
\end{formula}

\begin{formula}[무작위 순열의 순환]
$X_r=$ 길이 $r$인 순환의 개수라 하면
\[
\Exp[X_r]=\frac1r,\qquad \Exp[C_n]=H_n.
\]
특히 고정점의 기댓값은 $1$이다.
\end{formula}

\noindent\textbf{경우의 수 문제를 만나면.} 직접 세기보다 여사건·대칭·전단사로 단순화되는지 먼저 본다. 제한 조건이 있으면 포함배제, 간격·블록, 반사원리와 카탈란 수를 확인하고, 점화가 자연스러우면 생성함수나 선형점화식으로 옮긴다.\par\medskip

\finalchapter{6}{Probability and Statistics}

\begin{formula}[조건부확률과 베이즈 정리]
\[
\Prob(A\mid B)=\frac{\Prob(A\cap B)}{\Prob(B)},\qquad
\Prob(A\mid B)=\frac{\Prob(B\mid A)\Prob(A)}{\Prob(B)}.
\]
\end{formula}

\begin{definition}[독립]
\[
A\indep B\Longleftrightarrow\Prob(A\cap B)=\Prob(A)\Prob(B).
\]
서로소 사건과 독립사건은 다르다. 양의 확률을 갖는 서로소 사건은 독립일 수 없다.
\end{definition}

\begin{formula}[기댓값과 분산]
\[
\Exp[aX+bY]=a\Exp[X]+b\Exp[Y]
\]
은 독립성 없이 성립한다.
\[
\Var(X)=\Exp[X^2]-\Exp[X]^2.
\]
독립이면 $\Exp[XY]=\Exp[X]\Exp[Y]$, $\Var(X+Y)=\Var(X)+\Var(Y)$.
\end{formula}

\begin{proposition}[독립확률변수 곱의 기댓값]
독립인 적분가능 확률변수에 대해
\[
\Exp[X_1\cdots X_n]=\prod_i\Exp[X_i].
\]
합의 기댓값과 달리 곱의 기댓값에서는 독립성이 중요하다.
\end{proposition}

\begin{formula}[전체기댓값·전체분산·공분산]
\[
\Exp[X]=\Exp[\Exp[X\mid Y]],
\]
\[
\Var(X)=\Exp[\Var(X\mid Y)]+\Var(\Exp[X\mid Y]),
\]
\[
\Cov(X,Y)=\Exp[XY]-\Exp[X]\Exp[Y].
\]
독립이면 공분산은 $0$이지만 그 역은 일반적으로 거짓이다.
\end{formula}

\begin{formula}[꼬리합 공식]

\begin{formula}[유클리드 알고리즘과 베주 항등식]

\gcd(a,n)\mid b.

a^{-1}\pmod n\text{가 존재}\Longleftrightarrow\gcd(a,n)=1.

\begin{fact}[작은 법에서의 제곱잉여]

x^2\pmod4\in\{0,1\},\qquad

정수해 존재성 문제에서는 $2,4,8,3,5,9$ 같은 작은 법을 먼저 본다.

\begin{formula}[오일러 피 함수]

\varphi(n)=n\prod_{p\mid n}\left(1-\frac1p\right).

특히 $p$가 소수이면 $\varphi(p^k)=p^k-p^{k-1}$.

\begin{formula}[곱셈적 위수]

큰 지수는 곱셈적 위수로 줄인다.

\begin{theorem}[중국인의 나머지 정리]

\begin{theorem}[베르트랑 공준]

\begin{theorem}[이차상호법칙]

\begin{formula}[약수의 개수와 합]
\[
n=\prod p_i^{a_i}\quad\Longrightarrow\quad\tau(n)=\prod(a_i+1).
\]
$\tau(n)$이 홀수일 필요충분조건은 $n$이 완전제곱수인 것이다.
고정된 $\tau(n)$에서 최소 $n$을 찾을 때 큰 지수를 작은 소수부터 배치한다.
\end{formula}

\begin{formula}[르장드르·쿠머 정리]
\[
v_p(n!)=\sum_{k\ge1}\left\lfloor\frac n{p^k}\right\rfloor.
\]
쿠머 정리: $v_p\binom nk$는 $p$진법에서 $k+(n-k)$를 더할 때 발생하는 받아올림의 개수와 같다.
\end{formula}


\begin{theorem}[지수 올리기 보조정리 (LTE)]
홀수 소수 $p$에 대해 $p\mid a-b$이고 $p\nmid ab$이면, 모든 $n\in\N$에 대해
\[
v_p(a^n-b^n)=v_p(a-b)+v_p(n).
\]
또한 $p\mid a+b$이고 $n$이 홀수이면 $a^n+b^n$에도 같은 형태를 쓴다. $p=2$에는 별도의 보정항이 있으므로 조건을 확인한다.
\end{theorem}

\begin{theorem}[루카 정리]
$p$가 소수이고
\[
n=\sum_i n_ip^i,\qquad k=\sum_i k_ip^i\qquad(0\le n_i,k_i<p)
\]
이면
\[
\binom nk\equiv\prod_i\binom{n_i}{k_i}\pmod p.
\]
큰 이항계수를 소수로 나눈 나머지를 물으면 $p$진법의 각 자리로 분해한다.
\end{theorem}

\]
\end{fact}

\noindent\textbf{정수론 문제를 만나면.} 정수해의 존재 여부는 작은 법의 합동식부터 보고, 큰 지수는 $\gcd$를 확인한 뒤 페르마--오일러 정리와 곱셈적 위수로 줄인다. 서로 다른 법이 함께 나오면 중국인의 나머지 정리, $a^n\pm b^n$의 $p$-진 지수는 LTE, 이항계수를 소수로 나눈 나머지는 루카·쿠머, $n!$의 $p$-진 지수는 르장드르 공식을 먼저 떠올린다. 소수의 분포가 나오면 유클리드·오일러에서 가우스·리만, 소수정리로 이어지는 흐름을 연결한다.\par\medskip

\finalchapter{9}{Algebra}

\begin{fact}[유한군]
\[
|S_n|=n!,\qquad |A_n|=\frac{n!}{2}\quad(n\ge2).
\]
라그랑주 정리에 의해 부분군의 위수는 군의 위수를 나눈다.
\end{fact}

\begin{theorem}[궤도--안정자 정리]
군 $G$가 $X$에 작용하면 유한군의 경우
\[
|Gx|=\frac{|G|}{|G_x|}.
\]
\end{theorem}

\begin{theorem}[제1동형정리]
군 준동형 $\varphi:G\to H$에 대해
\[
\boxed{G/\ker\varphi\cong\im\varphi}.
\]
특히 $\varphi$가 단사일 필요충분조건은 $\ker\varphi=\{e\}$이다.
\end{theorem}

\begin{formula}[순환군 준동형]
\[
|\operatorname{Hom}(\Z/m\Z,\Z/n\Z)|=\gcd(m,n).
\]
$\Z/n\Z$의 생성원 수는 $\varphi(n)$이다.
\end{formula}

\begin{formula}[순열]
순열의 위수는 서로소 순환들의 길이의 최소공배수이다:
\[
\boxed{\ord(\sigma)=\lcmop(\ell_1,\ldots,\ell_r)}.
\]
\end{formula}

\begin{fact}[유한생성군]
유한군은 유한생성군이다. 유한생성군은 많아야 가산이다.
따라서 비가산군인 $\C^\times$는 유한생성군일 수 없다.
또한 유한개의 점을 뺀 원판의 기본군은 유한 계수의 자유군이다.
\end{fact}

\begin{theorem}[켤레근 정리]
$p(x)\in\R[x]$이고 $p(\alpha)=0$이면
\[
p(\overline\alpha)=0.
\]
중복도도 같다. 따라서 실수계수 다항식의 비실수근은 켤레쌍으로 나타난다.
\end{theorem}


\begin{theorem}[유리근 정리와 아이젠슈타인 판정법]
정수계수 다항식
\[
p(x)=a_nx^n+\cdots+a_0
\]
의 기약분수형 유리근 $r/s$에 대해 $r\mid a_0$, $s\mid a_n$이다. 또한 어떤 소수 $p$가 모든 비최고차항 계수를 나누고, $p\nmid a_n$, $p^2\nmid a_0$이면 $p(x)$는 $\Q$ 위에서 기약이다 (아이젠슈타인).
\end{theorem}

\begin{formula}[비에트 공식과 로그미분]
$p(x)=a\prod_i(x-r_i)$이면 계수와 근의 기본대칭식은 비에트 관계를 만족한다.
또한
\[
\frac{p'(x)}{p(x)}=\sum_i\frac1{x-r_i}.
\]
근의 역수합 같은 문제에 매우 빠르다.
\end{formula}

\begin{formula}[1의 거듭제곱근]
\[
\omega_n=e^{2\pi i/n},\qquad 1+\omega_n+\cdots+\omega_n^{n-1}=0.
\]
그리고
\[
\prod_{k=1}^{n-1}(1-\omega_n^k)=n.
\]
\end{formula}

\begin{theorem}[아벨--루피니 정리]
5차 이상의 일반 다항방정식에는 사칙연산과 근호만으로 모든 해를 나타내는 보편적 공식이 존재하지 않는다.
``모든 5차방정식이 풀리지 않는다''는 뜻은 아니다.
\end{theorem}

\begin{fact}[사원수]
해밀턴의 사원수 $\Hquat$에서는
\[
i^2=j^2=k^2=ijk=-1,
\qquad ij=k,\quad ji=-k.
\]
대표적인 비가환 나눗셈대수이다.
\end{fact}

\noindent\textbf{대수 문제를 만나면.} 군의 위수와 부분군이면 라그랑주 정리, 군 작용이면 궤도--안정자, 준동형이면 핵과 제1동형정리를 먼저 본다. 다항식은 유리근·켤레근·아이젠슈타인 판정과 비에트 공식을 확인한 뒤 더 무거운 도구로 간다. 근호로 풀 수 있는가를 묻는 순간 카르다노·페라리에서 아벨·갈루아로 이어지는 흐름을 떠올린다.\par\medskip

\finalchapter{10}{Topology}


\begin{formula}[위상수학 기본 표기]
\[
A^\circ=\interior A,\qquad
\overline A=\closure A,\qquad
\partial A=\overline A\setminus A^\circ.
\]
위상동형사상은 연속인 전단사이며 역함수도 연속인 사상이다.
\end{formula}

\begin{fact}[위상동형 불변량]
콤팩트성, 연결성, 연결성분의 개수 등은 위상동형 불변량이다.
$\R$은 비콤팩트이고 $S^1$은 콤팩트이므로 서로 위상동형이 아니다.
\end{fact}

\begin{formula}[오일러 특성]
유한 세포분할에서
\[
\chi=V-E+F.
\]
닫힌 가향곡면 $S_g$에 대해
\[
\chi(S_g)=2-2g.
\]
\end{formula}

\begin{fact}[기본군]
\[
\pi_1(S^1)\cong\Z,
\qquad
\pi_1(S^n)=0\quad(n\ge2),
\qquad
\pi_1(T^2)\cong\Z^2.
\]
원의 감김수, 구면의 단순연결성, 토러스의 두 독립 폐곡선을 연결한다.
\end{fact}

\noindent\textbf{대표적인 위상수학 정리.}
\begin{itemize}
\item \textbf{브라우어 고정점 정리}: 닫힌 공의 연속 자기함수는 고정점을 갖는다.
\item \textbf{털난 공 정리}: $S^2$에는 영점이 없는 연속 접벡터장이 존재하지 않는다.
\item \textbf{보르수크--울람 정리}: $f:S^n\to\R^n$ 연속이면 어떤 $x$에 대해 $f(x)=f(-x)$.
\end{itemize}

\begin{fact}[뫼비우스 띠와 클라인 병]
뫼비우스 띠는 경계가 있는 비가향곡면이고, 클라인 병은 경계가 없는 닫힌 비가향곡면이다.
다각형의 변을 붙여 곡면을 만드는 문제가 나오면 가향성과 경계의 유무를 먼저 본다.
\end{fact}

\begin{fact}[푸앵카레 추측]
닫힌 단순연결 $3$차원 다양체는 $S^3$와 위상동형이다.
그리고리 페렐만 (Grigori Perelman)이 리치 흐름을 이용해 해결했고 2006 필즈상을 거부했다.
\end{fact}

\begin{fact}[서스턴의 여덟 기하]
\[
S^3,\ \R^3,\ \mathbb H^3,\ S^2\times\R,\ \mathbb H^2\times\R,\
\widetilde{\SL_2(\R)},\ \mathrm{Nil},\ \mathrm{Sol}.
\]
3차원 다양체의 기하화에 나타나는 여덟 가지 모형 기하이다.
\end{fact}

\noindent\textbf{위상 문제를 만나면.} 좌표보다 연결성·콤팩트성·경계·오일러 특성·기본군처럼 위상동형으로 보존되는 양을 먼저 찾는다. ``모양이 달라 보여도 같은가''를 묻는 문제에서는 불변량 하나로 구별할 수 있는지부터 본다. 3차원 다양체가 나오면 푸앵카레에서 서스턴·페렐만으로 이어지는 흐름을 연결한다.\par\medskip

\begin{formula}[피보나치 수열]
\[
F_0=0,\quad F_1=1,\quad F_n=\frac{\varphi^n-\psi^n}{\sqrt5},
\]
\[
\varphi=\frac{1+\sqrt5}{2},\qquad \psi=\frac{1-\sqrt5}{2},\qquad
\frac{F_{n+1}}{F_n}\to\varphi.
\]
\end{formula}

\begin{fact}[콘웨이 상수]
읽고 말하기 수열(look-and-say sequence)의 연속한 항 길이 비는 일반적으로
\[
\lambda_{\rm Conway}\approx1.303577269
\]
로 수렴한다. 존 콘웨이와 함께 외울 만한 특수 숫자이다.
\end{fact}

\begin{fact}[리만 제타함수의 핵심 사실]
\[
\zeta(2)=\frac{\pi^2}{6}.
\]
해석적 연속으로
\[
\zeta(-1)=-\frac1{12}.
\]
단 $s=1$에는 극점이 있으며, ``원래 급수가 $-1/12$로 수렴한다''는 뜻이 아니다.
\end{fact}

\noindent\textbf{비에트 점핑.} 대칭적인 정수 조건에서 한 변수를 근으로 보는 이차방정식을 만든다. 다른 정수근을 얻어 더 작은 양의 해로 내려가고 최소성과 모순시키는 것이 기본 구조이다.

\medskip
\noindent\textbf{뫼비우스형 점화식.}
\[
a_{n+1}=\frac{\alpha a_n+\beta}{\gamma a_n+\delta}
\]
꼴에서는 고정점을 찾고, 그 고정점과의 차이의 역수 또는 교차비를 시도한다. 예를 들어
\[
a_{n+1}=\frac1{2-a_n}\quad\Rightarrow\quad b_n=\frac1{1-a_n}
\]
이면 $b_{n+1}=b_n+1$이다.

\begin{formula}[자주 쓰는 부등식]
$a,b,c>0$에서
\[
\frac{a+b+c}{3}\ge(abc)^{1/3}\qquad\text{(산술평균--기하평균)},
\]
\[
(a_1^2+\cdots+a_n^2)(b_1^2+\cdots+b_n^2)\ge(a_1b_1+\cdots+a_nb_n)^2
\qquad\text{(코시--슈바르츠)}.
\]
또한
\[
\frac a{b+c}+\frac b{c+a}+\frac c{a+b}\ge\frac32
\]
(네스빗 부등식).
\end{formula}


\begin{definition}[내시 균형]
다른 플레이어들의 전략을 고정했을 때 어느 한 플레이어도 혼자 전략을 바꾸어 자신의 보수를 높일 수 없는 전략조합이다.
\end{definition}

\begin{algorithm}[$2\times2$ 영합게임]
행 플레이어의 보수행렬이
\[
M=\begin{pmatrix}a&b\\ c&d\end{pmatrix},\qquad \Delta=a-b-c+d\neq0
\]
이고 내부 혼합균형이 존재하면
\[
p=\frac{d-c}{\Delta},\qquad q=\frac{d-b}{\Delta},\qquad v=\frac{ad-bc}{\Delta}.
\]
확률이 $[0,1]$ 안에 있는지 반드시 확인한다.
\end{algorithm}
