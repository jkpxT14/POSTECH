cmake_minimum_required(VERSION 3.16)
project(rpsc_engine VERSION 0.21.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(rpsc_core
    src/orientation.cpp src/position.cpp src/movegen.cpp src/evaluate.cpp
    src/tt.cpp src/search.cpp src/notation.cpp src/perft.cpp src/engine.cpp src/protocol.cpp)
target_include_directories(rpsc_core PUBLIC src)
if(MSVC)
  target_compile_options(rpsc_core PRIVATE /O2 /W4)
else()
  target_compile_options(rpsc_core PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
add_executable(rpsc-engine src/main.cpp)
target_link_libraries(rpsc-engine PRIVATE rpsc_core)
enable_testing()
add_executable(rpsc-engine-tests tests/engine_tests.cpp)
target_link_libraries(rpsc-engine-tests PRIVATE rpsc_core)
add_test(NAME rpsc-engine-tests COMMAND rpsc-engine-tests)
add_executable(rpsc-search-regressions tests/search_regressions.cpp)
target_link_libraries(rpsc-search-regressions PRIVATE rpsc_core)
add_test(NAME rpsc-search-regressions COMMAND rpsc-search-regressions)
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
  add_test(NAME rpsc-browser-format3-regressions
           COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/browser_engine_tests.cjs)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME rpsc-handbook-records
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/handbook_records.py)
endif()
find_program(BASH_EXECUTABLE bash)
if(BASH_EXECUTABLE)
  add_test(NAME rpsc-protocol-school-role-boundary
           COMMAND ${BASH_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_semantics.sh $<TARGET_FILE:rpsc-engine>)
endif()
