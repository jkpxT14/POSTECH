\begin{problem}{1}
수열 $(a_n)$이
\[
    a_0=2,
    \qquad
    a_1=\sqrt2,
    \qquad
    a_{n+2}=\sqrt2\,a_{n+1}-a_n
\]
을 만족한다. $a_{2026}$을 구하시오.
\end{problem}
