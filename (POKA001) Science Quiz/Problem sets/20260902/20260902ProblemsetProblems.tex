\begin{problem}{1}
수열 $(a_n)$이
\[
    a_0=2,
    \qquad
    a_1=\sqrt2,
    \qquad
    a_{n+2}=\sqrt2\,a_{n+1}-a_n
\]
을 만족한다. $a_{2026}$을 구하시오.
\end{problem}

\begin{problem}{2}
\[
    f(x)=\frac{1}{1-x-x^2}
\]
라 하자. $f^{(2026)}(0)$을 Fibonacci sequence를 이용하여 나타내시오.
\end{problem}

\begin{problem}{3}
$\mathbf{J}_n$을 모든 성분이 $1$인 $n\times n$ 행렬이라 하고
\[
    \mathbf{A}=\mathbf{I}_n+\mathbf{J}_n
\]
이라 하자. $\mathbf{A}$의 모든 eigenvalue와 multiplicity를 구하고, $\mathbf{A}^{2026}$을 계산하시오.
\end{problem}
