\begin{problem}{1}
수열 $(a_n)$이
\[
    a_0=2\cos\left(\frac{2\pi}{7}\right),
    \qquad
    a_{n+1}=a_n^2-2
\]
을 만족한다. $a_{2026}$을 구하시오.
\end{problem}

\begin{problem}{2}
$\alpha,\beta,\gamma$가 다항식
\[
    x^3-3x+1
\]
의 세 근일 때, 다음 값을 구하시오.
\[
    \frac{1}{1-\alpha}
    +\frac{1}{1-\beta}
    +\frac{1}{1-\gamma}.
\]
\end{problem}

\begin{problem}{3}
$z\in\C\setminus\{0\}$가
\[
    z+\frac{1}{z}=1
\]
을 만족한다. 다음 값을 구하시오.
\[
    z^{2026}+\frac{1}{z^{2026}}.
\]
\end{problem}
